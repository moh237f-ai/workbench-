    83→/** One PDF through the pipeline: blob → IDB, cover render, store record. Throws on failure (caller purges). */
    84→async function ingestPdf(file: File, id: string, folderId: string | null): Promise<void> {
    85→  await putBlob(id, file);
    86→  const doc = await loadDocFromBlob(file);
    87→  const cover = await pageToDataUrl(doc, 1, 420, 0.72);
    88→  await saveCover(id, cover);
    89→  useWB.getState().addFile({
    90→    id,
    91→    title: file.name.replace(/\.pdf$/i, ''),
    92→    fileName: file.name,
    93→    size: file.size,
    94→    pageCount: doc.numPages,
    95→    folderId,
    96→  });
    97→}

    99→/** Upload pipeline for loose PDF files: store blob → render cover → register file (toast per file). */
   100→async function importPdfFiles(list: File[]): Promise<void> {
   101→  bumpImporting(list.length);
   102→  for (const file of list) {
   103→    const id = uid();
   104→    try {
   105→      await ingestPdf(file, id, null);
   106→      toast('Added to library', file.name.replace(/\.pdf$/i, ''));
   107→    } catch {
   108→      await purgeFileArtifacts(id).catch(() => undefined);
   109→      toast('Import failed', file.name);
   110→    } finally {
   111→      bumpImporting(-1);
   112→    }
   113→  }
   114→}

   116→const IMPORT_FOLDER_CAP = 60;
   117→
   118→/** Import a whole system folder: builds the folder tree from each file's webkitRelativePath
   119→ *  (e.g. "Math/ODE/chapter1.pdf" → Math → ODE), reusing folders by name + parent. */
   120→async function importPdfFolder(list: File[]): Promise<void> {
   121→  const pdfs = list.filter((f) => f.type === 'application/pdf' || /\.pdf$/i.test(f.name));
   122→  const skipped = list.length - pdfs.length;
   123→  if (skipped > 0) {
   124→    toast('Skipped non-PDFs', `${skipped} file${skipped === 1 ? '' : 's'} ignored — only PDFs are imported.`);
   125→  }
   126→  if (!pdfs.length) {
   127→    toast('No PDFs found', 'Pick a folder that contains PDF files.');
   128→    return;
   129→  }
   130→  let work = pdfs;
   131→  if (work.length > IMPORT_FOLDER_CAP) {
   132→    work = work.slice(0, IMPORT_FOLDER_CAP);
   133→    toast('Import capped', `Only the first ${IMPORT_FOLDER_CAP} PDFs will be imported (found ${pdfs.length}).`);
   134→  }
   135→
   136→  /* relative path → folder id cache (create or reuse by name + parent) */
   137→  const cache = new Map<string, string>();
   138→  let created = 0;
   139→  const ensureFolderPath = (segments: string[]): string | null => {
   140→    let parentId: string | null = null;
   141→    let path = '';
   142→    for (const seg of segments) {
   143→      path = path ? `${path}/${seg}` : seg;
   144→      const hit = cache.get(path);
   145→      if (hit) {
   146→        parentId = hit;
   147→        continue;
   148→      }
   149→      const st = useWB.getState();
   150→      const found = st.libraryFolders.find((f) => f.name === seg && f.parentId === parentId);
   151→      const fid = found ? found.id : st.addLibraryFolder(seg, parentId).id;
   152→      if (!found) created += 1;
   153→      cache.set(path, fid);
   154→      parentId = fid;
   155→    }
   156→    return parentId;
   157→  };
   158→
   159→  let imported = 0;
   160→  let failed = 0;
   161→  const targetFolders = new Set<string>();
   162→  bumpImporting(work.length);
   163→  for (const file of work) {
   164→    const id = uid();
   165→    try {
   166→      const segments = file.webkitRelativePath.split('/').filter(Boolean);
   167→      segments.pop(); // strip the file name
   168→      const folderId = segments.length ? ensureFolderPath(segments) : null;
   169→      await ingestPdf(file, id, folderId);
   170→      if (folderId) targetFolders.add(folderId);
   171→      imported += 1;
   172→    } catch {
   173→      failed += 1;
   174→      await purgeFileArtifacts(id).catch(() => undefined);
   175→    } finally {
   176→      bumpImporting(-1);
   177→    }
   178→  }
   179→
   180→  const title = imported
   181→    ? targetFolders.size
   182→      ? `Imported ${imported} ${imported === 1 ? 'file' : 'files'} into ${targetFolders.size} ${
   183→          targetFolders.size === 1 ? 'folder' : 'folders'
   184→        }`
   185→      : `Imported ${imported} ${imported === 1 ? 'file' : 'files'}`
   186→    : 'Import failed';
   187→  const desc: string[] = [];
   188→  if (created) desc.push(`${created} new ${created === 1 ? 'folder' : 'folders'} created`);
   189→  if (failed) desc.push(`${failed} failed`);
   190→  toast(title, desc.length ? desc.join(' · ') : undefined);
   191→}

   298→/** useFileCover plus a "gave up" flag: distinguishes loading (shimmer) from failed (icon tile). */
   299→function useCoverWithFallback(file: ResourceFile): { cover: string | null; failed: boolean } {
   300→  const cover = useFileCover(file);
   301→  const epoch = `${file.id}::${file.coverMode}::${file.coverPage}`;
   302→  const [timedOut, setTimedOut] = useState<{ epoch: string; done: boolean }>({ epoch: '', done: false });
   303→  useEffect(() => {
   304→    const t = setTimeout(() => setTimedOut({ epoch, done: true }), 2400);
   305→    return () => clearTimeout(t);
   306→  }, [epoch]);
   307→  return { cover, failed: !cover && timedOut.epoch === epoch && timedOut.done };
   308→}

   310→/* ---------- folder tree helpers (pure, module scope) ---------- */
   311→
   312→type FolderNode = { folder: LibraryFolder; children: FolderNode[]; depth: number };
   313→
   314→/** Root folders + nested subfolders; orphans (missing parent) are treated as roots. */
   315→function buildFolderTree(folders: LibraryFolder[]): FolderNode[] {
   316→  const byParent = new Map<string | null, LibraryFolder[]>();
   317→  for (const f of folders) {
   318→    const key = f.parentId && folders.some((x) => x.id === f.parentId) ? f.parentId : null;
   319→    const arr = byParent.get(key);
   320→    if (arr) arr.push(f);
   321→    else byParent.set(key, [f]);
   322→  }
   323→  const make = (parent: string | null, depth: number): FolderNode[] =>
   324→    (byParent.get(parent) ?? []).map((folder) => ({ folder, children: make(folder.id, depth + 1), depth }));
   325→  return make(null, 0);
   326→}
   327→
   328→/** folderId → set of itself + all descendants (for counts + cycle guards). */
   329→function selfAndDescendants(folders: LibraryFolder[]): Map<string, Set<string>> {
   330→  const childrenOf = new Map<string, LibraryFolder[]>();
   331→  for (const f of folders) {
   332→    if (!f.parentId || !folders.some((x) => x.id === f.parentId)) continue;
   333→    const arr = childrenOf.get(f.parentId);
   334→    if (arr) arr.push(f);
   335→    else childrenOf.set(f.parentId, [f]);
   336→  }
   337→  const memo = new Map<string, Set<string>>();
   338→  const collect = (id: string): Set<string> => {
   339→    const cached = memo.get(id);
   340→    if (cached) return cached;
   341→    const set = new Set<string>([id]);
   342→    memo.set(id, set); /* seed before recursion — breaks any (impossible-by-store) parent cycle */
   343→    for (const child of childrenOf.get(id) ?? []) {
   344→      for (const d of collect(child.id)) set.add(d);
   345→    }
   346→    return set;
   347→  };
   348→  for (const f of folders) collect(f.id);
   349→  return memo;
   350→}

   352→/** Ancestor chain for a folder id (root first). Empty when unknown. */
   353→function folderPathOf(folders: LibraryFolder[], id: string): LibraryFolder[] {
   354→  const map = new Map(folders.map((f) => [f.id, f]));
   355→  const path: LibraryFolder[] = [];
   356→  const seen = new Set<string>();
   357→  let cur = map.get(id);
   358→  while (cur && !seen.has(cur.id)) {
   359→    seen.add(cur.id);
   360→    path.unshift(cur);
   361→    cur = cur.parentId ? map.get(cur.parentId) : undefined;
   362→  }
   363→  return path;
   364→}

   434→/* ---------- filter + sort ---------- */
   435→
   436→function matchesFilter(f: ResourceFile, filter: LibFilter): boolean {
   437→  if (filter === 'archived') return f.archived;
   438→  if (f.archived) return false;
   439→  const pct = wbFileProgress(f);
   440→  if (filter === 'reading') return pct > 0 && pct < 100;
   441→  if (filter === 'unread') return pct === 0;
   442→  if (filter === 'deadline') return !!f.deadline && (wbFileDeadlineStatus(f) !== 'ok' || pct < 100);
   443→  return true;
   444→}

   446→/** Folder filter: 'all' | 'unfiled' | folder id (files directly in that folder only). */
   447→function matchesFolder(f: ResourceFile, folderFilter: string): boolean {
   448→  if (folderFilter === 'all') return true;
   449→  if (folderFilter === 'unfiled') return !f.folderId;
   450→  return f.folderId === folderFilter;
   451→}

   453→/** Search: case-insensitive substring over title, fileName and tags. */
   454→function matchesSearch(f: ResourceFile, q: string): boolean {
   455→  if (!q) return true;
   456→  return (
   457→    f.title.toLowerCase().includes(q) ||
   458→    f.fileName.toLowerCase().includes(q) ||
   459→    f.tags.some((t) => t.toLowerCase().includes(q))
   460→  );
   461→}

   463→function sortFiles(list: ResourceFile[], sort: LibSort): ResourceFile[] {
   464→  const arr = [...list];
   465→  if (sort === 'recent') {
   466→    arr.sort((a, b) => (b.lastReadAt || b.addedAt).localeCompare(a.lastReadAt || a.addedAt));
   467→  } else if (sort === 'deadline') {
   468→    arr.sort((a, b) => {
   469→      if (!a.deadline || !b.deadline) return a.deadline ? -1 : b.deadline ? 1 : 0;
   470→      return a.deadline.localeCompare(b.deadline);
   471→    });
   472→  } else if (sort === 'progress') {
   473→    arr.sort((a, b) => wbFileProgress(b) - wbFileProgress(a));
   474→  } else {
   475→    arr.sort((a, b) => a.title.localeCompare(b.title));
   476→  }
   477→  return arr;
   478→}

   537→

   544→      if (!prev.has(id)) return prev;
   545→      const n = new Set(prev);
   546→      n.delete(id);
   547→      return n;
   548→    });
   549→  };

   554→      if (willOpen === prev.has(id)) return prev;

   556→      if (willOpen) n.add(id);
   557→      else n.delete(id);
   558→      return n;
   559→    });
   560→    setCollapsedIds((prev) => {
   561→      if (!willOpen === prev.has(id)) return prev;
   562→      const n = new Set(prev);
   563→      if (willOpen) n.delete(id);
   564→      else n.add(id);
   565→      return n;
   566→    });
   567→  };
   568→
   569→  /* icon picker popover: fixed + viewport-clamped at open time, closes on outside click / Esc */
   570→  useEffect(() => {
   571→    if (!iconPop) return;
   572→    const onDown = (e: MouseEvent) => {
   573→      const t = e.target as HTMLElement;

   579→        e.stopPropagation();
   580→        setIconPop(null);
   581→      }

   584→    document.addEventListener('keydown', onKey, true);

   586→      document.removeEventListener('mousedown', onDown);
   587→      document.removeEventListener('keydown', onKey, true);

   665→    e.dataTransfer.setData('text/plain', folder.id);
   666→    try {
   667→      e.dataTransfer.setData('application/x-wb-folder', folder.id);

   670→    }
   671→    e.dataTransfer.effectAllowed = 'move';
   672→  };

   674→  const handleDrop = (e: React.DragEvent, rowId: string) => {
   675→    if (!Array.from(e.dataTransfer.types).includes('text/plain')) return;

   738→          }}
   739→          onKeyDown={(e) => {

   741→            rowKey(e, () => setFolderFilter(folder.id));
   742→          }}
   743→          onDoubleClick={() => startRename(folder)}
   744→          draggable={renameId !== folder.id}
   745→          onDragStart={(e) => onRowDragStart(e, folder)}
   746→          onDragOver={(e) => onRowDragOver(e, folder.id)}
   747→          onDragLeave={() => onRowDragLeave(folder.id)}

   749→        >
   750→          {hasKids ? (
   751→            <button
   752→              type="button"
   753→              className={`lib-tree-caret ${isOpen ? 'open' : ''}`}
   754→              aria-label={isOpen ? `Collapse ${folder.name}` : `Expand ${folder.name}`}
   755→              onClick={(e) => {
   756→                e.stopPropagation();
   757→                toggleExpand(folder.id);
   758→              }}
   759→            >
   760→              <ChevronRight size={12} />
   761→            </button>
   762→          ) : (
   763→            <span className="lib-tree-caret spacer" aria-hidden="true" />
   764→          )}
   765→          <Icon size={13} className="lib-tree-ico" aria-hidden="true" />
   766→          <span
   767→            className="lib-tree-dot"
   768→            style={{ backgroundColor: folder.color }}
   769→            role="button"
   770→            tabIndex={0}
   771→            title={`Cycle color (currently ${folder.color})`}
   772→            aria-label={`Change color of folder ${folder.name}`}
   773→            onClick={(e) => {
   774→              e.stopPropagation();
   775→              cycleColor(folder);
   776→            }}
   777→            onDoubleClick={(e) => e.stopPropagation()}
   778→            onKeyDown={(e) => {
   779→              if (e.key === 'Enter' || e.key === ' ') {
   780→                e.preventDefault();
   781→                e.stopPropagation();
   782→                cycleColor(folder);
   783→              }
   784→            }}
   785→          />
   786→          {renameId === folder.id ? (
   787→            <input
   788→              className="lib-folder-input"
   789→              value={renameText}
   790→              autoFocus
   791→              aria-label={`Rename folder ${folder.name}`}
   792→              onChange={(e) => setRenameText(e.target.value)}

   799→              onBlur={commitRename}

   805→          <span className="lib-tree-acts">
   806→            <button
   807→              type="button"
   808→              className="lib-tree-act"
   809→              data-ico-btn
   810→              title={`New subfolder in ${folder.name}`}
   811→              aria-label={`Add subfolder to ${folder.name}`}

   813→                e.stopPropagation();

   815→                setCreateParent(folder.id);
   816→                setCreateText('');
   817→                forceExpand(folder.id);
   818→              }}

   820→              <Plus size={12} />

   822→            <button
   823→              type="button"
   824→              className="lib-tree-act"
   825→              data-ico-btn
   826→              title="Pick icon"
   827→              aria-label={`Pick icon for ${folder.name}`}

   829→                e.stopPropagation();
   830→                openIconPop(e, folder);
   831→              }}
   832→            >
   833→              <Palette size={12} />
   834→            </button>
   835→            <button
   836→              type="button"
   837→              className="lib-tree-act danger"
   838→              title={`Delete folder "${folder.name}"`}
   839→              aria-label={`Delete folder ${folder.name}`}
   840→              onClick={(e) => {
   841→                e.stopPropagation();
   842→                removeFolder(folder);
   843→              }}
   844→            >
   845→              <X size={11} />
   846→            </button>
   847→          </span>
   848→        </div>
   849→        {createParent === folder.id && (
   850→          <div className="lib-subrow" style={{ ['--lib-d' as string]: depth + 1 }}>
   851→            <input
   852→              className="lib-folder-input lib-tree-new"
   853→              value={createText}
   854→              autoFocus
   855→              placeholder="Subfolder name…"
   856→              aria-label="New subfolder name"
   857→              onChange={(e) => setCreateText(e.target.value)}
   858→              onKeyDown={(e) => {
   859→                if (e.key === 'Enter') finishCreate(true);
   860→                else if (e.key === 'Escape') {
   861→                  setCreateParent(null);
   862→                  setCreateText('');
   863→                }
   864→              }}
   865→              onBlur={() => finishCreate(false)}
   866→            />
   867→          </div>
   868→        )}
   869→        {isOpen && children.map(renderRow)}
   870→      </React.Fragment>
   871→    );
   872→  };
   873→
   874→  return (
   875→    <aside className={`lib-tree ${treeOpen ? '' : 'closed'}`} aria-label="Library folder tree">
   876→      <div className="lib-tree-head">
   877→        <span className="lib-tree-title">
   878→          <ListTree size={13} aria-hidden="true" /> Folders
   879→        </span>
   880→        <button
   881→          type="button"
   882→          className="lib-tree-fold"
   883→          onClick={() => setTreeOpen((v) => !v)}
   884→          aria-expanded={treeOpen}
   885→          aria-label={treeOpen ? 'Collapse folder tree' : 'Expand folder tree'}
   886→          title={treeOpen ? 'Collapse folder tree' : 'Expand folder tree'}
   887→        >
   888→          <ChevronDown size={14} />
   889→        </button>
   890→      </div>
   891→      {treeOpen && (
   892→        <div className="lib-tree-body" role="tree" aria-label="Folders">
   893→          <div
   894→            role="treeitem"
   895→            aria-selected={folderFilter === 'all'}
   896→            tabIndex={0}
   897→            className={`lib-tree-row vroot ${folderFilter === 'all' ? 'active' : ''} ${
   898→              dropId === 'all' ? 'drop' : ''
   899→            }`}
   900→            title="All files — drop here to unfile"
   901→            onClick={() => setFolderFilter('all')}
   902→            onKeyDown={(e) => rowKey(e, () => setFolderFilter('all'))}
   903→            onDragOver={(e) => onRowDragOver(e, 'all')}
   904→            onDragLeave={() => onRowDragLeave('all')}
   905→            onDrop={(e) => handleDrop(e, 'all')}
   906→          >
   907→            <span className="lib-tree-caret spacer" aria-hidden="true" />
   908→            <LibraryIco size={13} className="lib-tree-ico" aria-hidden="true" />

   912→          <div
   913→            role="treeitem"
   914→            aria-selected={folderFilter === 'unfiled'}
   915→            tabIndex={0}

   917→              dropId === 'unfiled' ? 'drop' : ''
   918→            }`}
   919→            title="Files not in any folder — drop a file here to unfile it"
   920→            onClick={() => setFolderFilter('unfiled')}

   924→            onDrop={(e) => handleDrop(e, 'unfiled')}

   926→            <span className="lib-tree-caret spacer" aria-hidden="true" />
   927→            <Inbox size={13} className="lib-tree-ico" aria-hidden="true" />

   929→            <span className="lib-folder-count">{unfiledCount}</span>

   931→          {tree.map(renderRow)}
   932→          {createParent === 'root' ? (
   933→            <div className="lib-subrow" style={{ ['--lib-d' as string]: 0 }}>
   934→              <input
   935→                className="lib-folder-input lib-tree-new"
   936→                value={createText}
   937→                autoFocus
   938→                placeholder="Folder name…"
   939→                aria-label="New folder name"
   940→                onChange={(e) => setCreateText(e.target.value)}
   941→                onKeyDown={(e) => {
   942→                  if (e.key === 'Enter') finishCreate(true);
   943→                  else if (e.key === 'Escape') {
   944→                    setCreateParent(null);
   945→                    setCreateText('');
   946→                  }
   947→                }}
   948→                onBlur={() => finishCreate(false)}
   949→              />
   950→            </div>
   951→          ) : (
   952→            <button
   953→              type="button"
   954→              className="lib-newfolder-btn"
   955→              onClick={() => {
   956→                setRenameId(null);
   957→                setIconPop(null);
   958→                setCreateParent('root');
   959→              }}
   960→            >
   961→              <FolderPlus size={13} /> New folder
   962→            </button>
   963→          )}
   964→        </div>
   965→      )}
   966→      {iconPop &&
   967→        createPortal(
   968→          <div
   969→            className="lib-icon-pop"
   970→            style={{ top: iconPop.top, left: iconPop.left }}
   971→            role="dialog"
   972→            aria-label="Pick a folder icon"
   973→          >
   974→            {FOLDER_ICONS.map((key) => {
   975→              const Ico = folderIconOf(key);
   976→              const sel = folders.find((f) => f.id === iconPop.id)?.icon === key;
   977→              return (
   978→                <button
   979→                  key={key}
   980→                  type="button"
   981→                  className={`lib-icon-cell ${sel ? 'sel' : ''}`}
   982→                  title={key}
   983→                  aria-label={`Icon: ${key}`}
   984→                  aria-pressed={sel}
   985→                  onClick={() => {
   986→                    setFolderIcon(iconPop.id, key);
   987→                    setIconPop(null);
   988→                  }}
   989→                >
   990→                  <Ico size={15} />
   991→                </button>
   992→              );
   993→            })}
   994→          </div>,
   995→          document.body
   996→        )}
   997→    </aside>
   998→  );
   999→}

  1039→  };
  1040→
  1041→  /* the whole card is a drag source (its file id as text/plain) — but never from form controls */
  1042→  const onCardDragStart = (e: React.DragEvent) => {
  1043→    const t = e.target as HTMLElement;
  1044→    if (t.closest('input, textarea, select, button, a, label')) {
  1045→      e.preventDefault();
  1046→      return;

  1050→    setDragging(true);
  1051→  };

  1054→    if (!confirm(`Delete "${file.title}"? Its PDF, cover and reading history will be removed.`)) return;
  1055→    void purgeFileArtifacts(file.id).catch(() => undefined);
  1056→    deleteFile(file.id);

  1064→      pct >= 100 ? 'done'
  1065→        : st === 'over' ? `${Math.abs(du)}d overdue`
  1066→          : du === 0 ? 'due today'
