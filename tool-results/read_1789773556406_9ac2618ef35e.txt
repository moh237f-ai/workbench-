    70→type RGB = [number, number, number];
    71→
    72→function mixRgb(a: RGB, b: RGB, t: number): RGB {
    73→  const c = (i: number): number => Math.round(a[i] + (b[i] - a[i]) * t);
    74→  return [c(0), c(1), c(2)];
    75→}
    76→
    77→function rgbCss(c: RGB): string {
    78→  return `rgb(${c[0]}, ${c[1]}, ${c[2]})`;
    79→}
    80→
    81→function clamp01v(v: number): number {
    82→  return v < 0 ? 0 : v > 1 ? 1 : v;
    83→}

    85→/** Tint for the surface behind the canvas so themed pages have no white glare. */
    86→function pageSurfaceBg(mode: ReaderThemeMode, custom: { brightness: number; sepia: number }): string {
    87→  const white: RGB = [255, 255, 255];
    88→  switch (mode) {
    89→    case 'dark':
    90→      return '#15181a';
    91→    case 'warm':
    92→      return '#f3ecdc';
    93→    case 'custom': {
    94→      let c = mixRgb(white, [0xef, 0xe3, 0xcd], clamp01v(custom.sepia / 100));
    95→      const dim = clamp01v((100 - custom.brightness) / 100);
    96→      c = mixRgb(c, [0x16, 0x19, 0x18], dim * 0.9);
    97→      return rgbCss(c);
    98→    }
    99→    default:
   100→      return '#ffffff';
   101→  }
   102→}
   103→
   104→function clampPage(p: number, total: number): number {
   105→  const max = Math.max(1, total);
   106→  const n = Math.round(p);
   107→  return Math.max(1, Math.min(max, Number.isFinite(n) ? n : 1));
   108→}

   110→function fmtRem(secs: number): string {
   111→  const s = Math.max(0, Math.floor(secs));
   112→  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`;
   113→}
   114→
   115→function themeFilterOf(mode: ReaderThemeMode, custom: { brightness: number; contrast: number; sepia: number }): string {
   116→  switch (mode) {
   117→    case 'dark':
   118→      return 'invert(1) hue-rotate(180deg) brightness(0.92) contrast(1.06)';
   119→    case 'warm':
   120→      return 'sepia(0.45) brightness(1.05)';
   121→    case 'custom':
   122→      return `brightness(${custom.brightness / 100}) contrast(${custom.contrast / 100}) sepia(${custom.sepia / 100})`;
   123→    default:
   124→      return 'none';
   125→  }
   126→}
