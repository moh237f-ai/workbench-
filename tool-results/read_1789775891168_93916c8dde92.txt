    83→/** One PDF through the pipeline: blob → IDB, cover render, store record. Throws on failure (caller purges). */
    84→async function ingestPdf(file: File, id: string, folderId: string | null): Promise<void> {
    85→  await putBlob(id, file);
    86→  const doc = await loadDocFromBlob(file);
    87→  const cover = await pageToDataUrl(doc, 1, 420, 0.72);
    88→  await saveCover(id, cover);
    89→  useWB.getState().addFile({
    90→    id,
    91→    title: file.name.replace(/\.pdf$/i, ''),
    92→    fileName: file.name,
    93→    size: file.size,
    94→    pageCount: doc.numPages,
    95→    folderId,
    96→  });
    97→}

    99→/** Upload pipeline for loose PDF files: store blob → render cover → register file (toast per file). */
   100→async function importPdfFiles(list: File[]): Promise<void> {
   101→  bumpImporting(list.length);
   102→  for (const file of list) {
   103→    const id = uid();
   104→    try {
   105→      await ingestPdf(file, id, null);
   106→      toast('Added to library', file.name.replace(/\.pdf$/i, ''));
   107→    } catch {
   108→      await purgeFileArtifacts(id).catch(() => undefined);
   109→      toast('Import failed', file.name);
   110→    } finally {
   111→      bumpImporting(-1);
   112→    }
   113→  }
   114→}

   116→const IMPORT_FOLDER_CAP = 60;
   117→
   118→/** Import a whole system folder: builds the folder tree from each file's webkitRelativePath
   119→ *  (e.g. "Math/ODE/chapter1.pdf" → Math → ODE), reusing folders by name + parent. */
   120→async function importPdfFolder(list: File[]): Promise<void> {
   121→  const pdfs = list.filter((f) => f.type === 'application/pdf' || /\.pdf$/i.test(f.name));
   122→  const skipped = list.length - pdfs.length;
   123→  if (skipped > 0) {
   124→    toast('Skipped non-PDFs', `${skipped} file${skipped === 1 ? '' : 's'} ignored — only PDFs are imported.`);
   125→  }
   126→  if (!pdfs.length) {
   127→    toast('No PDFs found', 'Pick a folder that contains PDF files.');
   128→    return;
   129→  }
   130→  let work = pdfs;
   131→  if (work.length > IMPORT_FOLDER_CAP) {
   132→    work = work.slice(0, IMPORT_FOLDER_CAP);
   133→    toast('Import capped', `Only the first ${IMPORT_FOLDER_CAP} PDFs will be imported (found ${pdfs.length}).`);
   134→  }
   135→
   136→  /* relative path → folder id cache (create or reuse by name + parent) */
   137→  const cache = new Map<string, string>();
   138→  let created = 0;
   139→  const ensureFolderPath = (segments: string[]): string | null => {
   140→    let parentId: string | null = null;
   141→    let path = '';
   142→    for (const seg of segments) {
   143→      path = path ? `${path}/${seg}` : seg;
   144→      const hit = cache.get(path);
   145→      if (hit) {
   146→        parentId = hit;
   147→        continue;
   148→      }
   149→      const st = useWB.getState();
   150→      const found = st.libraryFolders.find((f) => f.name === seg && f.parentId === parentId);
   151→      const fid = found ? found.id : st.addLibraryFolder(seg, parentId).id;
   152→      if (!found) created += 1;
   153→      cache.set(path, fid);
   154→      parentId = fid;
   155→    }
   156→    return parentId;
   157→  };
   158→
   159→  let imported = 0;
   160→  let failed = 0;
   161→  const targetFolders = new Set<string>();
   162→  bumpImporting(work.length);
   163→  for (const file of work) {
   164→    const id = uid();
   165→    try {
   166→      const segments = file.webkitRelativePath.split('/').filter(Boolean);
   167→      segments.pop(); // strip the file name
   168→      const folderId = segments.length ? ensureFolderPath(segments) : null;
   169→      await ingestPdf(file, id, folderId);
   170→      if (folderId) targetFolders.add(folderId);
   171→      imported += 1;
   172→    } catch {
   173→      failed += 1;
   174→      await purgeFileArtifacts(id).catch(() => undefined);
   175→    } finally {
   176→      bumpImporting(-1);
   177→    }
   178→  }
   179→
   180→  const title = imported
   181→    ? targetFolders.size
   182→      ? `Imported ${imported} ${imported === 1 ? 'file' : 'files'} into ${targetFolders.size} ${
   183→          targetFolders.size === 1 ? 'folder' : 'folders'
   184→        }`
   185→      : `Imported ${imported} ${imported === 1 ? 'file' : 'files'}`
   186→    : 'Import failed';
   187→  const desc: string[] = [];
   188→  if (created) desc.push(`${created} new ${created === 1 ? 'folder' : 'folders'} created`);
   189→  if (failed) desc.push(`${failed} failed`);
   190→  toast(title, desc.length ? desc.join(' · ') : undefined);
   191→}

   298→/** useFileCover plus a "gave up" flag: distinguishes loading (shimmer) from failed (icon tile). */
   299→function useCoverWithFallback(file: ResourceFile): { cover: string | null; failed: boolean } {
   300→  const cover = useFileCover(file);
   301→  const epoch = `${file.id}::${file.coverMode}::${file.coverPage}`;
   302→  const [timedOut, setTimedOut] = useState<{ epoch: string; done: boolean }>({ epoch: '', done: false });
   303→  useEffect(() => {
   304→    const t = setTimeout(() => setTimedOut({ epoch, done: true }), 2400);
   305→    return () => clearTimeout(t);
   306→  }, [epoch]);
   307→  return { cover, failed: !cover && timedOut.epoch === epoch && timedOut.done };
   308→}

   310→/* ---------- folder tree helpers (pure, module scope) ---------- */
   311→
   312→type FolderNode = { folder: LibraryFolder; children: FolderNode[]; depth: number };
   313→
   314→/** Root folders + nested subfolders; orphans (missing parent) are treated as roots. */
   315→function buildFolderTree(folders: LibraryFolder[]): FolderNode[] {
   316→  const byParent = new Map<string | null, LibraryFolder[]>();
   317→  for (const f of folders) {
   318→    const key = f.parentId && folders.some((x) => x.id === f.parentId) ? f.parentId : null;
   319→    const arr = byParent.get(key);
   320→    if (arr) arr.push(f);
   321→    else byParent.set(key, [f]);
   322→  }
   323→  const make = (parent: string | null, depth: number): FolderNode[] =>
   324→    (byParent.get(parent) ?? []).map((folder) => ({ folder, children: make(folder.id, depth + 1), depth }));
   325→  return make(null, 0);
   326→}
   327→
   328→/** folderId → set of itself + all descendants (for counts + cycle guards). */
   329→function selfAndDescendants(folders: LibraryFolder[]): Map<string, Set<string>> {
   330→  const childrenOf = new Map<string, LibraryFolder[]>();
   331→  for (const f of folders) {
   332→    if (!f.parentId || !folders.some((x) => x.id === f.parentId)) continue;
   333→    const arr = childrenOf.get(f.parentId);
   334→    if (arr) arr.push(f);
   335→    else childrenOf.set(f.parentId, [f]);
   336→  }
   337→  const memo = new Map<string, Set<string>>();
   338→  const collect = (id: string): Set<string> => {
   339→    const cached = memo.get(id);
   340→    if (cached) return cached;
   341→    const set = new Set<string>([id]);
   342→    memo.set(id, set); /* seed before recursion — breaks any (impossible-by-store) parent cycle */
   343→    for (const child of childrenOf.get(id) ?? []) {
   344→      for (const d of collect(child.id)) set.add(d);
   345→    }
   346→    return set;
   347→  };
   348→  for (const f of folders) collect(f.id);
   349→  return memo;
   350→}

   352→/** Ancestor chain for a folder id (root first). Empty when unknown. */
   353→function folderPathOf(folders: LibraryFolder[], id: string): LibraryFolder[] {
   354→  const map = new Map(folders.map((f) => [f.id, f]));
   355→  const path: LibraryFolder[] = [];
   356→  const seen = new Set<string>();
   357→  let cur = map.get(id);
   358→  while (cur && !seen.has(cur.id)) {
   359→    seen.add(cur.id);
   360→    path.unshift(cur);
   361→    cur = cur.parentId ? map.get(cur.parentId) : undefined;
   362→  }
   363→  return path;
   364→}

   366→/** Folder list in tree order (pre-order) with depth-prefixed labels — for selects that mirror the tree. */
   367→function flattenFolderOptions(folders: LibraryFolder[]): { id: string; label: string }[] {
   368→  const out: { id: string; label: string }[] = [];
   369→  const walk = (nodes: FolderNode[]): void => {
   370→    for (const n of nodes) {
   371→      out.push({ id: n.folder.id, label: `${'— '.repeat(Math.min(n.depth, 6))}${n.folder.name}` });
   372→      walk(n.children);
   373→    }
   374→  };
   375→  walk(buildFolderTree(folders));
   376→  return out;
   377→}

   449→function matchesFilter(f: ResourceFile, filter: LibFilter): boolean {
   450→  if (filter === 'archived') return f.archived;
   451→  if (f.archived) return false;
   452→  const pct = wbFileProgress(f);
   453→  if (filter === 'reading') return pct > 0 && pct < 100;
   454→  if (filter === 'unread') return pct === 0;
   455→  if (filter === 'deadline') return !!f.deadline && (wbFileDeadlineStatus(f) !== 'ok' || pct < 100);
   456→  return true;
   457→}

   459→/** Folder filter: 'all' | 'unfiled' | folder id (files directly in that folder only). */
   460→function matchesFolder(f: ResourceFile, folderFilter: string): boolean {
   461→  if (folderFilter === 'all') return true;
   462→  if (folderFilter === 'unfiled') return !f.folderId;
   463→  return f.folderId === folderFilter;
   464→}

   466→/** Search: case-insensitive substring over title, fileName and tags. */
   467→function matchesSearch(f: ResourceFile, q: string): boolean {
   468→  if (!q) return true;
   469→  return (
   470→    f.title.toLowerCase().includes(q) ||
   471→    f.fileName.toLowerCase().includes(q) ||
   472→    f.tags.some((t) => t.toLowerCase().includes(q))
   473→  );
   474→}

   476→function sortFiles(list: ResourceFile[], sort: LibSort): ResourceFile[] {
   477→  const arr = [...list];
   478→  if (sort === 'recent') {
   479→    arr.sort((a, b) => (b.lastReadAt || b.addedAt).localeCompare(a.lastReadAt || a.addedAt));
   480→  } else if (sort === 'deadline') {
   481→    arr.sort((a, b) => {
   482→      if (!a.deadline || !b.deadline) return a.deadline ? -1 : b.deadline ? 1 : 0;
   483→      return a.deadline.localeCompare(b.deadline);
   484→    });
   485→  } else if (sort === 'progress') {
   486→    arr.sort((a, b) => wbFileProgress(b) - wbFileProgress(a));
   487→  } else {
   488→    arr.sort((a, b) => a.title.localeCompare(b.title));
   489→  }
   490→  return arr;
   491→}

   532→      for (const id of descendants.get(folder.id) ?? [folder.id]) n += direct.get(id) ?? 0;
   533→      map.set(folder.id, n);
   534→    }
   535→    return map;
   536→  }, [files, folders, descendants]);
   537→
   538→  const unfiledCount = files.filter((f) => !f.folderId).length;
   539→
   540→  /* ancestors of the filtered folder are always visible (auto-expand, user can still collapse) */
   541→  const filterPath = useMemo(
   542→    () => (folderFilter === 'all' || folderFilter === 'unfiled' ? [] : folderPathOf(folders, folderFilter)),
   543→    [folders, folderFilter]
   544→  );

   548→    return s;
   549→  }, [filterPath]);
   550→

   557→      if (!prev.has(id)) return prev;
   558→      const n = new Set(prev);
   559→      n.delete(id);
   560→      return n;
   561→    });
   562→  };
   563→
   564→  const toggleExpand = (id: string) => {

   566→    setExpanded((prev) => {
   567→      if (willOpen === prev.has(id)) return prev;
   568→      const n = new Set(prev);
   569→      if (willOpen) n.add(id);
   570→      else n.delete(id);
   571→      return n;
   572→    });
   573→    setCollapsedIds((prev) => {
   574→      if (!willOpen === prev.has(id)) return prev;
   575→      const n = new Set(prev);
   576→      if (willOpen) n.delete(id);
   577→      else n.add(id);
   578→      return n;
   579→    });
   580→  };
   581→
   582→  /* icon picker popover: fixed + viewport-clamped at open time, closes on outside click / Esc */
   583→  useEffect(() => {
   584→    if (!iconPop) return;
   585→    const onDown = (e: MouseEvent) => {
   586→      const t = e.target as HTMLElement;
   587→      if (t.closest('.lib-icon-pop') || t.closest('[data-ico-btn]')) return;
   588→      setIconPop(null);
   589→    };

   594→      }
   595→    };

   597→    document.addEventListener('keydown', onKey, true);
   598→    return () => {
   599→      document.removeEventListener('mousedown', onDown);
   600→      document.removeEventListener('keydown', onKey, true);

   658→
   659→  /* drag & drop: file cards carry text/plain = file id; folder rows carry their folder id */
   660→  const onRowDragOver = (e: React.DragEvent, id: string) => {
   661→    const types = Array.from(e.dataTransfer.types);
   662→    if (!types.includes('text/plain')) return;

   664→    if (id === 'unfiled' && types.includes('application/x-wb-folder')) return;
   665→    e.preventDefault();

   678→      e.preventDefault();
   679→      return;

   681→    e.dataTransfer.setData('text/plain', folder.id);
   682→    try {
   683→      e.dataTransfer.setData('application/x-wb-folder', folder.id);
   684→    } catch {
   685→      /* some engines reject custom types — the text/plain payload covers it */
   686→    }
   687→    e.dataTransfer.effectAllowed = 'move';

   690→  const handleDrop = (e: React.DragEvent, rowId: string) => {
   691→    if (!Array.from(e.dataTransfer.types).includes('text/plain')) return;
   692→    e.preventDefault();
   693→    e.stopPropagation();
   694→    setDropId(null);
   695→    const fid = e.dataTransfer.getData('text/plain');
   696→    if (!fid) return;
   697→    const isFile = files.some((f) => f.id === fid);
   698→    const isFolder = !isFile && folders.some((f) => f.id === fid);

   716→      if (targetId) {
   717→        forceExpand(targetId);
   718→        toast('Folder moved', `${from} → ${folders.find((x) => x.id === targetId)?.name ?? ''}`);
   719→      } else {

   729→      e.preventDefault();
   730→      fn();
   731→    }
   732→  };

   743→          role="treeitem"
   744→          aria-selected={folderFilter === folder.id}
   745→          aria-level={depth + 1}
   746→          aria-expanded={hasKids ? isOpen : undefined}

   748→          className={`lib-tree-row ${folderFilter === folder.id ? 'active' : ''} ${
   749→            dropId === folder.id ? 'drop' : ''
   750→          } ${renameId === folder.id ? 'renaming' : ''}`}
   751→          style={{ ['--lib-d' as string]: depth }}
   752→          title={`${n} ${n === 1 ? 'file' : 'files'} (incl. subfolders) · double-click to rename`}
   753→          onClick={() => {
   754→            if (renameId !== folder.id) setFolderFilter(folder.id);
   755→          }}
   756→          onKeyDown={(e) => {
   757→            if (renameId === folder.id) return;
   758→            rowKey(e, () => setFolderFilter(folder.id));
   759→          }}
   760→          onDoubleClick={() => startRename(folder)}
   761→          draggable={renameId !== folder.id}
   762→          onDragStart={(e) => onRowDragStart(e, folder)}
   763→          onDragOver={(e) => onRowDragOver(e, folder.id)}
   764→          onDragLeave={() => onRowDragLeave(folder.id)}
   765→          onDrop={(e) => handleDrop(e, folder.id)}
   766→        >
   767→          {hasKids ? (
   768→            <button
   769→              type="button"
   770→              className={`lib-tree-caret ${isOpen ? 'open' : ''}`}
   771→              aria-label={isOpen ? `Collapse ${folder.name}` : `Expand ${folder.name}`}
   772→              onClick={(e) => {
   773→                e.stopPropagation();
   774→                toggleExpand(folder.id);
   775→              }}
   776→            >
   777→              <ChevronRight size={12} />
   778→            </button>
   779→          ) : (
   780→            <span className="lib-tree-caret spacer" aria-hidden="true" />
   781→          )}
   782→          <Icon size={13} className="lib-tree-ico" aria-hidden="true" />
   783→          <span
   784→            className="lib-tree-dot"
   785→            style={{ backgroundColor: folder.color }}
   786→            role="button"
   787→            tabIndex={0}
   788→            title={`Cycle color (currently ${folder.color})`}
   789→            aria-label={`Change color of folder ${folder.name}`}
   790→            onClick={(e) => {
   791→              e.stopPropagation();
   792→              cycleColor(folder);
   793→            }}
   794→            onDoubleClick={(e) => e.stopPropagation()}
   795→            onKeyDown={(e) => {
   796→              if (e.key === 'Enter' || e.key === ' ') {
   797→                e.preventDefault();
   798→                e.stopPropagation();
   799→                cycleColor(folder);

   814→                else if (e.key === 'Escape') setRenameId(null);
   815→              }}
   816→              onBlur={commitRename}

   824→              type="button"
   825→              className="lib-tree-act"
   826→              data-ico-btn
   827→              title={`New subfolder in ${folder.name}`}
   828→              aria-label={`Add subfolder to ${folder.name}`}

   830→                e.stopPropagation();
   831→                setRenameId(null);
   832→                setCreateParent(folder.id);
   833→                setCreateText('');
   834→                forceExpand(folder.id);

   840→              type="button"

   846→                e.stopPropagation();
   847→                openIconPop(e, folder);
   848→              }}
   849→            >
   850→              <Palette size={12} />
   851→            </button>
   852→            <button
   853→              type="button"
   854→              className="lib-tree-act danger"
   855→              title={`Delete folder "${folder.name}"`}
   856→              aria-label={`Delete folder ${folder.name}`}
   857→              onClick={(e) => {
   858→                e.stopPropagation();
   859→                removeFolder(folder);
   860→              }}
   861→            >
   862→              <X size={11} />
   863→            </button>
   864→          </span>
   865→        </div>
   866→        {createParent === folder.id && (
   867→          <div className="lib-subrow" style={{ ['--lib-d' as string]: depth + 1 }}>
   868→            <input
   869→              className="lib-folder-input lib-tree-new"
   870→              value={createText}
   871→              autoFocus
   872→              placeholder="Subfolder name…"
   873→              aria-label="New subfolder name"
   874→              onChange={(e) => setCreateText(e.target.value)}
   875→              onKeyDown={(e) => {
   876→                if (e.key === 'Enter') finishCreate(true);
   877→                else if (e.key === 'Escape') {
   878→                  setCreateParent(null);
   879→                  setCreateText('');
   880→                }
   881→              }}
   882→              onBlur={() => finishCreate(false)}
   883→            />
   884→          </div>
   885→        )}
   886→        {isOpen && children.map(renderRow)}
   887→      </React.Fragment>
   888→    );
   889→  };
   890→
   891→  return (
   892→    <aside className={`lib-tree ${treeOpen ? '' : 'closed'}`} aria-label="Library folder tree">
   893→      <div className="lib-tree-head">
   894→        <span className="lib-tree-title">
   895→          <ListTree size={13} aria-hidden="true" /> Folders
   896→        </span>
   897→        <button
   898→          type="button"
   899→          className="lib-tree-fold"
   900→          onClick={() => setTreeOpen((v) => !v)}
   901→          aria-expanded={treeOpen}
   902→          aria-label={treeOpen ? 'Collapse folder tree' : 'Expand folder tree'}
   903→          title={treeOpen ? 'Collapse folder tree' : 'Expand folder tree'}
   904→        >
   905→          <ChevronDown size={14} />
   906→        </button>
   907→      </div>
   908→      {treeOpen && (
   909→        <div className="lib-tree-body" role="tree" aria-label="Folders">
   910→          <div
   911→            role="treeitem"
   912→            aria-selected={folderFilter === 'all'}
   913→            aria-level={1}
   914→            tabIndex={0}
   915→            className={`lib-tree-row vroot ${folderFilter === 'all' ? 'active' : ''} ${
   916→              dropId === 'all' ? 'drop' : ''
   917→            }`}
   918→            title="All files — drop here to unfile"
   919→            onClick={() => setFolderFilter('all')}

   943→            onDrop={(e) => handleDrop(e, 'unfiled')}

   945→            <span className="lib-tree-caret spacer" aria-hidden="true" />
   946→            <Inbox size={13} className="lib-tree-ico" aria-hidden="true" />

   948→            <span className="lib-folder-count">{unfiledCount}</span>

   950→          {tree.map(renderRow)}
   951→          {createParent === 'root' ? (
   952→            <div className="lib-subrow" style={{ ['--lib-d' as string]: 0 }}>
   953→              <input
   954→                className="lib-folder-input lib-tree-new"

   956→                autoFocus
   957→                placeholder="Folder name…"
   958→                aria-label="New folder name"
   959→                onChange={(e) => setCreateText(e.target.value)}
   960→                onKeyDown={(e) => {
   961→                  if (e.key === 'Enter') finishCreate(true);
   962→                  else if (e.key === 'Escape') {
   963→                    setCreateParent(null);
   964→                    setCreateText('');
   965→                  }
   966→                }}
   967→                onBlur={() => finishCreate(false)}
   968→              />
   969→            </div>
   970→          ) : (
   971→            <button
   972→              type="button"
   973→              className="lib-newfolder-btn"
   974→              onClick={() => {
   975→                setRenameId(null);
   976→                setIconPop(null);
   977→                setCreateParent('root');
   978→              }}
   979→            >
   980→              <FolderPlus size={13} /> New folder
   981→            </button>
   982→          )}
   983→        </div>
   984→      )}
   985→      {iconPop &&
   986→        createPortal(
   987→          <div
   988→            className="lib-icon-pop"
   989→            style={{ top: iconPop.top, left: iconPop.left }}
   990→            role="dialog"
   991→            aria-label="Pick a folder icon"
   992→          >
   993→            {FOLDER_ICONS.map((key) => {
   994→              const Ico = folderIconOf(key);
   995→              const sel = folders.find((f) => f.id === iconPop.id)?.icon === key;
   996→              return (
   997→                <button
   998→                  key={key}
   999→                  type="button"
  1000→                  className={`lib-icon-cell ${sel ? 'sel' : ''}`}
  1001→                  title={key}
  1002→                  aria-label={`Icon: ${key}`}
  1003→                  aria-pressed={sel}
  1004→                  onClick={() => {
  1005→                    setFolderIcon(iconPop.id, key);
  1006→                    setIconPop(null);
  1007→                  }}
  1008→                >
  1009→                  <Ico size={15} />
  1010→                </button>
  1011→              );
  1012→            })}
  1013→          </div>,
  1014→          document.body
  1015→        )}
  1016→    </aside>
  1017→  );
  1018→}

  1371→      clearSelection();

  1373→    }
  1374→    selectMany(visible.map((f) => f.id), 'replace');

  1381→      toast('Nothing to move', 'The selected files are already there.');

  1388→  };

  1391→    const st = useWB.getState();
  1392→    const targets = st.files.filter((f) => ids.has(f.id) && f.archived !== archived);

  1395→        archived ? 'Nothing to archive' : 'Nothing to unarchive',

  1400→    st.snapshot();
  1401→    for (const f of targets) st.updateFile(f.id, { archived });
  1402→    toast(
  1403→      `${archived ? 'Archived' : 'Unarchived'} ${targets.length} ${targets.length === 1 ? 'file' : 'files'}`

  1416→      toast(`Deleted ${victims.length} ${victims.length === 1 ? 'file' : 'files'}`);
  1417→    })();
  1418→  };
  1419→
  1420→  return (
  1421→    <div className="lib-bulk" role="toolbar" aria-label="Bulk actions for selected files">
  1422→      <span className="lib-bulk-count">
  1423→        <CheckSquare size={13} aria-hidden="true" />

  1425→      </span>
  1426→      <Btn size="sm" variant="ghost" onClick={selectAllOrNone} disabled={!visible.length}>
  1427→        {allSelected ? <X size={13} /> : <CheckSquare size={13} />} {allSelected ? 'None' : 'All'}
  1428→      </Btn>
  1429→      <label className="lib-bulk-move" title="Move the selected files to a folder">
