    43→function roadmapProgress(rm: Roadmap, liveMap: Map<string, RmNodeLive>): number {
    44→  const vals: number[] = [];
    45→  rm.nodes.forEach((n) => {
    46→    const live = liveMap.get(n.id);
    47→    if (!live) return;
    48→    if (n.kind === 'note') return; // notes carry no progress
    49→    vals.push(live.pct >= 0 ? live.pct : live.done ? 100 : 0);
    50→  });
    51→  if (!vals.length) return 0;
    52→  return Math.round(vals.reduce((a, b) => a + b, 0) / vals.length);
    53→}

    55→/* ---------- execution order computation (module-scope pure) ---------- */
    56→
    57→interface OrderRow {
    58→  id: string;
    59→  idx: number;
    60→  node: RoadmapNode;
    61→  live: RmNodeLive | undefined;
    62→  est: number;
    63→  cum: number;
    64→  blocked: boolean;
    65→}
    66→
    67→function computeOrder(
    68→  roadmap: Roadmap,
    69→  liveMap: Map<string, RmNodeLive>
    70→): { rows: OrderRow[]; totalEst: number; critical: Set<string> } {
    71→  const order = topoOrder(roadmap.nodes, roadmap.edges);
    72→  const estOf = (id: string): number => liveMap.get(id)?.estMin ?? 0;
    73→
    74→  /* running cumulative down the list */
    75→  const rows: OrderRow[] = [];
    76→  let cum = 0;
    77→  order.forEach((id, idx) => {
    78→    const node = roadmap.nodes.find((n) => n.id === id);
    79→    if (!node) return;
    80→    const est = estOf(id);
    81→    cum += est;
    82→    const blocked = roadmap.edges.some((e) => {
    83→      if (e.to !== id) return false;
    84→      const st = liveMap.get(e.from)?.status;
    85→      return !!st && st !== 'done' && st !== 'none';
    86→    });
    87→    rows.push({ id, idx, node, live: liveMap.get(id), est, cum, blocked });
    88→  });
    89→
    90→  /* critical path: longest est chain (DP over topo order) */
    91→  const best = new Map<string, { total: number; prev: string | null }>();
    92→  order.forEach((id) => {
    93→    let prev: string | null = null;
    94→    let prevTotal = -1;
    95→    roadmap.edges.forEach((e) => {
    96→      if (e.to !== id) return;
    97→      const t = best.get(e.from)?.total ?? 0;
    98→      if (t > prevTotal) {
    99→        prevTotal = t;
   100→        prev = e.from;
   101→      }
   102→    });
   103→    best.set(id, { total: estOf(id) + Math.max(0, prevTotal), prev });
   104→  });
   105→  let end: string | null = null;
   106→  let max = 0;
   107→  best.forEach((v, id) => {
   108→    if (v.total > max) {
   109→      max = v.total;
   110→      end = id;
   111→    }
   112→  });
   113→  const critical = new Set<string>();
   114→  let cur: string | null = end && max > 0 ? end : null;
   115→  while (cur) {
   116→    critical.add(cur);
   117→    cur = best.get(cur)?.prev ?? null;
   118→  }
   119→
   120→  return { rows, totalEst: cum, critical };
   121→}

   123→function Ring({ pct }: { pct: number }) {
   124→  const r = 15.5;
   125→  const c = 2 * Math.PI * r;
   126→  return (
   127→    <div className="rm-ring" aria-hidden="true">
   128→      <svg viewBox="0 0 38 38" width={38} height={38}>
   129→        <circle className="rm-ring-track" cx={19} cy={19} r={r} />
   130→        <circle
   131→          className="rm-ring-val"
   132→          cx={19} cy={19} r={r}
   133→          strokeDasharray={`${(c * pct) / 100} ${c}`}
   134→          transform="rotate(-90 19 19)"
   135→        />
   136→      </svg>
   137→      <span className="rm-ring-pct mono">{pct}%</span>
   138→    </div>
   139→  );

   281→function RoadmapWorkspace({ roadmap }: { roadmap: Roadmap }) {
   282→  const items = useWB((s) => s.items);

   289→  const removeRoadmapEdge = useWB((s) => s.removeRoadmapEdge);
   290→  const updateRoadmapEdge = useWB((s) => s.updateRoadmapEdge);
   291→  const setActiveRoadmap = useWB((s) => s.setActiveRoadmap);
   292→
   293→  const canvasRef = useRef<RoadmapCanvasHandle>(null);
   294→  const [selection, setSelection] = useState<RmSelection | null>(null);
   295→  const [linkMode, setLinkMode] = useState(false);
   296→  const [linkSource, setLinkSource] = useState<string | null>(null);
   297→  const [orderOpen, setOrderOpen] = useState(false);
   298→  const [pop, setPop] = useState<PopState>(null);
   299→  const [popSearch, setPopSearch] = useState('');
   300→  const [zoom, setZoom] = useState(1);
   301→  const [fitKey, setFitKey] = useState(0);
   302→
   303→  const liveMap = useMemo(() => {
   304→    const m = new Map<string, RmNodeLive>();
   305→    roadmap.nodes.forEach((n) => m.set(n.id, nodeLive(n, items, files)));
   306→    return m;
   307→  }, [roadmap.nodes, items, files]);
   308→
   309→  const selectedNode =
   310→    selection?.kind === 'node' ? roadmap.nodes.find((n) => n.id === selection.id) || null : null;
   311→  const selectedEdge =
   312→    selection?.kind === 'edge' ? roadmap.edges.find((e) => e.id === selection.id) || null : null;
   313→
   314→  /* keyboard: Esc closes pop / cancels link / clears selection; Delete removes */
   315→  useEffect(() => {
   316→    const onKey = (e: KeyboardEvent) => {
   317→      const t = e.target as HTMLElement;
   318→      if (['INPUT', 'TEXTAREA', 'SELECT'].includes(t?.tagName) || t?.isContentEditable) return;
   319→      if (e.key === 'Escape') {
   320→        if (pop) setPop(null);
   321→        else if (linkSource) setLinkSource(null);
   322→        else if (selection) setSelection(null);
   323→        return;
   324→      }
   325→      if ((e.key === 'Delete' || e.key === 'Backspace') && selection) {
   326→        e.preventDefault();
   327→        if (selection.kind === 'node') {
   328→          removeRoadmapNode(roadmap.id, selection.id);
   329→          toast('Node removed');
   330→        } else {
   331→          removeRoadmapEdge(roadmap.id, selection.id);
   332→          toast('Link removed');
   333→        }
   334→        setSelection(null);
   335→      }
   336→    };
   337→    window.addEventListener('keydown', onKey);
   338→    return () => window.removeEventListener('keydown', onKey);
   339→  }, [pop, linkSource, selection, roadmap.id, removeRoadmapNode, removeRoadmapEdge]);
   340→
   341→  /* ---------- node creation ---------- */
   342→
   343→  const spawn = (partial: Omit<RoadmapNode, 'id'>) => {
   344→    const spot = canvasRef.current?.dropSpot() ?? { x: 80, y: 80 };
   345→    const node = addRoadmapNode(roadmap.id, { ...partial, x: spot.x, y: spot.y });
   346→    setPop(null);
   347→    setPopSearch('');
   348→    setSelection({ kind: 'node', id: node.id });
   349→    return node;
   350→  };
   351→
   352→  const pickList = useMemo(() => {
   353→    if (!pop || pop.mode !== 'pick') return [] as { id: string; title: string; meta: string[] }[];
   354→    const q = popSearch.trim().toLowerCase();
   355→    if (pop.kind === 'file') {
   356→      return files
   357→        .filter((f) => !f.archived)
   358→        .filter((f) => !q || f.title.toLowerCase().includes(q) || f.fileName.toLowerCase().includes(q))
   359→        .slice(0, 40)
   360→        .map((f) => {
   361→          const live = { pct: 0, pagesRead: 0, pagesTotal: f.pageCount } as const;
   362→          void live;
   363→          return {
   364→            id: f.id,
   365→            title: f.title,
   366→            meta: [`${f.pageCount} pages`, f.deadline ? `due ${f.deadline.slice(5)}` : 'no deadline'],
   367→          };
   368→        });
   369→    }
   370→    return items
   371→      .filter((i) => i.type === pop.kind)
   372→      .filter((i) => !['archived'].includes(i.status))
   373→      .filter((i) => !q || i.title.toLowerCase().includes(q))
   374→      .sort((a, b) => (a.status === 'done' ? 1 : 0) - (b.status === 'done' ? 1 : 0))
   375→      .slice(0, 40)
   376→      .map((i) => ({
   377→        id: i.id,
   378→        title: i.title,
   379→        meta: [
   380→          i.status.replace('_', ' '),
   381→          i.deadline ? `due ${i.deadline.slice(5)}` : 'no deadline',
   382→          i.estimateMin ? `${i.estimateMin}m` : '',
   383→        ].filter(Boolean),
   384→      }));
   385→  }, [pop, popSearch, items, files]);
   386→
   387→  /* ---------- auto layout ---------- */
   388→
   389→  const autoLayout = () => {
   390→    const order = topoOrder(roadmap.nodes, roadmap.edges);
   391→    const depth = new Map<string, number>();
   392→    roadmap.nodes.forEach((n) => depth.set(n.id, 0));
   393→    order.forEach((id) => {
   394→      roadmap.edges.forEach((e) => {
   395→        if (e.from !== id) return;
   396→        depth.set(e.to, Math.max(depth.get(e.to) ?? 0, (depth.get(id) ?? 0) + 1));
   397→      });
   398→    });
   399→    const layers = new Map<number, RoadmapNode[]>();
   400→    roadmap.nodes.forEach((n) => {
   401→      const d = depth.get(n.id) ?? 0;
   402→      const arr = layers.get(d) || [];
   403→      arr.push(n);
   404→      layers.set(d, arr);
   405→    });
   406→    layers.forEach((arr, d) => {
   407→      arr.sort((a, b) => a.y - b.y);
   408→      arr.forEach((n, i) => updateRoadmapNode(roadmap.id, n.id, { x: 60 + d * 290, y: 60 + i * 150 }));
   409→    });
   410→    setFitKey((k) => k + 1);
   411→    toast('Auto-layout applied', 'Layers follow your connections.');
   412→  };
   413→
   414→  /* ---------- order panel data ---------- */
   415→
   416→  const orderRows = useMemo(
   417→    () => computeOrder(roadmap, liveMap),
   418→    [roadmap, liveMap]
   419→  );
   420→

   423→  return (
   424→    <div className="rm-screen fade-in">
   425→      <div className="rm-toolbar">
   426→        <Btn size="sm" onClick={() => setActiveRoadmap(null)} title="Back to all roadmaps">
   427→          <ChevronLeft size={14} /> Roadmaps
   428→        </Btn>
   429→        <div className="rm-toolbar-sep" />
   430→        <div className="rm-toolbar-titles">
   431→          <input
   432→            className="rm-title-input"
   433→            value={roadmap.title}
   434→            aria-label="Roadmap title"
   435→            onChange={(e) => updateRoadmap(roadmap.id, { title: e.target.value })}
   436→          />
   437→          <input
   438→            className="rm-desc-input"
   439→            value={roadmap.description}
   440→            placeholder="Add a short description…"
   441→            aria-label="Roadmap description"
   442→            onChange={(e) => updateRoadmap(roadmap.id, { description: e.target.value })}
   443→          />
   444→        </div>
   445→
   446→        {/* add node */}
   447→        <div className="rm-pop-wrap">
   448→          <Btn
   449→            size="sm"
   450→            variant="primary"
   451→            onClick={() => setPop(pop ? null : { mode: 'menu' })}
   452→            title="Add a node"
   453→          >
   454→            <Plus size={14} /> Add node
   455→          </Btn>
   456→          {pop && (
   457→            <div className="rm-pop" role="dialog" aria-label="Add node">
   458→              {pop.mode === 'menu' && (
   459→                <>
   460→                  <div className="rm-pop-head">
   461→                    <span className="rm-pop-head-title">Add to the canvas</span>
   462→                  </div>
   463→                  {RM_KIND_META.map((k) => (
   464→                    <button
   465→                      key={k.kind}
   466→                      type="button"
   467→                      className="rm-pop-item"
   468→                      onClick={() => {
   469→                        if (k.kind === 'milestone') setPop({ mode: 'milestone' });
   470→                        else if (k.kind === 'note') setPop({ mode: 'note' });

   474→                      <k.Icon />

   750→                        {crit && <span className="rm-crit-chip">critical</span>}
   751→                        {r.blocked && (
   752→                          <span className="rm-blocked-ico" title="Waiting on an unfinished node">
   753→                            <svg viewBox="0 0 24 24" width="11" height="11" fill="none" stroke="currentColor" strokeWidth="2.4" strokeLinecap="round" strokeLinejoin="round">
   754→                              <path d="M10.3 3.9 1.8 18a2 2 0 0 0 1.7 3h17a2 2 0 0 0 1.7-3L13.7 3.9a2 2 0 0 0-3.4 0Z" />
   755→                              <path d="M12 9v4" />
   756→                              <path d="M12 17h.01" />

   758→                          </span>
   759→                        )}
   760→                        {r.live?.deadline && (
   761→                          <span className={`mono rm-dl-${r.live.deadlineTone ?? 'ok'}`}>
   762→                            {r.live.deadline.slice(5)}
   763→                          </span>
   764→                        )}
   765→                      </span>
   766→                    </span>
   767→                    <span className="rm-row-times">
   768→                      <span className="rm-est mono">{r.est ? fmtDuration(r.est) : '—'}</span>
   769→                      <span className="rm-cum mono">Σ {fmtDuration(r.cum)}</span>

   771→                  </button>

   773→              })}
   774→              {!orderRows.rows.length && (
   775→                <div className="rm-empty-hint">Add nodes to see their execution order here.</div>

   777→            </div>
   778→            <div className="rm-order-foot">
   779→              Estimates use item estimates and your reading pace. Critical path is highlighted —

   781→            </div>
   782→          </aside>
   783→        )}
   784→      </div>
   785→    </div>
   786→  );

   791→function EdgeLabelInput({ initial, onCommit }: { initial: string; onCommit: (v: string) => void }) {
   792→  const [value, setValue] = useState(initial);
   793→  return (
   794→    <input
   795→      className="rm-edge-edit"
   796→      value={value}
   797→      placeholder="Link label…"
   798→      aria-label="Link label"
   799→      onChange={(e) => setValue(e.target.value)}
   800→      onBlur={() => onCommit(value.trim())}
   801→      onKeyDown={(e) => {
   802→        if (e.key === 'Enter') (e.target as HTMLInputElement).blur();
   803→      }}
   804→    />
   805→  );
   806→}
   807→
   808→function MiniForm({
   809→  fields,
   810→  submitLabel,
   811→  onSubmit,
   812→  onCancel,
   813→}: {
   814→  fields: { key: string; placeholder: string; type?: string; autoFocus?: boolean }[];
   815→  submitLabel: string;
   816→  onSubmit: (values: Record<string, string>) => void;
   817→  onCancel: () => void;
   818→}) {
   819→  const [values, setValues] = useState<Record<string, string>>({});
   820→  return (
   821→    <div className="rm-mini-form">
   822→      {fields.map((f) => (
   823→        <input
   824→          key={f.key}
   825→          className="rm-mini-input"
   826→          type={f.type || 'text'}
   827→          autoFocus={f.autoFocus}
   828→          placeholder={f.placeholder}
   829→          value={values[f.key] || ''}
   830→          onChange={(e) => setValues((v) => ({ ...v, [f.key]: e.target.value }))}
   831→          onKeyDown={(e) => {
   832→            if (e.key === 'Enter') onSubmit(values);
   833→            if (e.key === 'Escape') onCancel();
   834→          }}
   835→        />
   836→      ))}
   837→      <div className="rm-mini-actions">
   838→        <Btn size="sm" onClick={onCancel}>Cancel</Btn>
   839→        <Btn size="sm" variant="primary" onClick={() => onSubmit(values)}>{submitLabel}</Btn>
   840→      </div>
   841→    </div>
   842→  );
   843→}
