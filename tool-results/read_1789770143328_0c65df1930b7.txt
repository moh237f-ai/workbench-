    61→const snap = (n: number): number => Math.round(n / SNAP_MIN) * SNAP_MIN;

    64→function timeboxDurMin(it: Item): number {
    65→  if (it.plannedStart && it.plannedEnd) {
    66→    const d = timeDiffMin(it.plannedStart.slice(11, 16), it.plannedEnd.slice(11, 16));
    67→    if (d > 0) return d;
    68→  }
    69→  return it.estimateMin || 30;
    70→}

    72→interface Slot {
    73→  start: number;
    74→  end: number;
    75→}
    76→
    77→/** first free slot ≥ max(8:00, now-if-today) in 5-min steps */
    78→function nextFreeSlot(duration: number, occupied: Slot[], fromMin: number): number {
    79→  let start = snap(Math.max(480, fromMin));
    80→  const limit = 1440 - duration;
    81→  while (start <= limit) {
    82→    if (occupied.every((b) => start + duration <= b.start || start >= b.end)) return start;
    83→    start += SNAP_MIN;
    84→  }
    85→  const lastEnd = occupied.reduce((m, b) => Math.max(m, b.end), 0);
    86→  return clamp(Math.max(480, lastEnd), 480, 1380);
    87→}

   249→function QuickAddPop({
   250→  date,
   251→  startMin,
   252→  x,
   253→  y,
   254→  onClose,
   255→}: {
   256→  date: string;
   257→  startMin: number;
   258→  x: number;
   259→  y: number;
   260→  onClose: () => void;
   261→}) {
   262→  const files = useWB((s) => s.files);
   263→  const addItem = useWB((s) => s.addItem);
   264→  const addFilePlan = useWB((s) => s.addFilePlan);
   265→  const snapshot = useWB((s) => s.snapshot);
   266→
   267→  const [title, setTitle] = useState('');
   268→  const [dur, setDur] = useState(30);
   269→  const [kind, setKind] = useState<'task' | 'reading'>('task');
   270→  const [fileId, setFileId] = useState<string | null>(null);
   271→  const [query, setQuery] = useState('');
   272→  const popRef = useRef<HTMLDivElement | null>(null);
   273→
   274→  const pool = useMemo(
   275→    () =>
   276→      files
   277→        .filter((f) => !f.archived && wbFileProgress(f) < 100)
   278→        .sort(
   279→          (a, b) =>
   280→            (a.deadline || '9999-12-31').localeCompare(b.deadline || '9999-12-31') ||
   281→            a.title.localeCompare(b.title)
   282→        ),
   283→    [files]
   284→  );
   285→  const filtered = useMemo(() => {
   286→    const q = query.trim().toLowerCase();
   287→    return q ? pool.filter((f) => f.title.toLowerCase().includes(q)) : pool;
   288→  }, [pool, query]);
   289→
   290→  /* Esc + click-outside close */
   291→  useEffect(() => {
   292→    const onKey = (e: KeyboardEvent) => {
   293→      if (e.key === 'Escape') {
   294→        e.stopPropagation();
   295→        onClose();
   296→      }
   297→    };
   298→    const onPointer = (e: PointerEvent) => {
   299→      if (popRef.current && !popRef.current.contains(e.target as Node)) onClose();
   300→    };
   301→    window.addEventListener('keydown', onKey, true);
   302→    window.addEventListener('pointerdown', onPointer, true);
   303→    return () => {
   304→      window.removeEventListener('keydown', onKey, true);
   305→      window.removeEventListener('pointerdown', onPointer, true);
   306→    };
   307→  }, [onClose]);
   308→
   309→  const endMin = startMin + dur;
   310→  const selFile = pool.find((f) => f.id === fileId) || null;
   311→
   312→  const submit = () => {
   313→    if (kind === 'task') {
   314→      const t = title.trim() || 'Untitled task';
   315→      snapshot();
   316→      addItem({
   317→        title: t,
   318→        type: 'task',
   319→        status: 'todo',
   320→        estimateMin: dur,
   321→        plannedStart: dateTimeFromMin(date, startMin),
   322→        plannedEnd: dateTimeFromMin(date, endMin),
   323→        startDate: date,
   324→      });
   325→      toast('Block added', `${t} · ${minToHHMM(startMin)}–${minToHHMM(endMin)}`);
   326→    } else {
   327→      if (!selFile) return;
   328→      snapshot();
   329→      addFilePlan(selFile.id, date, startMin, dur);
   330→      toast('Reading block added', `${selFile.title} · ${minToHHMM(startMin)}–${minToHHMM(endMin)}`);
   331→    }
   332→    onClose();
   333→  };
   334→
   335→  return (
   336→    <div
   337→      ref={popRef}
   338→      className="plan-pop"
   339→      style={{ left: x, top: y }}
   340→      role="dialog"
   341→      aria-label={`Quick add block at ${minToHHMM(startMin)}`}
   342→    >
   343→      <div className="plan-pop-head">
   344→        <span className="plan-pop-title">
   345→          New block <span className="mono">at {minToHHMM(startMin)}</span>
   346→        </span>
   347→        <button
   348→          type="button"
   349→          className="plan-pop-x"
   350→          onClick={onClose}
   351→          aria-label="Close quick add"
   352→          title="Close"
   353→        >
   354→          <X size={13} />
   355→        </button>
   356→      </div>
   357→
   358→      {kind === 'task' && (
   359→        <input
   360→          className="plan-pop-input"
   361→          value={title}
   362→          onChange={(e) => setTitle(e.target.value)}
   363→          placeholder="What needs doing?"
   364→          maxLength={120}
   365→          autoFocus
   366→          onKeyDown={(e) => {
   367→            if (e.key === 'Enter') {
   368→              e.preventDefault();
   369→              submit();
   370→            }
   371→          }}
   372→        />
   373→      )}
   374→
   375→      <div className="plan-pop-row">
   376→        <span className="plan-pop-lab">Duration</span>
   377→        <Seg<string>
   378→          options={QUICK_DURS.map((d) => ({ id: String(d), label: `${d}m` }))}
   379→          value={String(dur)}
   380→          onChange={(v) => setDur(Number(v))}
   381→        />
   382→      </div>
   383→
   384→      <div className="plan-pop-row">
   385→        <span className="plan-pop-lab">Type</span>
   386→        <Seg<'task' | 'reading'>
   387→          options={[
   388→            { id: 'task', label: 'Task' },
   389→            { id: 'reading', label: 'Reading' },
   390→          ]}
   391→          value={kind}
   392→          onChange={setKind}
   393→        />
   394→      </div>
   395→
   396→      {kind === 'reading' && (
   397→        <div className="plan-pop-files">
   398→          {pool.length > 8 && (
   399→            <div className="plan-pop-search">
   400→              <Search size={12} aria-hidden />
   401→              <input
   402→                value={query}
   403→                onChange={(e) => setQuery(e.target.value)}
   404→                placeholder="Find a file…"
   405→                aria-label="Search files"
   406→              />
   407→            </div>
   408→          )}
   409→          <div className="plan-pop-filelist plan-scroll">
   410→            {filtered.map((f) => {
   411→              const left = Math.max(0, f.pageCount - wbFileReadPages(f));
   412→              return (
   413→                <button
   414→                  key={f.id}
   415→                  type="button"
   416→                  className={`plan-pop-file${fileId === f.id ? ' sel' : ''}`}
   417→                  aria-pressed={fileId === f.id}
   418→                  onClick={() => setFileId(f.id)}
   419→                  title={f.title}
   420→                >
   421→                  <BookOpen size={13} aria-hidden />
   422→                  <span className="plan-pop-ftitle">{f.title}</span>
   423→                  <span className="chip chip-ok">{left}p left</span>
   424→                </button>

   426→            })}
   427→            {!filtered.length && (
   428→              <div className="plan-pop-nofiles">
   429→                {pool.length ? 'No matching files.' : 'No unread files in the library.'}
   430→              </div>
   431→            )}
   432→          </div>
   433→        </div>
   434→      )}
   435→
   436→      <div className="plan-pop-foot">
   437→        <Btn size="sm" variant="ghost" onClick={onClose}>
   438→          Cancel
   439→        </Btn>
   440→        <Btn size="sm" variant="primary" disabled={kind === 'reading' && !selFile} onClick={submit}>
   441→          Add
   442→        </Btn>
   443→      </div>
   444→    </div>
   445→  );
   446→}

   448→/* ==================== Up Next card (isolated so the 1s ticker stays cheap) ==================== */

   450→interface UnBlock {
   451→  kind: DragKind;

   459→function UpNextCard({
   460→  date,
   461→  timeboxed,
   462→  fileBlocks,
   463→  fileById,
   464→}: {
   465→  date: string;
   466→  timeboxed: Item[];
   467→  fileBlocks: FilePlanBlock[];
   468→  fileById: Map<string, ResourceFile>;
   469→}) {
   470→  const { openItemDetail } = useViewActions();
   471→  const isToday = date === todayStr();
   472→  const now = useNow(isToday);
   473→
   474→  const blocks = useMemo<UnBlock[]>(() => {
   475→    const list: UnBlock[] = [
   476→      ...timeboxed
   477→        .filter((it) => it.status !== 'done')
   478→        .map((it) => {
   479→          const s = plannedStartToMin(it.plannedStart ?? '');
   480→          return {
   481→            kind: 'item' as const,
   482→            id: it.id,
   483→            fileId: null,
   484→            title: it.title,
   485→            start: s,
   486→            end: s + timeboxDurMin(it),
   487→          };
   488→        }),
   489→      ...fileBlocks
   490→        .filter((p) => !p.done)
   491→        .map((p) => ({
   492→          kind: 'file' as const,
   493→          id: p.id,
   494→          fileId: p.fileId,
   495→          title: fileById.get(p.fileId)?.title || 'Reading block',
   496→          start: p.startMin,
   497→          end: p.startMin + p.minutes,
   498→        })),
   499→    ];
   500→    list.sort((a, b) => a.start - b.start);
   501→    return list;
   502→  }, [timeboxed, fileBlocks, fileById]);
   503→
   504→  const nowMin = isToday ? now.getHours() * 60 + now.getMinutes() : -1;
   505→  const running = isToday ? blocks.find((b) => b.start <= nowMin && nowMin < b.end) : undefined;
   506→  const next = running ? undefined : blocks.find((b) => b.start >= nowMin);
   507→  const active = running ?? next;
   508→
   509→  const open = (b: UnBlock) => {
   510→    if (b.kind === 'item') openItemDetail(b.id);
   511→    else if (b.fileId) window.dispatchEvent(new CustomEvent('wb-open-reader', { detail: b.fileId }));
   512→  };
   513→
   514→  let body: React.ReactNode;
   515→  if (active) {
   516→    const meta = running
   517→      ? `${minToHHMM(running.start)}–${minToHHMM(running.end)} · ${fmtDuration(Math.max(0, running.end - nowMin))} left`
   518→      : isToday
   519→        ? `at ${minToHHMM(next!.start)} · in ${fmtDuration(Math.max(0, next!.start - nowMin))}`
   520→        : `at ${minToHHMM(next!.start)}`;
   521→    body = (
   522→      <button
   523→        type="button"
   524→        className="plan-un-row"
   525→        onClick={() => open(active)}
   526→        title={running ? 'Open current block' : 'Open next block'}
   527→      >
   528→        {running && <span className="plan-un-dot" aria-hidden />}
   529→        <span className="plan-un-main">
   530→          <span className="plan-un-kicker">{running ? 'Now' : isToday ? 'Next' : 'First block'}</span>
   531→          <span className="plan-un-title">{active.title}</span>
   532→          <span className="plan-un-meta mono">{meta}</span>
   533→        </span>
   534→        <ChevronRight size={14} className="plan-un-go" aria-hidden />
   535→      </button>
   536→    );
   537→  } else {
   538→    body = (
   539→      <div className="plan-un-empty">
   540→        {isToday ? 'Nothing left today — nice.' : 'Nothing planned for this day.'}
   541→      </div>
   542→    );
   543→  }
   544→
   545→  return (
   546→    <div className="card plan-card">
   547→      <div className="plan-card-head">
   548→        <span className="plan-card-title">Up next</span>
   549→      </div>
   550→      {body}
   551→    </div>
   552→  );
   553→}

   557→function DayPlanner({ date }: { date: string }) {
   558→  const items = useWB((s) => s.items);
   559→  const files = useWB((s) => s.files);
   560→  const filePlans = useWB((s) => s.filePlans);
   561→  const capacity = useWB((s) => s.dailyCapacityMin);
   562→  const updateItem = useWB((s) => s.updateItem);
   563→  const addItem = useWB((s) => s.addItem);
   564→  const updateFilePlan = useWB((s) => s.updateFilePlan);
   565→  const deleteFilePlan = useWB((s) => s.deleteFilePlan);
   566→  const addFilePlan = useWB((s) => s.addFilePlan);
   567→  const snapshot = useWB((s) => s.snapshot);
   568→  const confirmMsg = useConfirm();
   569→  const { openItemDetail } = useViewActions();
   570→
   571→  const scrollRef = useRef<HTMLDivElement | null>(null);
   572→  const lanesRef = useRef<HTMLDivElement | null>(null);
   573→  const [drag, setDrag] = useState<DragState | null>(null);
   574→  const [pop, setPop] = useState<PopState | null>(null);
   575→  const [poolDrag, setPoolDrag] = useState<PoolDrag | null>(null);
   576→
   577→  /* auto-scroll to 08:00 on mount / date change (a stale popover hides via pop.date !== date) */
   578→  useEffect(() => {
   579→    const el = scrollRef.current;
   580→    if (el) el.scrollTop = Math.max(0, (480 - DAY_START_MIN) * PX_PER_MIN - 8);
   581→  }, [date]);
   582→
   583→  /* safety net: discard drag if pointerup/cancel escapes the captured element */
   584→  useEffect(() => {
   585→    if (!drag) return;
   586→    const clear = () => setDrag(null);
   587→    window.addEventListener('pointerup', clear);
   588→    window.addEventListener('pointercancel', clear);
   589→    return () => {
   590→      window.removeEventListener('pointerup', clear);
   591→      window.removeEventListener('pointercancel', clear);
   592→    };
   593→  }, [drag]);
   594→
   595→  /* ---------- data ---------- */
   596→  const timeboxed = useMemo(() => wbTimeboxedItems(items, date), [items, date]);
   597→  const fileBlocks = useMemo(() => wbFilePlansForDate(filePlans, date), [filePlans, date]);
   598→  const fileById = useMemo(() => {
   599→    const m = new Map<string, ResourceFile>();
   600→    files.forEach((f) => m.set(f.id, f));
   601→    return m;
   602→  }, [files]);
   603→
   604→  const occupied = useMemo<Slot[]>(
   605→    () => [
   606→      ...timeboxed.map((it) => {
   607→        const s = plannedStartToMin(it.plannedStart ?? '');
   608→        return { start: s, end: s + timeboxDurMin(it) };
   609→      }),
   610→      ...fileBlocks.map((p) => ({ start: p.startMin, end: p.startMin + p.minutes })),
   611→    ],
   612→    [timeboxed, fileBlocks]
   613→  );
   614→
   615→  const plannedItemMin = useMemo(() => timeboxed.reduce((s, it) => s + timeboxDurMin(it), 0), [timeboxed]);
   616→  const plannedFileMin = useMemo(() => fileBlocks.reduce((s, p) => s + p.minutes, 0), [fileBlocks]);
   617→  const plannedMin = plannedItemMin + plannedFileMin;
   618→  const overCap = plannedMin > capacity;
   619→  const freeMin = Math.max(0, capacity - plannedMin);
   620→  const blockCount = timeboxed.length + fileBlocks.length;
   621→  const doneCount =
   622→    timeboxed.filter((it) => it.status === 'done').length + fileBlocks.filter((p) => p.done).length;
   623→
   624→  const suggested = useMemo(
   625→    () =>
   626→      items
   627→        .filter((i) => {
   628→          if (['done', 'archived', 'inbox'].includes(i.status)) return false;
   629→          if (i.plannedStart && i.plannedStart.slice(0, 10) === date) return false;
   630→          if (wbIsBlocked(items, i)) return false;
   631→          const du = i.deadline ? daysUntil(i.deadline) : null;
   632→          return (du !== null && du <= 3) || i.priority === 'p0' || i.priority === 'p1';
   633→        })
   634→        .sort((a, b) => {
   635→          const p = wbPriorityWeight(a.priority) - wbPriorityWeight(b.priority);
   636→          if (p) return p;
   637→          return (a.deadline || '9999-12-31').localeCompare(b.deadline || '9999-12-31');
   638→        })
   639→        .slice(0, 30),
   640→    [items, date]
   641→  );
   642→
   643→  const readingGoals = useMemo(
   644→    () =>
   645→      files
   646→        .filter((f) => {
   647→          if (f.archived) return false;
   648→          const prog = wbFileProgress(f);
   649→          if (prog >= 100) return false;
   650→          if (f.deadline && daysUntil(f.deadline) <= 7) return true;
   651→          return prog > 0 || !!f.deadline; /* in progress, or unread with a deadline */
   652→        })
   653→        .sort((a, b) => (a.deadline || '9999-12-31').localeCompare(b.deadline || '9999-12-31'))
   654→        .slice(0, 30),
   655→    [files]
   656→  );
   657→
   658→  /* ---------- scheduling ---------- */
   659→  const nowMinutes = (): number =>
   660→    date === todayStr() ? new Date().getHours() * 60 + new Date().getMinutes() : 0;
   661→
   662→  const scheduleItem = (it: Item) => {
   663→    const dur = it.estimateMin || 30;
   664→    const start = nextFreeSlot(dur, occupied, nowMinutes());
   665→    updateItem(
   666→      it.id,
   667→      {
   668→        plannedStart: dateTimeFromMin(date, start),
   669→        plannedEnd: dateTimeFromMin(date, start + dur),
   670→        startDate: date,
   671→      },
   672→      { snapshot: true }
   673→    );
   674→    toast('Scheduled', `${it.title} at ${minToHHMM(start)}`);
   675→  };
   676→
   677→  const addReadingBlock = (f: ResourceFile) => {
   678→    const start = nextFreeSlot(30, occupied, nowMinutes());
   679→    addFilePlan(f.id, date, start, 30);
   680→    toast('Reading block added', `${f.title} at ${minToHHMM(start)} · 30m`);
   681→  };
   682→
   683→  const autoPlan = () => {
   684→    const occ = occupied.map((o) => ({ ...o }));
   685→    let planned = plannedMin;
   686→    let blocks = 0;
   687→    let addedMin = 0;
   688→    let snapped = false;
   689→    const ensureSnap = () => {
   690→      if (!snapped) {
   691→        snapshot();
   692→        snapped = true;
   693→      }
   694→    };
   695→    for (const it of suggested) {
   696→      if (planned >= capacity) break;
   697→      const dur = it.estimateMin || 30;
   698→      const start = nextFreeSlot(dur, occ, nowMinutes());
   699→      ensureSnap();
   700→      updateItem(it.id, {
   701→        plannedStart: dateTimeFromMin(date, start),
   702→        plannedEnd: dateTimeFromMin(date, start + dur),
   703→        startDate: date,
   704→      });
   705→      occ.push({ start, end: start + dur });
   706→      planned += dur;
   707→      blocks++;
   708→      addedMin += dur;
   709→    }
   710→    let readings = 0;
   711→    for (const f of readingGoals) {
   712→      if (readings >= 2 || planned >= capacity) break;
   713→      const start = nextFreeSlot(30, occ, nowMinutes());
   714→      ensureSnap();
   715→      addFilePlan(f.id, date, start, 30);
   716→      occ.push({ start, end: start + 30 });
   717→      planned += 30;
   718→      blocks++;
   719→      addedMin += 30;
   720→      readings++;
   721→    }
   722→    if (!blocks) {
   723→      toast('Nothing to plan', 'Capacity is already committed.');
   724→      return;
   725→    }
   726→    toast('Auto-planned', `Planned ${blocks} block${blocks === 1 ? '' : 's'} · ${fmtDuration(addedMin)}`);
   727→  };
   728→
   729→  /* ---------- timeline header actions ---------- */
   730→  const copyYesterday = () => {
   731→    const y = addDaysStr(date, -1);
   732→    const yItems = wbTimeboxedItems(items, y);
   733→    const yPlans = wbFilePlansForDate(filePlans, y);
   734→    const n = yItems.length + yPlans.length;
   735→    if (!n) {
   736→      toast('Nothing to copy', 'No blocks on yesterday.');
   737→      return;
   738→    }
   739→    snapshot();
   740→    yItems.forEach((it) => {
   741→      const s = plannedStartToMin(it.plannedStart ?? '');
   742→      const dur = timeboxDurMin(it);
   743→      addItem({
   744→        title: it.title,
   745→        type: it.type,
   746→        priority: it.priority,
   747→        color: it.color,
   748→        estimateMin: dur,
   749→        plannedStart: dateTimeFromMin(date, s),
   750→        plannedEnd: dateTimeFromMin(date, s + dur),
   751→        startDate: date,
   752→        status: 'todo',
   753→      });
   754→    });
   755→    yPlans.forEach((p) => addFilePlan(p.fileId, date, p.startMin, p.minutes));
   756→    toast('Copied from yesterday', `Copied ${n} block${n === 1 ? '' : 's'} from yesterday.`);
   757→  };
   758→
   759→  const clearDay = () => {
   760→    if (!blockCount) return;
   761→    if (!confirmMsg(`Clear all blocks on ${fmtDateNice(date)}?`)) return;
   762→    snapshot();
   763→    timeboxed.forEach((it) => updateItem(it.id, { plannedStart: null, plannedEnd: null }));
   764→    fileBlocks.forEach((p) => deleteFilePlan(p.id));
   765→    toast('Day cleared', `${blockCount} block${blockCount === 1 ? '' : 's'} removed.`);
   766→  };
   767→
   768→  const scrollToNow = () => {
   769→    const el = scrollRef.current;
   770→    if (!el) return;
   771→    const n = new Date();
   772→    const nowMin = n.getHours() * 60 + n.getMinutes();
   773→    el.scrollTo({
   774→      top: Math.max(0, (nowMin - DAY_START_MIN) * PX_PER_MIN - el.clientHeight / 2),
   775→      behavior: 'smooth',
   776→    });
   777→  };
   778→
   779→  /* ---------- quick add (click on empty timeline) ---------- */
   780→  const openQuickAdd = (e: React.MouseEvent<HTMLDivElement>) => {
   781→    const t = e.target as HTMLElement;
   782→    if (t.closest('.plan-block, .plan-pop')) return;
   783→    const rect = lanesRef.current?.getBoundingClientRect();
   784→    if (!rect) return;
   785→    const minute = clamp(snapQ(DAY_START_MIN + (e.clientY - rect.top) / PX_PER_MIN), 0, 1425);
   786→    const x = clamp(e.clientX + 12, 8, Math.max(8, window.innerWidth - POP_W - 8));
   787→    const y = clamp(e.clientY - 10, 8, Math.max(8, window.innerHeight - 432));
   788→    setPop({ date, x, y, startMin: minute });
   789→  };
   790→
   791→  /* ---------- pointer drag (move + resize) ---------- */
   792→  const openItemBlock = (id: string) => openItemDetail(id);
   793→  const openFileBlock = (planId: string) => {
   794→    const p = fileBlocks.find((x) => x.id === planId);
   795→    if (p) window.dispatchEvent(new CustomEvent('wb-open-reader', { detail: p.fileId }));
   796→  };
   797→
   798→  const beginDrag = (
   799→    e: React.PointerEvent<HTMLElement>,
   800→    kind: DragKind,
   801→    id: string,
   802→    mode: 'move' | 'resize',
   803→    origStart: number,
   804→    origDur: number
   805→  ) => {
   806→    if (e.pointerType === 'mouse' && e.button !== 0) return;
   807→    e.stopPropagation();
   808→    try {
   809→      e.currentTarget.setPointerCapture(e.pointerId);
   810→    } catch {
   811→      /* pointer already released — drag simply won't start */
   812→    }
   813→    setDrag({
   814→      kind,
   815→      id,
   816→      mode,
   817→      pointerId: e.pointerId,
   818→      startY: e.clientY,
   819→      origStart,
   820→      origDur,
   821→      start: origStart,
   822→      dur: origDur,
   823→      movedPx: 0,
   824→    });
   825→  };
   826→
   827→  const onDragMove = (e: React.PointerEvent<HTMLElement>) => {
   828→    if (!drag || drag.pointerId !== e.pointerId) return;
   829→    const dy = e.clientY - drag.startY;
   830→    const dMin = dy / PX_PER_MIN;
   831→    const movedPx = Math.max(drag.movedPx, Math.abs(dy));
   832→    if (drag.mode === 'move') {
   833→      const start = clamp(snap(drag.origStart + dMin), 0, 1440 - drag.dur);
   834→      setDrag({ ...drag, start, movedPx });
   835→    } else {
   836→      const dur = clamp(snap(drag.origDur + dMin), 15, 1440 - drag.start);
   837→      setDrag({ ...drag, dur, movedPx });
   838→    }
   839→  };
   840→
   841→  const onDragEnd = (e: React.PointerEvent<HTMLElement>, cancelled: boolean) => {
   842→    if (!drag || drag.pointerId !== e.pointerId) return;
   843→    const d = drag;
   844→    setDrag(null);
   845→    if (cancelled) return;
   846→    if (d.movedPx < 4) {
   847→      if (d.mode === 'move') {
   848→        if (d.kind === 'item') openItemBlock(d.id);
   849→        else openFileBlock(d.id);
   850→      }
   851→      return;
   852→    }
   853→    if (d.mode === 'move' && d.start === d.origStart) return;
   854→    if (d.mode === 'resize' && d.dur === d.origDur) return;
   855→    if (d.kind === 'item') {
   856→      const patch: Partial<Item> = {
   857→        plannedStart: dateTimeFromMin(date, d.start),
   858→        plannedEnd: dateTimeFromMin(date, d.start + d.dur),
   859→        startDate: date,
   860→      };
   861→      if (d.mode === 'resize') patch.estimateMin = d.dur;
   862→      updateItem(d.id, patch, { snapshot: true });
   863→    } else {
   864→      snapshot();
   865→      updateFilePlan(d.id, { startMin: d.start, minutes: d.dur });
   866→    }
   867→  };
   868→
   869→  const dragFor = (kind: DragKind, id: string): DragState | null =>
   870→    drag && drag.kind === kind && drag.id === id ? drag : null;
   871→
   872→  /* ---------- drag-to-schedule from the backlog ---------- */
   873→  const beginPoolDrag = (
   874→    e: React.PointerEvent<HTMLElement>,
   875→    kind: DragKind,
   876→    id: string,
   877→    title: string,
   878→    dur: number
   879→  ) => {
   880→    if (e.pointerType === 'mouse' && e.button !== 0) return;
   881→    e.stopPropagation();
   882→    try {
   883→      e.currentTarget.setPointerCapture(e.pointerId);
   884→    } catch {
   885→      /* pointer already released — drag simply won't start */
   886→    }
   887→    setPoolDrag({
   888→      kind,
   889→      id,
   890→      title,
   891→      dur,
   892→      pointerId: e.pointerId,
   893→      startX: e.clientX,
   894→      startY: e.clientY,
   895→      movedPx: 0,
   896→      overTimeline: false,
   897→      minute: -1,
   898→    });
   899→  };
   900→
   901→  const onPoolDragMove = (e: React.PointerEvent<HTMLElement>) => {
   902→    if (!poolDrag || poolDrag.pointerId !== e.pointerId) return;
   903→    const movedPx = Math.max(
   904→      poolDrag.movedPx,
   905→      Math.max(Math.abs(e.clientX - poolDrag.startX), Math.abs(e.clientY - poolDrag.startY))
   906→    );
   907→    const rect = lanesRef.current?.getBoundingClientRect();
   908→    if (!rect) {
   909→      setPoolDrag({ ...poolDrag, movedPx, overTimeline: false });
   910→      return;
   911→    }
   912→    const overTimeline =
   913→      e.clientX >= rect.left &&
   914→      e.clientX <= rect.right &&
   915→      e.clientY >= rect.top &&
   916→      e.clientY <= rect.bottom;
   917→    const minute = clamp(
   918→      snapQ(DAY_START_MIN + (e.clientY - rect.top) / PX_PER_MIN),
   919→      0,
   920→      1440 - poolDrag.dur
   921→    );
   922→    setPoolDrag({ ...poolDrag, movedPx, overTimeline, minute });
   923→  };
   924→
   925→  const onPoolDragEnd = (e: React.PointerEvent<HTMLElement>, cancelled: boolean) => {
   926→    if (!poolDrag || poolDrag.pointerId !== e.pointerId) return;
   927→    const d = poolDrag;
   928→    setPoolDrag(null);
   929→    if (cancelled || d.movedPx < 4 || !d.overTimeline) return;
   930→    const start = clamp(d.minute, 0, 1440 - d.dur);
   931→    if (d.kind === 'item') {
   932→      updateItem(
   933→        d.id,
   934→        {
   935→          plannedStart: dateTimeFromMin(date, start),
   936→          plannedEnd: dateTimeFromMin(date, start + d.dur),
   937→          startDate: date,
   938→        },
   939→        { snapshot: true }
   940→      );
   941→      toast('Scheduled', `${d.title} at ${minToHHMM(start)}`);
   942→    } else {
   943→      snapshot();
   944→      addFilePlan(d.id, date, start, d.dur);
   945→      toast('Reading block added', `${d.title} at ${minToHHMM(start)} · ${d.dur}m`);
   946→    }
   947→  };
   948→
   949→  /* ---------- overlap detection (drag-aware, O(n²) — n is small) ---------- */
   950→  const liveBlocks: { key: string; start: number; end: number }[] = [];
   951→  timeboxed.forEach((it) => {
   952→    const g = dragFor('item', it.id);
   953→    const s = g ? g.start : plannedStartToMin(it.plannedStart ?? '');
   954→    const d = g ? g.dur : timeboxDurMin(it);
   955→    liveBlocks.push({ key: `i${it.id}`, start: s, end: s + d });
   956→  });
   957→  fileBlocks.forEach((p) => {
   958→    const g = dragFor('file', p.id);
   959→    const s = g ? g.start : p.startMin;
   960→    const d = g ? g.dur : p.minutes;
   961→    liveBlocks.push({ key: `f${p.id}`, start: s, end: s + d });
   962→  });
   963→  const overlapKeys = new Set<string>();
   964→  for (let a = 0; a < liveBlocks.length; a++) {
   965→    for (let b = a + 1; b < liveBlocks.length; b++) {
   966→      if (liveBlocks[a].start < liveBlocks[b].end && liveBlocks[b].start < liveBlocks[a].end) {
   967→        overlapKeys.add(liveBlocks[a].key);
   968→        overlapKeys.add(liveBlocks[b].key);
   969→      }
   970→    }
   971→  }
   972→
   973→  /* ---------- geometry ---------- */
   974→  const blockTop = (startMin: number) => Math.max(0, Math.round((startMin - DAY_START_MIN) * PX_PER_MIN));
   975→  const blockHeight = (startMin: number, dur: number) =>
   976→    Math.max(24, Math.round((startMin + dur - Math.max(startMin, DAY_START_MIN)) * PX_PER_MIN));
   977→
   978→  const hours = useMemo(() => {
   979→    const out: number[] = [];
   980→    for (let h = DAY_START_MIN / 60; h <= DAY_END_MIN / 60; h++) out.push(h);
   981→    return out;
   982→  }, []);
   983→  const today = todayStr();
   984→
   985→  /* ---------- render ---------- */
   986→  return (
   987→    <div className="plan-day-grid">
   988→      {/* -------- timeline -------- */}
   989→      <div className="card plan-tl-card">
   990→        <div className="plan-tl-head">
   991→          <span className="plan-tl-count">
   992→            {blockCount
   993→              ? `${blockCount} block${blockCount === 1 ? '' : 's'} · ${fmtDuration(plannedMin)} planned`
   994→              : 'Timeline · 06:00 – 23:00'}
   995→          </span>
   996→          <div className="plan-head-actions">
   997→            {blockCount > 0 && (
   998→              <span className={`chip plan-done-chip ${doneCount === blockCount ? 'chip-ok' : 'chip-soon'}`}>
   999→                {doneCount}/{blockCount} done
  1000→              </span>
  1001→            )}
  1002→            <IconBtn small title="Copy yesterday's blocks onto this day" onClick={copyYesterday}>
  1003→              <Copy size={14} />
  1004→            </IconBtn>
  1005→            {blockCount > 0 && (
  1006→              <IconBtn small danger title="Clear all blocks on this day" onClick={clearDay}>
  1007→                <Eraser size={14} />
  1008→              </IconBtn>
  1009→            )}
  1010→            {date === today && (
  1011→              <IconBtn small title="Scroll to now" onClick={scrollToNow}>
  1012→                <Crosshair size={14} />
  1013→              </IconBtn>
  1014→            )}
  1015→          </div>
  1016→        </div>
  1017→        <div className="plan-tl-scroll plan-scroll" ref={scrollRef}>
  1018→          <div className="plan-tl" style={{ height: TIMELINE_H }}>
  1019→            <div className="plan-tl-gutter" aria-hidden>
  1020→              {hours.slice(0, -1).map((h) => (
  1021→                <span key={h} style={{ top: (h * 60 - DAY_START_MIN) * PX_PER_MIN - 7 }}>
  1022→                  {String(h).padStart(2, '0')}:00
  1023→                </span>
  1024→              ))}
  1025→            </div>
  1026→            <div
  1027→              className={`plan-tl-lanes${poolDrag ? ' droptarget' : ''}`}
  1028→              ref={lanesRef}
  1029→              onClick={openQuickAdd}
  1030→            >
  1031→              {hours.map((h) => (
  1032→                <div
  1033→                  key={h}
  1034→                  className="plan-hour-line"
  1035→                  style={{ top: Math.round((h * 60 - DAY_START_MIN) * PX_PER_MIN) }}
  1036→                />
  1037→              ))}
  1038→              {hours.slice(0, -1).map((h) => (
  1039→                <div
  1040→                  key={`h${h}`}
  1041→                  className="plan-half-line"
  1042→                  style={{ top: Math.round((h * 60 - DAY_START_MIN) * PX_PER_MIN + HOUR_PX / 2) }}
  1043→                />
  1044→              ))}
  1045→              {date === today && <NowLine />}
  1046→              {!blockCount && (
  1047→                <div className="plan-empty">
  1048→                  <div>
  1049→                    <div className="plan-empty-title">A clear day</div>
  1050→                    <div className="plan-empty-sub">
  1051→                      click any empty spot to add a block — or pull work in from the backlog
  1052→                    </div>
  1053→                  </div>
  1054→                </div>
  1055→              )}
  1056→
  1057→              {/* item timeboxes */}
  1058→              {timeboxed.map((it) => {
  1059→                const g = dragFor('item', it.id);
  1060→                const startMin = g ? g.start : plannedStartToMin(it.plannedStart ?? '');
  1061→                const durMin = g ? g.dur : timeboxDurMin(it);
  1062→                const clamped = startMin < DAY_START_MIN;
  1063→                const overl = overlapKeys.has(`i${it.id}`);
  1064→                const style: React.CSSProperties = {
  1065→                  top: blockTop(startMin),
  1066→                  height: blockHeight(startMin, durMin),
  1067→                };
  1068→                if (it.color) (style as Record<string, string | number>)['--stripe'] = it.color;
  1069→                return (
  1070→                  <div
  1071→                    key={it.id}
  1072→                    className={`plan-block plan-block-item${it.status === 'done' ? ' done' : ''}${g ? ' dragging' : ''}${overl ? ' plan-overlap' : ''}`}
  1073→                    style={style}
  1074→                    role="button"
  1075→                    tabIndex={0}
  1076→                    aria-label={`${it.title}, ${minToHHMM(startMin)} to ${minToHHMM(startMin + durMin)}${overl ? ', overlaps another block' : ''}`}
  1077→                    onPointerDown={(e) => beginDrag(e, 'item', it.id, 'move', startMin, durMin)}
  1078→                    onPointerMove={onDragMove}
  1079→                    onPointerUp={(e) => onDragEnd(e, false)}
  1080→                    onPointerCancel={(e) => onDragEnd(e, true)}
  1081→                    onKeyDown={(e) => {
  1082→                      if (e.key === 'Enter' || e.key === ' ') {
  1083→                        e.preventDefault();
  1084→                        openItemDetail(it.id);
  1085→                      }
  1086→                    }}
  1087→                  >
  1088→                    <div className="plan-block-time mono">
  1089→                      {clamped && <span aria-hidden>↑ </span>}
  1090→                      {minToHHMM(startMin)}–{minToHHMM(startMin + durMin)}
  1091→                    </div>
  1092→                    <div className="plan-block-title">{it.title}</div>
  1093→                    {overl && (
  1094→                      <span className="plan-overlap-ico" aria-hidden>
  1095→                        <AlertTriangle size={10} />
  1096→                      </span>
  1097→                    )}
  1098→                    <div
  1099→                      className="plan-block-resize"
  1100→                      role="separator"
  1101→                      aria-label="Resize block"
  1102→                      onPointerDown={(e) => beginDrag(e, 'item', it.id, 'resize', startMin, durMin)}
  1103→                    />
  1104→                  </div>
  1105→                );
  1106→              })}
  1107→
  1108→              {/* file reading blocks */}
  1109→              {fileBlocks.map((p) => {

  1386→function WeekPlanner({ date, onOpenDay }: { date: string; onOpenDay: (d: string) => void }) {

  1405→      days.map((d) => {
