    14→/* ---------- Labels manager ---------- */
    15→
    16→export function LabelsModal({ open, onClose }: { open: boolean; onClose: () => void }) {
    17→  const labels = useWB((s) => s.labels);
    18→  const items = useWB((s) => s.items);
    19→  const addLabel = useWB((s) => s.addLabel);
    20→  const renameLabel = useWB((s) => s.renameLabel);
    21→  const deleteLabel = useWB((s) => s.deleteLabel);
    22→  const [name, setName] = useState('');
    23→
    24→  const save = () => {
    25→    const n = name.trim();
    26→    if (!n) {
    27→      toast('Label name is required');
    28→      return;
    29→    }
    30→    addLabel(n);
    31→    setName('');
    32→  };
    33→
    34→  return (
    35→    <Modal
    36→      open={open}
    37→      onClose={onClose}
    38→      title="Manage labels"
    39→      footer={<Btn variant="primary" onClick={onClose}>Done</Btn>}
    40→    >
    41→      <div style={{ display: 'flex', gap: 8, marginBottom: 14 }} onSubmit={(e) => e.preventDefault()}>
    42→        <input
    43→          type="text"
    44→          value={name}
    45→          placeholder="New label name"
    46→          autoFocus
    47→          onChange={(e) => setName(e.target.value)}
    48→          onKeyDown={(e) => e.key === 'Enter' && save()}
    49→          style={{
    50→            flex: 1, padding: '9px 13px', border: '1px solid var(--hairline)',
    51→            borderRadius: 'var(--radius-wb-sm)', background: 'var(--surface-solid)', color: 'var(--ink)',
    52→          }}
    53→        />
    54→        <Btn variant="primary" size="sm" onClick={save}>
    55→          <Plus size={14} /> Add
    56→        </Btn>
    57→      </div>
    58→      {labels.length ? (
    59→        <div style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
    60→          {labels.map((l) => (
    61→            <div key={l.id} className="card" style={{ padding: '11px 13px', margin: 0 }}>
    62→              <div style={{ display: 'flex', alignItems: 'center', gap: 9 }}>
    63→                <span className="chip-dot" style={{ background: l.color, width: 11, height: 11 }} />
    64→                <input
    65→                  type="text"
    66→                  defaultValue={l.name}
    67→                  key={l.id + l.name}
    68→                  onBlur={(e) => {
    69→                    const v = e.target.value.trim();
    70→                    if (v && v !== l.name) {
    71→                      renameLabel(l.id, v);
    72→                      toast('Label renamed', v);
    73→                    }
    74→                  }}
    75→                  onKeyDown={(e) => {
    76→                    if (e.key === 'Enter') (e.target as HTMLInputElement).blur();
    77→                  }}
    78→                  style={{
    79→                    border: 'none', background: 'none', fontWeight: 650, flex: 1,
    80→                    color: 'var(--ink)', fontSize: 13.5, outline: 'none',
    81→                    borderBottom: '1px dashed transparent',
    82→                  }}
    83→                  aria-label={`Rename label ${l.name}`}
    84→                />
    85→                <span style={{ fontSize: 11, color: 'var(--faint)' }} className="mono">
    86→                  {l.itemIds.filter((iid) => items.some((i) => i.id === iid)).length} items
    87→                </span>
    88→                <button
    89→                  className="icon-btn icon-btn-sm icon-btn-danger"
    90→                  title={`Delete label ${l.name}`}
    91→                  onClick={() => {
    92→                    if (confirm(`Delete label "${l.name}"? It will be removed from all entries.`)) {
    93→                      deleteLabel(l.id);
    94→                      toast('Label deleted');
    95→                    }
    96→                  }}
    97→                >
    98→                  <Trash2 size={13} />
    99→                </button>
   100→              </div>
   101→            </div>
   102→          ))}
   103→          <div className="field-help">Click a label name to rename it. Changes save when you click away.</div>
   104→        </div>
   105→      ) : (
   106→        <div className="field-help">No labels yet — add one above, then attach it from an item&apos;s detail view.</div>
   107→      )}
   108→    </Modal>
   109→  );
   110→}

   168→/* ---------- Data (export / import / ICS) ---------- */
   169→
   170→export function DataModal({ open, onClose }: { open: boolean; onClose: () => void }) {
   171→  const exportData = useWB((s) => s.exportData);
   172→  const importData = useWB((s) => s.importData);
   173→  const items = useWB((s) => s.items);
   174→
   175→  const doExport = () => {
   176→    const blob = new Blob([exportData()], { type: 'application/json' });
   177→    const url = URL.createObjectURL(blob);
   178→    const a = document.createElement('a');
   179→    a.href = url;
   180→    a.download = `workbench-aurora-${todayStr()}.json`;
   181→    document.body.appendChild(a);
   182→    a.click();
   183→    document.body.removeChild(a);
   184→    URL.revokeObjectURL(url);
   185→    toast('Backup exported');
   186→  };
   187→
   188→  const doImport = (kind: 'json' | 'ics') => {
   189→    const input = document.createElement('input');
   190→    input.type = 'file';
   191→    input.accept = kind === 'json' ? 'application/json,.json' : '.ics,text/calendar';
   192→    input.onchange = () => {
   193→      const f = input.files?.[0];
   194→      if (!f) return;
   195→      const fr = new FileReader();
   196→      fr.onload = () => {
   197→        if (kind === 'json') {
   198→          try {
   199→            const raw = JSON.parse(String(fr.result));
   200→            if (!confirm('Replace the current workspace with this backup?')) return;
   201→            const err = importData(raw);
   202→            if (err) toast('Import failed', err);
   203→            else toast('Backup imported', 'Your workspace has been restored.');
   204→          } catch (e) {
   205→            toast('Import failed', e instanceof Error ? e.message : 'Invalid file');
   206→          }
   207→        } else {
   208→          const events = parseICS(String(fr.result));
   209→          if (!events.length) {
   210→            toast('No events found', 'The .ics file contained no importable events.');
   211→            return;
   212→          }
   213→          const { addItem } = useWB.getState();
   214→          events.forEach((ev) =>
   215→            addItem({ title: ev.title, type: 'task', status: 'todo', deadline: ev.deadline, plannedStart: ev.plannedStart })
   216→          );
   217→          toast('Calendar imported', `${events.length} event${events.length === 1 ? '' : 's'} converted to tasks`);
   218→        }
   219→      };
   220→      fr.readAsText(f);
   221→    };
   222→    input.click();
   223→  };
   224→
   225→  const doIcsExport = () => {
   226→    const timeboxed = items.filter((i) => i.plannedStart);
   227→    if (!timeboxed.length) {
   228→      toast('Nothing to export', 'Timebox a few tasks first — their blocks become calendar events.');
   229→      return;
   230→    }
   231→    const blob = new Blob([buildICS(timeboxed)], { type: 'text/calendar' });
   232→    const url = URL.createObjectURL(blob);
   233→    const a = document.createElement('a');
   234→    a.href = url;
   235→    a.download = `workbench-timeboxes-${todayStr()}.ics`;
   236→    document.body.appendChild(a);
   237→    a.click();
   238→    document.body.removeChild(a);
   239→    URL.revokeObjectURL(url);
   240→    toast('Calendar exported', `${timeboxed.length} time block${timeboxed.length === 1 ? '' : 's'}`);
   241→  };
   242→
   243→  const rowStyle: React.CSSProperties = {
   244→    display: 'flex', alignItems: 'center', gap: 14, padding: '14px 0',
   245→    borderBottom: '1px solid var(--hairline)',
   246→  };
   247→
   248→  return (
   249→    <Modal
   250→      open={open}
   251→      onClose={onClose}
   252→      title="Backup & data"
   253→      footer={<Btn variant="primary" onClick={onClose}>Done</Btn>}
   254→    >
   255→      <div style={rowStyle}>
   256→        <div style={{ flex: 1 }}>
   257→          <div style={{ fontWeight: 650, fontSize: 13.5 }}>Workspace backup</div>
   258→          <div className="field-help">Full JSON export/import of goals, tasks, entries, habits and journal.</div>
   259→        </div>
   260→        <Btn size="sm" onClick={doExport}><Download size={14} /> Export</Btn>
   261→        <Btn size="sm" onClick={() => doImport('json')}><Upload size={14} /> Import</Btn>
   262→      </div>
   263→      <div style={rowStyle}>
   264→        <div style={{ flex: 1 }}>
   265→          <div style={{ fontWeight: 650, fontSize: 13.5 }}>Calendar bridge</div>
   266→          <div className="field-help">Move timeboxed work in and out using standard .ics files.</div>
   267→        </div>
   268→        <Btn size="sm" onClick={doIcsExport}><Download size={14} /> .ics</Btn>
   269→        <Btn size="sm" onClick={() => doImport('ics')}><Upload size={14} /> .ics</Btn>
   270→      </div>
   271→      <div style={{ ...rowStyle, borderBottom: 'none' }}>
   272→        <div style={{ flex: 1 }}>
   273→          <div style={{ fontWeight: 650, fontSize: 13.5 }}>Storage</div>
   274→          <div className="field-help">Local-first: everything lives in this browser. Export regularly.</div>
   275→        </div>
   276→        <span className="chip chip-ok">Local</span>
   277→      </div>
   278→    </Modal>
   279→  );
   280→}

   358→      </div>
   359→    </Modal>
   360→  );

   363→/* ---------- Timebox ---------- */
   364→
   365→export function TimeboxModal({
   366→  itemId,
   367→  date,
   368→  onClose,
   369→}: {
   370→  itemId: string;
   371→  date?: string;
   372→  onClose: () => void;
   373→}) {
   374→  const items = useWB((s) => s.items);
   375→  const updateItem = useWB((s) => s.updateItem);
   376→  const item = items.find((i) => i.id === itemId) || null;
   377→
   378→  const d = date || item?.startDate || todayStr();
   379→  const currentStart =
   380→    item?.plannedStart && item.plannedStart.slice(0, 10) === d ? item.plannedStart.slice(11, 16) : '09:00';
   381→  const mins = Math.max(15, item?.estimateMin || 30);
   382→  const [h0, m0] = currentStart.split(':').map(Number);
   383→  const x0 = new Date(2000, 0, 1, h0, m0 + mins);
   384→  const defaultEnd = `${String(x0.getHours()).padStart(2, '0')}:${String(x0.getMinutes()).padStart(2, '0')}`;
   385→
   386→  const [day, setDay] = useState(d);
   387→  const [start, setStart] = useState(currentStart);
   388→  const [end, setEnd] = useState(
   389→    item?.plannedEnd && item.plannedEnd.slice(0, 10) === d ? item.plannedEnd.slice(11, 16) : defaultEnd
   390→  );
   391→
   392→  if (!item) return null;
   393→
   394→  const plannedOnDay = items
   395→    .filter((i) => i.id !== item.id && i.plannedStart && i.plannedStart.slice(0, 10) === day)
   396→    .reduce((s, i) => s + (Number(i.estimateMin) || 30), 0);
   397→
   398→  return (
   399→    <Modal
   400→      open
   401→      onClose={onClose}
   402→      title="Timebox task"
   403→      footer={
   404→        <>
   405→          <Btn
   406→            variant="danger"
   407→            onClick={() => {
   408→              updateItem(item.id, { plannedStart: null, plannedEnd: null });
   409→              toast('Timebox cleared');
   410→              onClose();
   411→            }}
   412→          >
   413→            Clear
   414→          </Btn>
   415→          <span className="spacer" />
   416→          <Btn onClick={onClose}>Cancel</Btn>
   417→          <Btn
   418→            variant="primary"
   419→            onClick={() => {
   420→              if (!start || !end || end <= start) {
   421→                toast('End time must be after start');
   422→                return;
   423→              }
   424→              updateItem(item.id, {
   425→                plannedStart: `${day}T${start}`,
   426→                plannedEnd: `${day}T${end}`,
   427→                startDate: day,
   428→              });
   429→              useWB.setState({ calendarSelectedDate: day });
   430→              toast('Time block saved', `${fmtDateNice(day)} · ${start}–${end}`);
   431→              onClose();
   432→            }}
   433→          >
   434→            <Check size={15} /> Save block
   435→          </Btn>
   436→        </>
   437→      }
   438→    >
   439→      <div className="field">
   440→        <label htmlFor="tb-date">Day</label>
   441→        <input id="tb-date" type="date" value={day} onChange={(e) => setDay(e.target.value)} />
   442→      </div>
   443→      <div className="field-row">
   444→        <div className="field">
   445→          <label htmlFor="tb-start">Start</label>
   446→          <input id="tb-start" type="time" value={start} onChange={(e) => setStart(e.target.value)} />
   447→        </div>
   448→        <div className="field">
   449→          <label htmlFor="tb-end">End</label>
   450→          <input id="tb-end" type="time" value={end} onChange={(e) => setEnd(e.target.value)} />
   451→        </div>
   452→      </div>
   453→      <div className="card" style={{ padding: '12px 14px', marginTop: 4 }}>
   454→        <div style={{ fontWeight: 650 }}>{item.title}</div>
   455→        <div className="card-sub" style={{ marginTop: 4 }}>
   456→          Estimated {fmtDuration(mins)} · Other blocks that day: {fmtDuration(plannedOnDay)}
   457→        </div>
   458→      </div>
   459→    </Modal>
   460→  );
   461→}

   465→export function DayPlannerModal({ date, onClose, onTimebox }: { date: string; onClose: () => void; onTimebox: (id: string, date: string) => void }) {
   466→  const items = useWB((s) => s.items);
   467→  const updateItem = useWB((s) => s.updateItem);
   468→
   469→  const blocked = items.filter((i) => i.status !== 'done' && i.status !== 'archived' && wbIsBlocked(items, i));
   470→  const dayItems = items
   471→    .filter((i) => i.plannedStart && i.plannedStart.slice(0, 10) === date && i.status !== 'archived')
   472→    .sort((a, b) => (a.plannedStart || '').localeCompare(b.plannedStart || ''));
   473→  const available = items
   474→    .filter((i) => !['done', 'archived', 'inbox'].includes(i.status) && !wbIsBlocked(items, i) && !i.plannedStart)
   475→    .sort((a, b) => wbPriorityWeight(a.priority) - wbPriorityWeight(b.priority))
   476→    .slice(0, 40);
   477→  const plannedMin = dayItems.reduce((s, i) => s + Math.max(1, Number(i.estimateMin) || 30), 0);
   478→  const [addId, setAddId] = useState('');
   479→
   480→  return (
   481→    <Modal open onClose={onClose} title={`Plan ${fmtDateNice(date)}`} wide>
   482→      <div
   483→        style={{
   484→          display: 'flex', justifyContent: 'space-between', alignItems: 'center',
   485→          padding: '10px 14px', borderRadius: 12, background: 'var(--pine-light)', marginBottom: 14,
   486→        }}
   487→      >
   488→        <strong>{fmtDateNice(date)}</strong>
   489→        <span className="mono" style={{ fontSize: 11, color: 'var(--muted2)' }}>{fmtDuration(plannedMin)} planned</span>
   490→      </div>
   491→      <div style={{ display: 'grid', gap: 8 }}>
   492→        {dayItems.length ? (
   493→          dayItems.map((i) => (
   494→            <div
   495→              key={i.id}
   496→              style={{
   497→                display: 'grid', gridTemplateColumns: '92px minmax(0,1fr) auto', alignItems: 'center',
   498→                gap: 12, padding: 10, border: '1px solid var(--hairline)', borderRadius: 12,
   499→              }}
   500→            >
   501→              <div className="mono" style={{ fontWeight: 700, fontSize: 11, color: 'var(--pine)' }}>
   502→                {i.plannedStart?.slice(11, 16)}–{i.plannedEnd?.slice(11, 16) || '—'}
   503→              </div>
   504→              <div style={{ minWidth: 0 }}>
   505→                <div style={{ fontWeight: 650 }}>{i.title}</div>
   506→                <div className="card-sub">
   507→                  {fmtDuration(i.estimateMin || 30)}
   508→                  {i.nextAction ? ` · ${i.nextAction}` : ''}
   509→                </div>
   510→              </div>
   511→              <Btn size="sm" variant="ghost" onClick={() => onTimebox(i.id, date)}>
   512→                <Pencil size={13} /> Edit
   513→              </Btn>
   514→            </div>
   515→          ))
   516→        ) : (
   517→          <div className="empty-state" style={{ padding: '22px 14px' }}>
   518→            <div className="es-title">No time blocks yet.</div>
   519→          </div>
   520→        )}
   521→      </div>

   552→/* ---------- Daily plan (capacity) ---------- */

   554→export function DailyPlanModal({ onClose }: { onClose: () => void }) {
   555→  const items = useWB((s) => s.items);
   556→  const dailyCapacityMin = useWB((s) => s.dailyCapacityMin);
   557→  const setDailyCapacity = useWB((s) => s.setDailyCapacity);
   558→  const updateItem = useWB((s) => s.updateItem);
   559→
   560→  const today = todayStr();
   561→  const planned = wbPlannedItems(items, today);
   562→  const used = wbPlannedMinutes(items, today);
   563→  const cap = dailyCapacityMin || 240;
   564→  const ready = wbReadyItems(items).filter((i) => i.type === 'task' && !i.startDate);
   565→  const [capacity, setCapacity] = useState(String(cap));
   566→
   567→  return (
   568→    <Modal
   569→      open
   570→      onClose={onClose}
   571→      title="Plan today"
   572→      wide
   573→      footer={
   574→        <>
   575→          <Btn
   576→            onClick={() => {
   577→              // auto-fill: add ready tasks by priority until capacity
   578→              let total = used;
   579→              const c = Number(capacity) || 240;
   580→              ready
   581→                .slice()
   582→                .sort((a, b) => wbPriorityWeight(a.priority) - wbPriorityWeight(b.priority))
   583→                .forEach((i) => {
   584→                  const m = Number(i.estimateMin) || 30;
   585→                  if (total + m <= c) {
   586→                    updateItem(i.id, { startDate: today, status: i.status === 'inbox' ? 'todo' : i.status });
   587→                    total += m;
   588→                  }
   589→                });
   590→              toast('Plan auto-filled', `${fmtDuration(Math.min(total, c))} of ${fmtDuration(c)} capacity`);
   591→            }}
   592→          >
   593→            Auto-fill capacity
   594→          </Btn>
   595→          <span className="spacer" />
   596→          <Btn variant="primary" onClick={() => { setDailyCapacity(Number(capacity) || 240); onClose(); }}>
   597→            <Check size={15} /> Done
   598→          </Btn>
   599→        </>
   600→      }
   601→    >
   602→      <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', gap: 12, flexWrap: 'wrap', marginBottom: 10 }}>
   603→        <div>
   604→          <div style={{ fontWeight: 650 }}>Today&apos;s capacity</div>
   605→          <div className="field-help">Keep the plan below what you can realistically finish.</div>
   606→        </div>
   607→        <select value={capacity} onChange={(e) => setCapacity(e.target.value)} style={{ minWidth: 120 }}>
   608→          {[120, 180, 240, 300, 360, 420, 480].map((m) => (
   609→            <option key={m} value={m}>{fmtDuration(m)}</option>
   610→          ))}
   611→        </select>
   612→      </div>
   613→      <div className={`capacity-track ${used > cap ? 'over' : ''}`}>
   614→        <span style={{ width: `${Math.min(100, (used / cap) * 100)}%` }} />
   615→      </div>
   616→      <div className="field-help" style={{ margin: '6px 0 16px' }}>
   617→        Planned {fmtDuration(used)} · {fmtDuration(Math.max(0, cap - used))} remaining
   618→      </div>
   619→
   620→      <div className="section-title" style={{ marginTop: 0 }}>Planned today · {planned.length}</div>
   621→      <div style={{ display: 'flex', flexDirection: 'column', gap: 8, marginBottom: 16 }}>
   622→        {planned.length ? (
   623→          planned.map((i) => (
   624→            <div key={i.id} style={{ display: 'flex', alignItems: 'center', gap: 10, padding: '9px 12px', border: '1px solid var(--hairline)', borderRadius: 12 }}>
   625→              <div style={{ flex: 1, minWidth: 0 }}>
   626→                <div style={{ fontWeight: 650, fontSize: 13.5 }}>{i.title}</div>
   627→                <div className="card-sub">{i.nextAction || 'No next action'} · {fmtDuration(i.estimateMin || 0)}</div>
   628→              </div>
   629→              <Btn size="sm" variant="ghost" onClick={() => updateItem(i.id, { startDate: null })}>
   630→                Unplan
   631→              </Btn>
   632→            </div>
   633→          ))
   634→        ) : (
   635→          <div className="field-help">Nothing planned yet.</div>
   636→        )}
   637→      </div>
   638→
   639→      <div className="section-title">Ready to plan · {ready.length}</div>
   640→      <div style={{ display: 'flex', flexDirection: 'column', gap: 8, maxHeight: 280, overflowY: 'auto', paddingRight: 4 }}>
   641→        {ready.length ? (
   642→          ready.map((i) => (
   643→            <label key={i.id} style={{ display: 'flex', alignItems: 'center', gap: 10, padding: '9px 12px', border: '1px solid var(--hairline)', borderRadius: 12, cursor: 'pointer' }}>
   644→              <input
   645→                type="checkbox"
   646→                checked={i.startDate === today}
   647→                onChange={() => updateItem(i.id, { startDate: i.startDate === today ? null : today })}
   648→                style={{ accentColor: 'var(--pine)' }}
   649→              />
   650→              <div style={{ flex: 1, minWidth: 0 }}>
   651→                <div style={{ fontWeight: 650, fontSize: 13.5 }}>{i.title}</div>
   652→                <div className="card-sub">
   653→                  {priorityLabel(i.priority)} · {fmtDuration(i.estimateMin || 30)}
   654→                  {i.deadline ? ` · due ${fmtDateNice(i.deadline)}` : ''}
   655→                </div>
   656→              </div>
   657→            </label>
   658→          ))
   659→        ) : (
   660→          <div className="field-help">No ready tasks need planning.</div>
   661→        )}
   662→      </div>
   663→    </Modal>
   664→  );
   665→}
