   572→   572→    });

   574→   574→      if (!willOpen === prev.has(id)) return prev;
   575→   575→      const n = new Set(prev);
   576→   576→      if (willOpen) n.delete(id);
   577→   577→      else n.add(id);
   578→   578→      return n;
   579→   579→    });
   580→   580→  };

   599→   599→      document.removeEventListener('mousedown', onDown);
   600→   600→      document.removeEventListener('keydown', onKey, true);

   659→   659→  /* drag & drop: file cards carry text/plain = file id; folder rows carry their folder id */
   660→   660→  const onRowDragOver = (e: React.DragEvent, id: string) => {
   661→   661→    const types = Array.from(e.dataTransfer.types);
   662→   662→    if (!types.includes('text/plain')) return;

   664→   664→    if (id === 'unfiled' && types.includes('application/x-wb-folder')) return;

   678→   678→      e.preventDefault();
   679→   679→      return;

   681→   681→    e.dataTransfer.setData('text/plain', folder.id);
   682→   682→    try {
   683→   683→      e.dataTransfer.setData('application/x-wb-folder', folder.id);
   684→   684→    } catch {

   686→   686→    }

   690→   690→  const handleDrop = (e: React.DragEvent, rowId: string) => {
   691→   691→    if (!Array.from(e.dataTransfer.types).includes('text/plain')) return;
   692→   692→    e.preventDefault();
   693→   693→    e.stopPropagation();

   696→   696→    if (!fid) return;
   697→   697→    const isFile = files.some((f) => f.id === fid);
   698→   698→    const isFolder = !isFile && folders.some((f) => f.id === fid);
   699→   699→    if (isFile) {
   700→   700→      const file = files.find((f) => f.id === fid);
   701→   701→      if (!file) return;

   703→   703→      if ((file.folderId ?? null) === folderId) return;

   705→   705→      const folder = folderId ? folders.find((x) => x.id === folderId) : undefined;
   706→   706→      toast(folder ? 'Moved to folder' : 'Removed from folder', folder ? folder.name : file.title);
   707→   707→      return;
   708→   708→    }
   709→   709→    if (isFolder) {
   710→   710→      if (rowId === 'unfiled') return; // folders can't be unfiled
   711→   711→      if (rowId !== 'all' && (descendants.get(fid)?.has(rowId) ?? false)) return; // own descendant → silent
   712→   712→      const targetId = rowId === 'all' ? null : rowId;
   713→   713→      if ((folders.find((x) => x.id === fid)?.parentId ?? null) === targetId) return; // same parent → no-op
   714→   714→      const from = folders.find((x) => x.id === fid)?.name ?? 'Folder';
   715→   715→      moveFolder(fid, targetId);
   716→   716→      if (targetId) {
   717→   717→        forceExpand(targetId);
   718→   718→        toast('Folder moved', `${from} → ${folders.find((x) => x.id === targetId)?.name ?? ''}`);
   719→   719→      } else {
   720→   720→        toast('Folder moved to root', from);
   721→   721→      }
   722→   722→    }
   723→   723→  };

   749→   749→            dropId === folder.id ? 'drop' : ''

   757→   757→            if (renameId === folder.id) return;

   759→   759→          }}
   760→   760→          onDoubleClick={() => startRename(folder)}

   796→   796→              if (e.key === 'Enter' || e.key === ' ') {

   798→   798→                e.stopPropagation();

   817→   817→            />
   818→   818→          ) : (
   819→   819→            <span className="lib-tree-name">{folder.name}</span>
   820→   820→          )}
   821→   821→          <span className="lib-folder-count">{n}</span>
   822→   822→          <span className="lib-tree-acts">
   823→   823→            <button
   824→   824→              type="button"

   895→   895→          <ListTree size={13} aria-hidden="true" /> Folders

   897→   897→        <button
