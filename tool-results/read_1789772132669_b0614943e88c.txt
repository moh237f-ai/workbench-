   234→   234→/** positions lack meaning when everything sits at the origin or in one card-sized pile */
   235→   235→function needsAutoLayout(nodes: RoadmapNode[]): boolean {
   236→   236→  if (nodes.length < 2) return false;
   237→   237→  if (nodes.every((n) => Math.abs(n.x) < 1 && Math.abs(n.y) < 1)) return true;
   238→   238→  let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
   239→   239→  nodes.forEach((n) => {
   240→   240→    minX = Math.min(minX, n.x);
   241→   241→    minY = Math.min(minY, n.y);
   242→   242→    maxX = Math.max(maxX, n.x);
   243→   243→    maxY = Math.max(maxY, n.y);
   244→   244→  });
   245→   245→  return maxX - minX < RM_NODE_W * 0.75 && maxY - minY < RM_NODE_H * 0.75;
   246→   246→}
