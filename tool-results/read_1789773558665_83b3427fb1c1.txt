    72→    72→function mixRgb(a: RGB, b: RGB, t: number): RGB {
    73→    73→  const c = (i: number): number => Math.round(a[i] + (b[i] - a[i]) * t);
    74→    74→  return [c(0), c(1), c(2)];
    75→    75→}

    86→    86→function pageSurfaceBg(mode: ReaderThemeMode, custom: { brightness: number; sepia: number }): string {
    87→    87→  const white: RGB = [255, 255, 255];
    88→    88→  switch (mode) {
    89→    89→    case 'dark':
    90→    90→      return '#15181a';
    91→    91→    case 'warm':
    92→    92→      return '#f3ecdc';
    93→    93→    case 'custom': {
    94→    94→      let c = mixRgb(white, [0xef, 0xe3, 0xcd], clamp01v(custom.sepia / 100));
    95→    95→      const dim = clamp01v((100 - custom.brightness) / 100);
    96→    96→      c = mixRgb(c, [0x16, 0x19, 0x18], dim * 0.9);
    97→    97→      return rgbCss(c);
    98→    98→    }
    99→    99→    default:
   100→   100→      return '#ffffff';
   101→   101→  }
   102→   102→}

   104→   104→function clampPage(p: number, total: number): number {
   105→   105→  const max = Math.max(1, total);
   106→   106→  const n = Math.round(p);
   107→   107→  return Math.max(1, Math.min(max, Number.isFinite(n) ? n : 1));
   108→   108→}

   116→   116→  switch (mode) {
