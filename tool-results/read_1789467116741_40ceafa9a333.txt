   253→   253→function deleteCascade(
   254→   254→  items: Item[],
   255→   255→  labels: Label[],
   256→   256→  expanded: Record<string, boolean>,
   257→   257→  id: string
   258→   258→): Item[] {
   259→   259→  const desc = (iid: string): string[] => {
   260→   260→    const out: string[] = [];
   261→   261→    items.filter((i) => i.parentId === iid).forEach((c) => {
   262→   262→      out.push(c.id);
   263→   263→      out.push(...desc(c.id));
   264→   264→    });
   265→   265→    return out;
   266→   266→  };
   267→   267→  const ids = [id, ...desc(id)];
   268→   268→  labels.forEach((l) => {
   269→   269→    l.itemIds = l.itemIds.filter((x) => !ids.includes(x));
   270→   270→  });
   271→   271→  ids.forEach((i) => delete expanded[i]);
   272→   272→  return items.filter((i) => !ids.includes(i.id));
   273→   273→}

  1026→  1026→export function wbCanDependOn(items: Item[], itemId: string, candidateId: string): boolean {
  1027→  1027→  if (!candidateId || candidateId === itemId) return false;
  1028→  1028→  if (wbDescendantIds(items, itemId).includes(candidateId)) return false;
  1029→  1029→  if (!wbItem(items, candidateId)) return false;
  1030→  1030→  // no cycles: candidate must not (transitively) depend on itemId
  1031→  1031→  const reaches = (startId: string, targetId: string, seen = new Set<string>()): boolean => {
  1032→  1032→    if (startId === targetId) return true;
  1033→  1033→    if (seen.has(startId)) return false;
  1034→  1034→    seen.add(startId);
  1035→  1035→    const it = wbItem(items, startId);
  1036→  1036→    return !!it && (it.dependsOn || []).some((d) => reaches(d, targetId, seen));
  1037→  1037→  };
  1038→  1038→  return !reaches(candidateId, itemId);
  1039→  1039→}

  1064→  1064→    source
  1065→  1065→      .filter((i) => i.type !== 'goal' || wbChildren(items, i.id).length === 0)
  1066→  1066→      .sort((a, b) => {
  1067→  1067→        const p = wbPriorityWeight(a.priority) - wbPriorityWeight(b.priority);
  1068→  1068→        if (p) return p;

  1070→  1070→        const bd = b.deadline ? new Date(b.deadline).getTime() : Infinity;

  1086→  1086→  if (blocked) a.push(`${blocked} item${blocked === 1 ? ' is' : 's are'} blocked by dependencies. Resolve them before adding more work.`);
  1087→  1087→  if (planned > capacity) a.push(`Today is over capacity by ${planned - capacity} minutes. Trim the plan before starting another block.`);
