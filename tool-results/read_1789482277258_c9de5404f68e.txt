    74→export function nodeLive(node: RoadmapNode, items: Item[], files: ResourceFile[]): RmNodeLive {
    75→  const base: RmNodeLive = {
    76→    title: node.label,
    77→    missing: false,
    78→    status: 'todo',
    79→    pct: -1,
    80→    deadline: null,
    81→    deadlineTone: null,
    82→    estMin: 0,
    83→    done: false,
    84→    pagesRead: 0,
    85→    pagesTotal: 0,
    86→  };
    87→
    88→  if (node.kind === 'goal' || node.kind === 'task') {
    89→    const item = items.find((i) => i.id === node.refId);
    90→    if (!item) return { ...base, missing: true, title: node.label || 'Deleted item' };
    91→    const pct = computeProgress(item, (id) => wbChildren(items, id));
    92→    const blocked = wbIsBlocked(items, item);
    93→    const done = item.status === 'done' || (pct >= 100 && item.status !== 'inbox');
    94→    const status: RmNodeStatus = done
    95→      ? 'done'
    96→      : blocked || item.status === 'waiting'
    97→        ? 'blocked'
    98→        : item.status === 'in_progress'
    99→          ? 'progress'
   100→          : 'todo';
   101→    return {
   102→      ...base,
   103→      title: item.title,
   104→      status,
   105→      pct,
   106→      deadline: item.deadline,
   107→      deadlineTone: deadlineStatus(item, pct),
   108→      estMin: item.estimateMin || 30,
   109→      done,
   110→    };
   111→  }
   112→
   113→  if (node.kind === 'file') {
   114→    const f = files.find((x) => x.id === node.refId);
   115→    if (!f) return { ...base, missing: true, title: node.label || 'Deleted file' };
   116→    const pct = wbFileProgress(f);
   117→    const read = wbFileReadPages(f);
   118→    const remaining = Math.max(0, f.pageCount - read);
   119→    const avgSec = wbFileAvgSecPerPage(f);
   120→    const perPageMin = avgSec > 0 ? Math.max(1, Math.round(avgSec / 60)) : 1; // fallback: 1 min/page
   121→    const tone = wbFileDeadlineStatus(f);
   122→    const status: RmNodeStatus = pct >= 100 ? 'done' : tone === 'over' ? 'blocked' : tone === 'soon' ? 'progress' : 'todo';
   123→    return {
   124→      ...base,
   125→      title: f.title,
   126→      status,
   127→      pct,
   128→      deadline: f.deadline,
   129→      deadlineTone: tone,
   130→      estMin: remaining * perPageMin,
   131→      done: pct >= 100,
   132→      pagesRead: read,
   133→      pagesTotal: f.pageCount,
   134→    };
   135→  }
   136→
   137→  if (node.kind === 'milestone') {
   138→    const done = node.done;
   139→    let tone: RmNodeLive['deadlineTone'] = null;
   140→    if (node.milestoneDate) {
   141→      if (done) tone = 'done';
   142→      else {
   143→        const du = daysUntil(node.milestoneDate);
   144→        tone = du < 0 ? 'over' : du <= 3 ? 'soon' : 'ok';
   145→      }
   146→    }
   147→    return {
   148→      ...base,
   149→      title: node.label || 'Milestone',
   150→      status: done ? 'done' : 'todo',
   151→      pct: done ? 100 : -1,
   152→      deadline: node.milestoneDate,
   153→      deadlineTone: tone,
   154→      estMin: 0,
   155→      done,
   156→    };
   157→  }
   158→
   159→  // note
   160→  return { ...base, title: node.label || 'Note', status: 'none' };
   161→}

   163→/* ---------- topological order (Kahn, cycle-safe) ---------- */
   164→
   165→export function topoOrder(nodes: RoadmapNode[], edges: RoadmapEdge[]): string[] {
   166→  const ids = nodes.map((n) => n.id);
   167→  if (!ids.length) return [];
   168→  const idSet = new Set(ids);
   169→  const indeg = new Map<string, number>();
   170→  const adj = new Map<string, string[]>();
   171→  ids.forEach((id) => {
   172→    indeg.set(id, 0);
   173→    adj.set(id, []);
   174→  });
   175→  edges.forEach((e) => {
   176→    if (!idSet.has(e.from) || !idSet.has(e.to) || e.from === e.to) return;
   177→    indeg.set(e.to, (indeg.get(e.to) ?? 0) + 1);
   178→    adj.get(e.from)!.push(e.to);
   179→  });
   180→
   181→  const orderIndex = new Map(ids.map((id, i) => [id, i]));
   182→  const ready: string[] = [];
   183→  const pushReady = (id: string) => {
   184→    ready.push(id);
   185→    ready.sort((a, b) => (orderIndex.get(a) ?? 0) - (orderIndex.get(b) ?? 0));
   186→  };
   187→  ids.forEach((id) => {
   188→    if ((indeg.get(id) ?? 0) === 0) pushReady(id);
   189→  });
   190→
   191→  const out: string[] = [];
   192→  const emitted = new Set<string>();
   193→  while (ready.length) {
   194→    const id = ready.shift()!;
   195→    out.push(id);
   196→    emitted.add(id);
   197→    for (const t of adj.get(id) ?? []) {
   198→      const d = (indeg.get(t) ?? 0) - 1;
   199→      indeg.set(t, d);
   200→      if (d === 0) pushReady(t);
   201→    }
   202→  }
   203→
   204→  // cycle remains → append leftover nodes by x-position (never crash)
   205→  if (out.length < ids.length) {
   206→    const byId = new Map(nodes.map((n) => [n.id, n]));
   207→    const rest = ids.filter((id) => !emitted.has(id));
   208→    rest.sort((a, b) => (byId.get(a)?.x ?? 0) - (byId.get(b)?.x ?? 0));
   209→    out.push(...rest);
   210→  }
   211→  return out;
   212→}

   226→export interface RoadmapCanvasProps {
   227→  roadmap: Roadmap;
   228→  items: Item[];

   242→const snap = (v: number) => Math.round(v / 4) * 4;
   243→const f = (n: number) => n.toFixed(1);
   244→
   245→const RoadmapCanvas = forwardRef<RoadmapCanvasHandle, RoadmapCanvasProps>(function RoadmapCanvas(props, ref) {
   246→  const {
   247→    roadmap, items, files, selection, onSelect, linkMode, linkSource, onLinkSource, onLink,
   248→    onMoveNode, onToggleMilestone, onZoomChange, fitKey,
   249→  } = props;
   250→
   251→  const wrapRef = useRef<HTMLDivElement>(null);
   252→  const [viewport, setViewport] = useState<Viewport>({ x: 0, y: 0, zoom: 1 });
   253→  const [size, setSize] = useState<{ w: number; h: number }>({ w: 0, h: 0 });
   254→  const [drag, setDrag] = useState<{ id: string; x: number; y: number } | null>(null);
   255→  const [panning, setPanning] = useState(false);
   256→
   257→  /* latest-value refs (read inside imperative handle + native listeners) */
   258→  const viewportRef = useRef(viewport);
   259→  const sizeRef = useRef(size);
   260→  const roadmapRef = useRef(roadmap);
   261→  const onZoomChangeRef = useRef(onZoomChange);
   262→
   263→  /* keep refs in sync after render (compiler-safe) */
   264→  useEffect(() => {
   265→    viewportRef.current = viewport;
   266→  }, [viewport]);
   267→  useEffect(() => {
   268→    sizeRef.current = size;
   269→  }, [size]);
   270→  useEffect(() => {
   271→    roadmapRef.current = roadmap;
   272→  }, [roadmap]);
   273→  useEffect(() => {
   274→    onZoomChangeRef.current = onZoomChange;
   275→  }, [onZoomChange]);
   276→
   277→  const dragInfoRef = useRef<{ id: string; px: number; py: number; ox: number; oy: number; moved: boolean } | null>(null);
   278→  const panInfoRef = useRef<{ px: number; py: number; x: number; y: number } | null>(null);
   279→  const didFitRef = useRef(false);
   280→
   281→  /* live data per node — recomputed when entities or nodes change */
   282→  const liveById = useMemo(() => {
   283→    const m = new Map<string, RmNodeLive>();
   284→    roadmap.nodes.forEach((n) => m.set(n.id, nodeLive(n, items, files)));
   285→    return m;
   286→  }, [roadmap.nodes, items, files]);
   287→
   288→  const nodeById = useMemo(() => new Map(roadmap.nodes.map((n) => [n.id, n])), [roadmap.nodes]);
   289→
   290→  /* selected / dragging / link-source node renders last (on top) */
   291→  const renderNodes = useMemo(() => {
   292→    const topId = drag?.id ?? (selection?.kind === 'node' ? selection.id : null) ?? linkSource;
   293→    const rest: RoadmapNode[] = [];
   294→    const top: RoadmapNode[] = [];
   295→    roadmap.nodes.forEach((n) => (n.id === topId ? top.push(n) : rest.push(n)));
   296→    return [...rest, ...top];
   297→  }, [roadmap.nodes, drag, selection, linkSource]);
   298→
   299→  const posOf = (n: RoadmapNode): { x: number; y: number } =>
   300→    drag && drag.id === n.id ? { x: drag.x, y: drag.y } : { x: n.x, y: n.y };
   301→
   302→  /* ---------- measure ---------- */
   303→  useEffect(() => {
   304→    const el = wrapRef.current;
   305→    if (!el) return;
   306→    const ro = new ResizeObserver((entries) => {
   307→      const cr = entries[0]?.contentRect;
   308→      if (cr) setSize({ w: cr.width, h: cr.height });
   309→    });
   310→    ro.observe(el);
   311→    return () => ro.disconnect();
   312→  }, []);
   313→
   314→  /* ---------- zoom (wheel, cursor-anchored) ---------- */
   315→  useEffect(() => {
   316→    const el = wrapRef.current;
   317→    if (!el) return;
   318→    const onWheel = (e: WheelEvent) => {
   319→      e.preventDefault();
   320→      const rect = el.getBoundingClientRect();
   321→      const mx = e.clientX - rect.left;
   322→      const my = e.clientY - rect.top;
   323→      setViewport((v) => {
   324→        const nz = clamp(v.zoom * Math.exp(-e.deltaY * 0.0016), ZOOM_MIN, ZOOM_MAX);
   325→        if (nz === v.zoom) return v;
   326→        const wx = (mx - v.x) / v.zoom;
   327→        const wy = (my - v.y) / v.zoom;
   328→        return { x: mx - wx * nz, y: my - wy * nz, zoom: nz };
   329→      });
   330→    };
   331→    el.addEventListener('wheel', onWheel, { passive: false });
   332→    return () => el.removeEventListener('wheel', onWheel);
   333→  }, []);
   334→
   335→  useEffect(() => {
   336→    onZoomChangeRef.current?.(viewport.zoom);
   337→  }, [viewport.zoom]);
   338→
   339→  /* ---------- imperative handle ---------- */
   340→
   341→  const fitView = useCallback(() => {
   342→    const { w, h } = sizeRef.current;
   343→    const nodes = roadmapRef.current.nodes;
   344→    if (!nodes.length || !w || !h) {
   345→      setViewport({ x: 0, y: 0, zoom: 1 });
   346→      return;
   347→    }
   348→    let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
   349→    nodes.forEach((n) => {
   350→      minX = Math.min(minX, n.x);
   351→      minY = Math.min(minY, n.y);
   352→      maxX = Math.max(maxX, n.x + RM_NODE_W);
   353→      maxY = Math.max(maxY, n.y + RM_NODE_H);
   354→    });
   355→    const bw = Math.max(1, maxX - minX);
   356→    const bh = Math.max(1, maxY - minY);
   357→    // On narrow screens don't shrink below a readable zoom — users pan instead.
   358→    const minFit = w < 640 ? 0.62 : ZOOM_MIN;
   359→    const zoom = clamp(Math.min((w - 90) / bw, (h - 90) / bh), minFit, 1.5);
   360→    setViewport({
   361→      x: (w - bw * zoom) / 2 - minX * zoom,
   362→      y: (h - bh * zoom) / 2 - minY * zoom,
   363→      zoom,
   364→    });
   365→  }, []);
   366→
   367→  const centerNode = useCallback((nodeId: string) => {
   368→    const { w, h } = sizeRef.current;
   369→    const n = roadmapRef.current.nodes.find((x) => x.id === nodeId);
   370→    if (!n || !w || !h) return;
   371→    setViewport((v) => ({
   372→      x: w / 2 - (n.x + RM_NODE_W / 2) * v.zoom,
   373→      y: h / 2 - (n.y + RM_NODE_H / 2) * v.zoom,
   374→      zoom: v.zoom,
   375→    }));
   376→  }, []);
   377→
   378→  const zoomBy = useCallback((factor: number) => {
   379→    const { w, h } = sizeRef.current;
   380→    setViewport((v) => {
   381→      const nz = clamp(v.zoom * factor, ZOOM_MIN, ZOOM_MAX);
   382→      if (nz === v.zoom) return v;
   383→      const wx = (w / 2 - v.x) / v.zoom;
   384→      const wy = (h / 2 - v.y) / v.zoom;
   385→      return { x: w / 2 - wx * nz, y: h / 2 - wy * nz, zoom: nz };
   386→    });
   387→  }, []);
   388→
   389→  const dropSpot = useCallback(() => {
   390→    const { w, h } = sizeRef.current;
   391→    const v = viewportRef.current;
   392→    const n = roadmapRef.current.nodes.length;
   393→    if (!w || !h) return { x: 80 + (n % 4) * 260, y: 80 + Math.floor(n / 4) * 140 };
   394→    // stagger grid around the world coords of the current view center
   395→    const cx = (w / 2 - v.x) / v.zoom;
   396→    const cy = (h / 2 - v.y) / v.zoom;
   397→    return {
   398→      x: Math.round(cx + (n % 4) * 260 - RM_NODE_W / 2),
   399→      y: Math.round(cy + Math.floor(n / 4) * 140 - RM_NODE_H / 2),
   400→    };
   401→  }, []);
   402→
   403→  useImperativeHandle(ref, () => ({ fitView, centerNode, zoomBy, dropSpot }), [fitView, centerNode, zoomBy, dropSpot]);
   404→
   405→  /* auto-fit once the canvas gets its first real size */
   406→  useEffect(() => {
   407→    if (didFitRef.current || !size.w || !size.h) return;
   408→    didFitRef.current = true;
   409→    if (roadmapRef.current.nodes.length > 0) {
   410→      const raf = requestAnimationFrame(() => fitView());
   411→      return () => cancelAnimationFrame(raf);
   412→    }
   413→  }, [size.w, size.h, fitView]);
   414→
   415→  /* fit on demand (after auto-layout etc.) */
   416→  useEffect(() => {
   417→    if (fitKey > 0) {
   418→      const raf = requestAnimationFrame(() => fitView());
   419→      return () => cancelAnimationFrame(raf);
   420→    }
   421→  }, [fitKey, fitView]);
   422→
   423→  /* ---------- background pan ---------- */
   424→
   425→  const onBgPointerDown = (e: React.PointerEvent<SVGRectElement>) => {
   426→    if (e.pointerType === 'mouse' && e.button !== 0) return;
   427→    e.preventDefault();
   428→    const v = viewportRef.current;
   429→    panInfoRef.current = { px: e.clientX, py: e.clientY, x: v.x, y: v.y };
   430→    setPanning(true);
   431→    e.currentTarget.setPointerCapture(e.pointerId);
   432→    onLinkSource(null); // background cancels the link source
   433→    onSelect(null); // background click clears the selection
   434→  };
   435→  const onBgPointerMove = (e: React.PointerEvent<SVGRectElement>) => {
   436→    const p = panInfoRef.current;
   437→    if (!p) return;
   438→    setViewport((v) => ({ ...v, x: p.x + (e.clientX - p.px), y: p.y + (e.clientY - p.py) }));
   439→  };
   440→  const endPan = (e: React.PointerEvent<SVGRectElement>) => {
   441→    panInfoRef.current = null;
   442→    setPanning(false);
   443→    if (e.currentTarget.hasPointerCapture(e.pointerId)) e.currentTarget.releasePointerCapture(e.pointerId);
   444→  };
   445→
   446→  /* ---------- node drag / select / link ---------- */
   447→
   448→  const onNodePointerDown = (e: React.PointerEvent<HTMLDivElement>, node: RoadmapNode) => {
   449→    if (e.pointerType === 'mouse' && e.button !== 0) return;
   450→    e.stopPropagation();
   451→    e.preventDefault();
   452→    if (linkMode) {
   453→      // capture semantics — clicking never drags in link mode
   454→      if (!linkSource) onLinkSource(node.id);
   455→      else if (linkSource === node.id) onLinkSource(null);
   456→      else onLink(linkSource, node.id);
   457→      return;
   458→    }
   459→    dragInfoRef.current = { id: node.id, px: e.clientX, py: e.clientY, ox: node.x, oy: node.y, moved: false };
   460→    e.currentTarget.setPointerCapture(e.pointerId);
   461→  };
   462→
   463→  const onNodePointerMove = (e: React.PointerEvent<HTMLDivElement>) => {
   464→    const d = dragInfoRef.current;
   465→    if (!d) return;
   466→    const zoom = viewportRef.current.zoom;
   467→    const dx = (e.clientX - d.px) / zoom;
   468→    const dy = (e.clientY - d.py) / zoom;
   469→    if (!d.moved && Math.abs(dx) < 3 && Math.abs(dy) < 3) return;
   470→    d.moved = true;
   471→    setDrag({ id: d.id, x: snap(d.ox + dx), y: snap(d.oy + dy) });
   472→  };
   473→
   474→  const onNodePointerUp = (e: React.PointerEvent<HTMLDivElement>, node: RoadmapNode) => {
   475→    const d = dragInfoRef.current;
   476→    dragInfoRef.current = null;
   477→    if (e.currentTarget.hasPointerCapture(e.pointerId)) e.currentTarget.releasePointerCapture(e.pointerId);
   478→    if (!d) return;
   479→    if (d.moved) {
   480→      const cur = drag && drag.id === d.id ? drag : { x: snap(d.ox), y: snap(d.oy) };
   481→      onMoveNode(d.id, cur.x, cur.y);
   482→      setDrag(null);
   483→    } else {
   484→      onSelect({ kind: 'node', id: node.id });
   485→    }
   486→  };
   487→
   488→  const onNodePointerCancel = () => {
   489→    dragInfoRef.current = null;
   490→    setDrag(null);
   491→  };
   492→
   493→  const onNodeKeyDown = (e: React.KeyboardEvent<HTMLDivElement>, node: RoadmapNode) => {
   494→    if (e.key !== 'Enter' && e.key !== ' ') return;
   495→    e.preventDefault();
   496→    if (linkMode) {
   497→      if (!linkSource) onLinkSource(node.id);
   498→      else if (linkSource !== node.id) onLink(linkSource, node.id);
   499→      return;
   500→    }
   501→    onSelect({ kind: 'node', id: node.id });
   502→  };
   503→
   504→  /* ---------- render ---------- */
   505→
   506→  return (
   507→    <div
   508→      ref={wrapRef}
   509→      className={`rm-canvas-wrap${panning ? ' is-panning' : ''}${linkMode ? ' linking' : ''}`}
   510→    >
   511→      <svg className="rm-canvas-svg" role="img" aria-label={`Roadmap canvas: ${roadmap.title}`}>
   512→        <defs>
   513→          <pattern
   514→            id="rm-dots"
   515→            width={24}
   516→            height={24}
   517→            patternUnits="userSpaceOnUse"
   518→            patternTransform={`translate(${f(viewport.x)} ${f(viewport.y)}) scale(${viewport.zoom.toFixed(3)})`}
   519→          >
   520→            <circle cx={1.4} cy={1.4} r={1.4} className="rm-dot" />
   521→          </pattern>
   522→          <marker
   523→            id="rm-arrow"
   524→            viewBox="0 0 10 10"
   525→            refX={8.5}
   526→            refY={5}
   527→            markerWidth={9}
   528→            markerHeight={9}
   529→            orient="auto"
   530→            markerUnits="userSpaceOnUse"
   531→          >
   532→            <path d="M 0 1 L 9 5 L 0 9 z" className="rm-arrow" />
   533→          </marker>
   534→          <marker
   535→            id="rm-arrow-sel"
   536→            viewBox="0 0 10 10"
   537→            refX={8.5}
   538→            refY={5}
   539→            markerWidth={9}
   540→            markerHeight={9}
   541→            orient="auto"
   542→            markerUnits="userSpaceOnUse"
   543→          >
   544→            <path d="M 0 1 L 9 5 L 0 9 z" className="rm-arrow-sel" />
   545→          </marker>
   546→        </defs>
   547→
   548→        {/* pan surface (screen space, pattern follows pan/zoom) */}
   549→        <rect
   550→          className="rm-bg"
   551→          width="100%"
   552→          height="100%"
   553→          fill="url(#rm-dots)"
   554→          onPointerDown={onBgPointerDown}
   555→          onPointerMove={onBgPointerMove}
   556→          onPointerUp={endPan}
   557→          onPointerCancel={endPan}
   558→        />
   559→
   560→        <g transform={`translate(${f(viewport.x)} ${f(viewport.y)}) scale(${viewport.zoom.toFixed(3)})`}>
   561→          {/* edges first */}
   562→          {roadmap.edges.map((edge) => {
   563→            const a = nodeById.get(edge.from);
   564→            const b = nodeById.get(edge.to);
   565→            if (!a || !b) return null;
   566→            const pa = posOf(a);
   567→            const pb = posOf(b);
   568→            const x1 = pa.x + RM_NODE_W;
   569→            const y1 = pa.y + RM_NODE_H / 2;
   570→            const x2 = pb.x;
   571→            const y2 = pb.y + RM_NODE_H / 2;
   572→            const mx = (x1 + x2) / 2;
   573→            const d = `M ${f(x1)} ${f(y1)} C ${f(mx)} ${f(y1)}, ${f(mx)} ${f(y2)}, ${f(x2)} ${f(y2)}`;
   574→            const midX = (x1 + 6 * mx + x2) / 8;
   575→            const midY = (y1 + y2) / 2;
   576→            const selected = selection?.kind === 'edge' && selection.id === edge.id;
   577→            return (
   578→              <g key={edge.id}>
   579→                <path
   580→                  d={d}
   581→                  className={`rm-edge${selected ? ' selected' : ''}`}
   582→                  markerEnd={`url(#${selected ? 'rm-arrow-sel' : 'rm-arrow'})`}
   583→                />
   584→                {edge.label ? (
   585→                  <text x={f(midX)} y={f(midY - 6)} className="rm-edge-label">
   586→                    {edge.label}
   587→                  </text>
   588→                ) : null}
   589→                {/* wide invisible hit target for easy clicking */}
   590→                <path
   591→                  d={d}
   592→                  className="rm-edge-hit"
   593→                  onPointerDown={(e) => e.stopPropagation()}
   594→                  onClick={(e) => {
   595→                    e.stopPropagation();
   596→                    onSelect({ kind: 'edge', id: edge.id });
   597→                  }}
   598→                />
   599→              </g>
   600→            );
   601→          })}
   602→
   603→          {/* nodes on top */}
   604→          {renderNodes.map((node) => {
   605→            const p = posOf(node);
   606→            const live = liveById.get(node.id);
   607→            const st = live?.status ?? 'todo';
   608→            const selected = selection?.kind === 'node' && selection.id === node.id;
   609→            const isSource = linkSource === node.id;
   610→            const cls = [
   611→              'rm-node',
   612→              selected ? 'selected' : '',
   613→              isSource ? 'linking-source' : '',
   614→              drag?.id === node.id ? 'dragging' : '',
   615→              live?.missing ? 'deleted-ref' : '',
   616→            ]
   617→              .filter(Boolean)
   618→              .join(' ');
   619→            const Icon = RM_KIND_ICON[node.kind];
   620→            const isProgressKind = node.kind === 'goal' || node.kind === 'task' || node.kind === 'file';
   621→            return (
   622→              <foreignObject key={node.id} x={f(p.x)} y={f(p.y)} width={RM_NODE_W} height={RM_NODE_H}>
   623→                <div
   624→                  className={cls}
   625→                  role="button"
   626→                  tabIndex={0}
   627→                  aria-label={`${node.kind} node: ${live?.title ?? node.label}`}
   628→                  onPointerDown={(e) => onNodePointerDown(e, node)}
   629→                  onPointerMove={onNodePointerMove}
   630→                  onPointerUp={(e) => onNodePointerUp(e, node)}
   631→                  onPointerCancel={onNodePointerCancel}
   632→                  onKeyDown={(e) => onNodeKeyDown(e, node)}
   633→                >
   634→                  <span className={`rm-node-stripe st-${st}`} />
   635→                  <div className="rm-node-head">
   636→                    <span className={`rm-node-ico st-${st}`}>
   637→                      <Icon />
   638→                    </span>
   639→                    <span className="rm-node-title">
   640→                      {live?.title || 'Untitled'}
   641→                      {live?.missing && <em className="rm-deleted-tag">deleted</em>}
   642→                    </span>
   643→                  </div>
   644→
   645→                  {isProgressKind && live && (
   646→                    <div className="rm-node-pbar">
   647→                      <span
   648→                        style={{
   649→                          width: `${clamp(live.pct, 0, 100)}%`,
   650→                          background: live.status === 'blocked' ? 'var(--coral)' : undefined,
   651→                        }}
   652→                      />
   653→                    </div>
   654→                  )}
   655→
   656→                  <div className="rm-node-meta">
   657→                    {(node.kind === 'goal' || node.kind === 'task') && live && (
   658→                      <>
   659→                        <span className={`rm-mini-chip tone-${st}`}>{STATUS_LABEL[st]}</span>
   660→                        <span className="mono">{live.pct}%</span>
   661→                        {live.deadline && (
   662→                          <span className={`mono rm-dl-${live.deadlineTone ?? 'ok'}`}>{live.deadline.slice(5)}</span>
   663→                        )}
   664→                      </>
   665→                    )}
   666→                    {node.kind === 'file' && live && (
   667→                      <>
   668→                        <span className="mono">{live.pct}%</span>
   669→                        <span>
   670→                          {live.pagesRead}/{live.pagesTotal}p
   671→                        </span>
   672→                        {live.deadline && (
   673→                          <span className={`mono rm-dl-${live.deadlineTone ?? 'ok'}`}>{live.deadline.slice(5)}</span>
   674→                        )}
   675→                      </>
   676→                    )}
   677→                    {node.kind === 'milestone' && (
   678→                      <>
   679→                        <span className={`mono ${live?.deadlineTone === 'over' ? 'rm-dl-over' : ''}`}>
   680→                          {node.milestoneDate ? node.milestoneDate.slice(5) : 'no date'}
   681→                        </span>
   682→                        <button
   683→                          type="button"
   684→                          className={`rm-ms-toggle${node.done ? ' on' : ''}`}
   685→                          aria-pressed={node.done}
   686→                          aria-label={node.done ? 'Mark milestone as not done' : 'Mark milestone as done'}
   687→                          title={node.done ? 'Mark as not done' : 'Mark as done'}
   688→                          onPointerDown={(e) => e.stopPropagation()}
   689→                          onClick={(e) => {
   690→                            e.stopPropagation();
   691→                            onToggleMilestone(node);
   692→                          }}
   693→                        />
   694→                      </>
   695→                    )}
   696→                    {node.kind === 'note' && <span className="rm-note-txt">{node.note || '—'}</span>}
   697→                  </div>
   698→                </div>
   699→              </foreignObject>
   700→            );
   701→          })}
   702→        </g>
   703→      </svg>
   704→
   705→      {roadmap.nodes.length === 0 && (
   706→        <div className="rm-canvas-hint">
   707→          <div>
   708→            <div className="rm-hint-title">This canvas is empty</div>
   709→            Use “Add node” to drop goals, tasks, files, milestones or notes —<br />
   710→            then switch on “Link” and connect them to shape the path.
   711→          </div>
   712→        </div>
   713→      )}
   714→
   715→      <div className="rm-canvas-badge mono">
   716→        {roadmap.nodes.length} nodes · {roadmap.edges.length} links · drag to pan · scroll to zoom
   717→      </div>
   718→    </div>
   719→  );
   720→});
   721→
   722→export default RoadmapCanvas;
