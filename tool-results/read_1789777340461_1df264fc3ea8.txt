  1006→function UpNextCard({
  1007→  date,
  1008→  timeboxed,
  1009→  fileBlocks,
  1010→  fileById,
  1011→}: {
  1012→  date: string;
  1013→  timeboxed: Item[];
  1014→  fileBlocks: FilePlanBlock[];
  1015→  fileById: Map<string, ResourceFile>;
  1016→}) {
  1017→  const { openItemDetail } = useViewActions();
  1018→  const isToday = date === todayStr();
  1019→  const now = useNow(isToday);
  1020→
  1021→  const blocks = useMemo<UnBlock[]>(() => {
  1022→    const list: UnBlock[] = [
  1023→      ...timeboxed
  1024→        .filter((it) => it.status !== 'done')
  1025→        .map((it) => {
  1026→          const s = plannedStartToMin(it.plannedStart ?? '');
  1027→          return {
  1028→            kind: 'item' as const,
  1029→            id: it.id,
  1030→            fileId: null,
  1031→            title: it.title,
  1032→            start: s,
  1033→            end: s + timeboxDurMin(it),
  1034→          };
  1035→        }),
  1036→      ...fileBlocks
  1037→        .filter((p) => !p.done)
  1038→        .map((p) => ({
  1039→          kind: 'file' as const,
  1040→          id: p.id,
  1041→          fileId: p.fileId,
  1042→          title: fileById.get(p.fileId)?.title || 'Reading block',
  1043→          start: p.startMin,
  1044→          end: p.startMin + p.minutes,
  1045→        })),
  1046→    ];
  1047→    list.sort((a, b) => a.start - b.start);
  1048→    return list;
  1049→  }, [timeboxed, fileBlocks, fileById]);
  1050→
  1051→  const nowMin = isToday ? now.getHours() * 60 + now.getMinutes() : -1;
  1052→  const running = isToday ? blocks.find((b) => b.start <= nowMin && nowMin < b.end) : undefined;
  1053→  const next = running ? undefined : blocks.find((b) => b.start >= nowMin);
  1054→  const active = running ?? next;
  1055→
  1056→  const open = (b: UnBlock) => {
  1057→    if (b.kind === 'item') openItemDetail(b.id);
  1058→    else if (b.fileId) window.dispatchEvent(new CustomEvent('wb-open-reader', { detail: b.fileId }));
  1059→  };

  1061→  let body: React.ReactNode;
  1062→  if (active) {
  1063→    const meta = running
  1064→      ? `${minToHHMM(running.start)}–${minToHHMM(running.end)} · ${fmtDuration(Math.max(0, running.end - nowMin))} left`
  1065→      : isToday
  1066→        ? `at ${minToHHMM(next!.start)} · in ${fmtDuration(Math.max(0, next!.start - nowMin))}`
  1067→        : `at ${minToHHMM(next!.start)}`;
  1068→    body = (
  1069→      <button
  1070→        type="button"
  1071→        className="plan-un-row"
  1072→        onClick={() => open(active)}
  1073→        title={running ? 'Open current block' : 'Open next block'}
  1074→      >
  1075→        {running && <span className="plan-un-dot" aria-hidden />}
  1076→        <span className="plan-un-main">
  1077→          <span className="plan-un-kicker">{running ? 'Now' : isToday ? 'Next' : 'First block'}</span>
  1078→          <span className="plan-un-title">{active.title}</span>
  1079→          <span className="plan-un-meta mono">{meta}</span>
  1080→        </span>
  1081→        <ChevronRight size={14} className="plan-un-go" aria-hidden />
  1082→      </button>
  1083→    );
  1084→  } else {
  1085→    body = (
  1086→      <div className="plan-un-empty">
  1087→        {isToday ? 'Nothing left today — nice.' : 'Nothing planned for this day.'}
  1088→      </div>
  1089→    );
  1090→  }
  1091→
  1092→  return (
  1093→    <div className="card plan-card">
  1094→      <div className="plan-card-head">
  1095→        <span className="plan-card-title">Up next</span>
  1096→      </div>
  1097→      {body}
  1098→    </div>
  1099→  );
