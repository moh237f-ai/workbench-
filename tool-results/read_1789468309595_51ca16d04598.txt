     1→/* ==================== Library: Reader Overlay (pdf.js canvas) ====================
     2→   Full-screen reading surface for a stored PDF.
     3→   - Fit-width / zoom rendering with serialized page paints (no render races)
     4→   - Per-page reading-second tracking (flushed to the store every 10s / page change / close)
     5→   - Focus-read integration (pomodoro started from the reader)
     6→   - Immersive mode with auto-hiding chrome */
     7→
     8→'use client';
     9→
    10→import React, { useCallback, useEffect, useMemo, useRef, useState } from 'react';
    11→import { createPortal } from 'react-dom';
    12→import type { PDFDocumentProxy } from 'pdfjs-dist';
    13→import {
    14→  BookCheck, BookPlus, ChevronLeft, ChevronRight, Clock, Eye, FileText, FileWarning,
    15→  MoveHorizontal, PanelTopClose, PanelTopOpen, Pause, Play, SkipForward, Square, SquareCheck,
    16→  StickyNote, Target, Timer, X, ZoomIn, ZoomOut,
    17→} from 'lucide-react';
    18→import {
    19→  PAGE_READ_THRESHOLD_SEC, useWB, wbFileDeadlineStatus, wbFileProgress, wbFileReadPages,
    20→  wbItemsForFile,
    21→} from '@/lib/workbench/store';
    22→import { daysUntil, fmtClock, getFocusTiming, isoDate } from '@/lib/workbench/helpers';
    23→import { loadFileDoc, renderPageToCanvas } from '@/lib/workbench/pdf';
    24→import type { Item } from '@/lib/workbench/types';
    25→import { Btn, IconBtn, toast, useNow } from '../ui/primitives';
    26→import './reader.css';
    27→
    28→const ZOOM_MIN = 0.5;
    29→const ZOOM_MAX = 3;
    30→const ZOOM_STEP = 1.1;
    31→const FLUSH_EVERY_MS = 10_000;
    32→const CHROME_IDLE_MS = 2_600;
    33→const MIN_SESSION_MS = 60_000;
    34→
    35→type LoadState = 'loading' | 'ready' | 'error';
    36→
    37→const ITEM_ICON: Record<Item['type'], React.ReactNode> = {
    38→  goal: <Target size={12} />,
    39→  task: <SquareCheck size={12} />,
    40→  note: <StickyNote size={12} />,
    41→};
    42→
    43→function clampPage(p: number, total: number): number {
    44→  const max = Math.max(1, total);
    45→  const n = Math.round(p);
    46→  return Math.max(1, Math.min(max, Number.isFinite(n) ? n : 1));
    47→}
    48→
    49→function fmtRem(secs: number): string {
    50→  const s = Math.max(0, Math.floor(secs));
    51→  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`;
    52→}
    53→
    54→export function ReaderOverlay({ fileId, onClose }: { fileId: string; onClose: () => void }) {
    55→  /* ---------- store ---------- */
    56→  const file = useWB((s) => s.files.find((f) => f.id === fileId));
    57→  const items = useWB((s) => s.items);
    58→  const activeTimer = useWB((s) => s.activeTimer);
    59→  const setFilePosition = useWB((s) => s.setFilePosition);
    60→  const addFilePageSeconds = useWB((s) => s.addFilePageSeconds);
    61→  const markFilePage = useWB((s) => s.markFilePage);
    62→  const logReadingSession = useWB((s) => s.logReadingSession);
    63→  const startFileFocus = useWB((s) => s.startFileFocus);
    64→  const pausePomodoro = useWB((s) => s.pausePomodoro);
    65→  const resumePomodoro = useWB((s) => s.resumePomodoro);
    66→  const advancePomodoro = useWB((s) => s.advancePomodoro);
    67→  const stopTimer = useWB((s) => s.stopTimer);
    68→
    69→  const now = useNow();
    70→
    71→  /* ---------- view state ---------- */
    72→  const [doc, setDoc] = useState<PDFDocumentProxy | null>(null);
    73→  const [loadState, setLoadState] = useState<LoadState>('loading');
    74→  const [loadError, setLoadError] = useState('');
    75→  const [page, setPage] = useState(() => clampPage(file?.currentPage ?? 1, file?.pageCount ?? 1));
    76→  const [pageInput, setPageInput] = useState(() => String(clampPage(file?.currentPage ?? 1, file?.pageCount ?? 1)));
    77→  const [fit, setFit] = useState(true);
    78→  const [scale, setScale] = useState(1);
    79→  const [baseW, setBaseW] = useState(0);
    80→  const [stageW, setStageW] = useState(0);
    81→  const [rendering, setRendering] = useState(false);
    82→  const [hasPaint, setHasPaint] = useState(false);
    83→  const [immersive, setImmersive] = useState(false);
    84→  const [chromeOn, setChromeOn] = useState(true);
    85→
    86→  const numPages = doc?.numPages ?? file?.pageCount ?? 1;
    87→  const effScale = fit && baseW > 0 && stageW > 0 ? Math.max(0.05, stageW / baseW) : scale;
    88→
    89→  /* ---------- refs (session tracking + render serialization) ---------- */
    90→  const stageRef = useRef<HTMLDivElement | null>(null);
    91→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
    92→  const pageInputRef = useRef<HTMLInputElement | null>(null);
    93→  const sessionStartRef = useRef(Date.now());
    94→  const startPageRef = useRef(page);
    95→  const pageRef = useRef(page);
    96→  const pageSecsRef = useRef(0); // visible seconds on the current page (display)
    97→  const pendingRef = useRef<Map<number, number>>(new Map()); // page -> unflushed seconds
    98→  const sessionTotalsRef = useRef<Map<number, number>>(new Map()); // page -> session seconds
    99→  const visitedRef = useRef<Set<number>>(new Set());
   100→  const newlyReadRef = useRef<Set<number>>(new Set());
   101→  const renderSeqRef = useRef(0);
   102→  const renderChainRef = useRef<Promise<void>>(Promise.resolve());
   103→  const idleTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   104→
   105→  /* immutable snapshot of pageSeconds/readPages at mount — threshold crossing math */
   106→  const snapshotRef = useRef<{ secs: Record<string, number>; read: Set<string> } | null>(null);
   107→  if (file && snapshotRef.current === null) {
   108→    snapshotRef.current = { secs: { ...file.pageSeconds }, read: new Set(file.readPages) };
   109→  }
   110→
   111→  /* ---------- derived ---------- */
   112→  const elapsedSec = Math.max(0, Math.floor((now.getTime() - sessionStartRef.current) / 1000));
   113→  const timeOnPage = pageSecsRef.current;
   114→  const fileProgress = file ? wbFileProgress(file) : 0;
   115→  const readCount = file ? wbFileReadPages(file) : 0;
   116→  const pageIsRead = !!file && file.readPages.includes(String(page));
   117→  const linkedItems = useMemo(() => wbItemsForFile(items, fileId), [items, fileId]);
   118→
   119→  const focusHere = !!activeTimer && activeTimer.fileId === fileId;
   120→  const focusTiming = focusHere && activeTimer ? getFocusTiming(activeTimer, now.getTime()) : null;
   121→  const ftPaused = focusTiming?.paused ?? false;
   122→  const ftRemaining = focusTiming?.remaining ?? 0;
   123→  const ftPhase = focusTiming?.phaseLabel ?? '';
   124→  const phaseTone = activeTimer?.pomodoro?.phase === 'break' ? 'break' : 'work';
   125→
   126→  const deadlineChip = useMemo(() => {
   127→    if (!file?.deadline) return null;
   128→    const status = wbFileDeadlineStatus(file);
   129→    if (!status) return null;
   130→    if (fileProgress >= 100) return { text: 'Complete', tone: 'ok' as const };
   131→    const d = daysUntil(file.deadline);
   132→    if (d < 0) return { text: `${Math.abs(d)}d overdue`, tone: 'over' as const };
   133→    if (d === 0) return { text: 'Due today', tone: 'soon' as const };
   134→    return { text: `${d}d left`, tone: status === 'soon' ? ('soon' as const) : ('ok' as const) };
   135→  }, [file, fileProgress]);
   136→
   137→  /* ---------- reading-time tracking ---------- */
   138→
   139→  const flushPending = useCallback(() => {
   140→    const pending = pendingRef.current;
   141→    if (pending.size === 0) return;
   142→    const snap = snapshotRef.current;
   143→    for (const [pg, secs] of pending) {
   144→      pending.delete(pg);
   145→      if (secs <= 0) continue;
   146→      addFilePageSeconds(fileId, pg, secs);
   147→      const sess = (sessionTotalsRef.current.get(pg) || 0) + secs;
   148→      sessionTotalsRef.current.set(pg, sess);
   149→      if (snap) {
   150→        const before = snap.secs[String(pg)] || 0;
   151→        if (!snap.read.has(String(pg)) && before < PAGE_READ_THRESHOLD_SEC && before + sess >= PAGE_READ_THRESHOLD_SEC) {
   152→          newlyReadRef.current.add(pg);
   153→        }
   154→      }
   155→    }
   156→  }, [fileId, addFilePageSeconds]);
   157→
   158→  /* 1-second tick — only while the tab is visible */
   159→  useEffect(() => {
   160→    const id = setInterval(() => {
   161→      if (document.visibilityState !== 'visible') return;
   162→      const p = pageRef.current;
   163→      pendingRef.current.set(p, (pendingRef.current.get(p) || 0) + 1);
   164→      pageSecsRef.current += 1;
   165→    }, 1000);
   166→    return () => clearInterval(id);
   167→  }, []);
   168→
   169→  /* mark the entry page as visited */
   170→  useEffect(() => {
   171→    visitedRef.current.add(pageRef.current);
   172→  }, []);
   173→
   174→  /* periodic flush */
   175→  useEffect(() => {
   176→    const id = setInterval(() => flushPending(), FLUSH_EVERY_MS);
   177→    return () => clearInterval(id);
   178→  }, [flushPending]);
   179→
   180→  /* final flush + session log on unmount */
   181→  useEffect(() => {
   182→    const sessionStart = sessionStartRef.current;
   183→    const startPage = startPageRef.current;
   184→    return () => {
   185→      flushPending();
   186→      const elapsed = Date.now() - sessionStart;
   187→      if (elapsed < MIN_SESSION_MS) return;
   188→      logReadingSession({
   189→        fileId,
   190→        date: isoDate(new Date(sessionStart)),
   191→        startedAt: new Date(sessionStart).toISOString(),
   192→        endedAt: new Date().toISOString(),
   193→        minutes: Math.max(1, Math.round(elapsed / 60000)),
   194→        pagesVisited: visitedRef.current.size,
   195→        pagesRead: newlyReadRef.current.size,
   196→        startPage,
   197→        endPage: pageRef.current,
   198→      });
   199→    };
   200→  }, [fileId, flushPending, logReadingSession]);
   201→
   202→  /* lock background scroll while the overlay is up */
   203→  useEffect(() => {
   204→    const prev = document.body.style.overflow;
   205→    document.body.style.overflow = 'hidden';
   206→    return () => {
   207→      document.body.style.overflow = prev;
   208→    };
   209→  }, []);
   210→
   211→  /* ---------- document loading ---------- */
   212→
   213→  useEffect(() => {
   214→    let alive = true;
   215→    loadFileDoc(fileId)
   216→      .then((d) => {
   217→        if (!alive) return;
   218→        setDoc(d);
   219→        setLoadState('ready');
   220→      })
   221→      .catch((err: unknown) => {
   222→        if (!alive) return;
   223→        setLoadState('error');
   224→        setLoadError(err instanceof Error ? err.message : 'Could not open this PDF.');
   225→      });
   226→    return () => {
   227→      alive = false;
   228→    };
   229→  }, [fileId]);
   230→
   231→  /* base page width (scale 1) for fit-width math */
   232→  useEffect(() => {
   233→    if (!doc) return;
   234→    let alive = true;
   235→    doc
   236→      .getPage(clampPage(page, doc.numPages))
   237→      .then((pg) => {
   238→        if (alive) setBaseW(pg.getViewport({ scale: 1 }).width);
   239→      })
   240→      .catch(() => {});
   241→    return () => {
   242→      alive = false;
   243→    };
   244→  }, [doc, page]);
   245→
   246→  /* clamp page if the doc turns out shorter than stored metadata */
   247→  useEffect(() => {
   248→    if (doc && page > doc.numPages) {
   249→      const target = clampPage(page, doc.numPages);
   250→      setPage(target);
   251→      setPageInput(String(target));
   252→      pageRef.current = target;
   253→      pageSecsRef.current = 0;
   254→    }
   255→  }, [doc, page]);
   256→
   257→  /* available stage width (debounced, exact against computed padding) */
   258→  useEffect(() => {
   259→    const stage = stageRef.current;
   260→    if (!stage) return;
   261→    let timer: ReturnType<typeof setTimeout> | null = null;
   262→    const measure = () => {
   263→      const cs = window.getComputedStyle(stage);
   264→      const pad = parseFloat(cs.paddingLeft || '0') + parseFloat(cs.paddingRight || '0');
   265→      setStageW(Math.max(0, stage.clientWidth - pad - 12));
   266→    };
   267→    const ro = new ResizeObserver(() => {
   268→      if (timer) clearTimeout(timer);
   269→      timer = setTimeout(measure, 140);
   270→    });
   271→    ro.observe(stage);
   272→    measure();
   273→    return () => {
   274→      ro.disconnect();
   275→      if (timer) clearTimeout(timer);
   276→    };
   277→  }, []);
   278→
   279→  /* serialized canvas painting — latest request wins, no concurrent renders */
   280→  useEffect(() => {
   281→    if (!doc || loadState !== 'ready' || effScale <= 0) return;
   282→    const seq = ++renderSeqRef.current;
   283→    const prev = renderChainRef.current;
   284→    const run = (async () => {
   285→      await prev;
   286→      if (seq !== renderSeqRef.current) return;
   287→      const canvas = canvasRef.current;
   288→      if (!canvas) return;
   289→      setRendering(true);
   290→      try {
   291→        await renderPageToCanvas(doc, page, canvas, { scale: effScale });
   292→        if (seq === renderSeqRef.current) {
   293→          setRendering(false);
   294→          setHasPaint(true);
   295→        }
   296→      } catch {
   297→        if (seq === renderSeqRef.current) setRendering(false);
   298→      }
   299→    })();
   300→    renderChainRef.current = run.catch(() => {});
   301→  }, [doc, page, effScale, loadState]);
   302→
   303→  /* ---------- navigation ---------- */
   304→
   305→  const goToPage = useCallback(
   306→    (next: number) => {
   307→      const total = doc?.numPages ?? file?.pageCount ?? 1;
   308→      const target = clampPage(next, total);
   309→      if (target !== page) {
   310→        flushPending();
   311→        visitedRef.current.add(target);
   312→        pageRef.current = target;
   313→        pageSecsRef.current = 0;
   314→        setPage(target);
   315→        setFilePosition(fileId, target);
   316→        stageRef.current?.scrollTo({ top: 0 });
   317→      }
   318→      setPageInput(String(target));
   319→    },
   320→    [doc, file, fileId, page, flushPending, setFilePosition]
   321→  );
   322→
   323→  const applyPageInput = useCallback(() => {
   324→    const n = parseInt(pageInput, 10);
   325→    if (Number.isFinite(n)) goToPage(n);
   326→    else setPageInput(String(page));
   327→  }, [pageInput, goToPage, page]);
   328→
   329→  const closeReader = useCallback(() => {
   330→    onClose();
   331→  }, [onClose]);
   332→
   333→  /* ---------- keyboard ---------- */
   334→
   335→  useEffect(() => {
   336→    const onKey = (e: KeyboardEvent) => {
   337→      const t = e.target as HTMLElement | null;
   338→      const inField =
   339→        !!t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.tagName === 'SELECT' || t.isContentEditable);
   340→      if (e.key === 'Escape') {
   341→        if (inField) {
   342→          pageInputRef.current?.blur();
   343→          setPageInput(String(page));
   344→          return;
   345→        }
   346→        if (document.querySelector('.modal-overlay')) return; // a modal above owns Esc
   347→        e.preventDefault();
   348→        if (immersive) {
   349→          setImmersive(false);
   350→          setChromeOn(true);
   351→        } else {
   352→          closeReader();
   353→        }
   354→        return;
   355→      }
   356→      if (inField || document.querySelector('.modal-overlay')) return;
   357→      if (e.key === 'ArrowLeft' || e.key === 'PageUp') {
   358→        e.preventDefault();
   359→        goToPage(page - 1);
   360→      } else if (e.key === 'ArrowRight' || e.key === 'PageDown') {
   361→        e.preventDefault();
   362→        goToPage(page + 1);
   363→      } else if (e.key === 'Home') {
   364→        e.preventDefault();
   365→        goToPage(1);
   366→      } else if (e.key === 'End') {
   367→        e.preventDefault();
   368→        goToPage(numPages);
   369→      }
   370→    };
   371→    window.addEventListener('keydown', onKey);
   372→    return () => window.removeEventListener('keydown', onKey);
   373→  }, [page, numPages, goToPage, immersive, closeReader]);
   374→
   375→  /* ---------- immersive chrome ---------- */
   376→
   377→  const bumpChrome = useCallback(() => {
   378→    setChromeOn(true);
   379→    if (idleTimerRef.current) clearTimeout(idleTimerRef.current);
   380→    idleTimerRef.current = setTimeout(() => setChromeOn(false), CHROME_IDLE_MS);
   381→  }, []);
   382→
   383→  useEffect(() => {
   384→    if (!immersive) return;
   385→    bumpChrome();
   386→    const onMove = () => bumpChrome();
   387→    window.addEventListener('mousemove', onMove);
   388→    return () => {
   389→      window.removeEventListener('mousemove', onMove);
   390→      if (idleTimerRef.current) {
   391→        clearTimeout(idleTimerRef.current);
   392→        idleTimerRef.current = null;
   393→      }
   394→    };
   395→  }, [immersive, bumpChrome]);
   396→
   397→  const toggleImmersive = useCallback(() => {
   398→    const next = !immersive;
   399→    setImmersive(next);
   400→    if (!next) {
   401→      if (idleTimerRef.current) {
   402→        clearTimeout(idleTimerRef.current);
   403→        idleTimerRef.current = null;
   404→      }
   405→      setChromeOn(true);
   406→    }
   407→  }, [immersive]);
   408→
   409→  /* tap/click on the page reveals chrome while immersive (touch support) */
   410→  const onStageClick = () => {
   411→    if (immersive && !chromeOn) bumpChrome();
   412→  };
   413→
   414→  /* ---------- zoom ---------- */
   415→
   416→  const zoomBy = useCallback(
   417→    (mult: number) => {
   418→      setScale(Math.min(ZOOM_MAX, Math.max(ZOOM_MIN, effScale * mult)));
   419→      setFit(false);
   420→    },
   421→    [effScale]
   422→  );
   423→
   424→  /* ---------- focus read ---------- */
   425→
   426→  const startFocusRead = () => {
   427→    if (startFileFocus(fileId)) toast('Focus read started', 'Pomodoro running — page time still counts.');
   428→    else toast('A timer is already running', 'End it before starting a focus read.');
   429→  };
   430→
   431→  const endFocusRead = () => {
   432→    stopTimer();
   433→    toast('Session logged');
   434→  };
   435→
   436→  /* auto-advance the pomodoro while focus-reading here */
   437→  useEffect(() => {
   438→    if (!focusHere || ftPaused || ftRemaining > 0) return;
   439→    advancePomodoro();
   440→    toast(
   441→      ftPhase === 'Focus' ? 'Break time! ☕' : 'Back to work! 🧠',
   442→      ftPhase === 'Focus' ? 'Focus phase logged.' : 'Break over — next cycle starts now.'
   443→    );
   444→  }, [focusHere, ftPaused, ftRemaining, ftPhase, advancePomodoro]);
   445→
   446→  /* ---------- render ---------- */
   447→
   448→  if (typeof document === 'undefined') return null;
   449→
   450→  const openLinked = (itemId: string) => {
   451→    window.dispatchEvent(new CustomEvent('wb-open-item', { detail: itemId }));
   452→  };
   453→
   454→  const showSpinner = loadState === 'loading' || (!!file && loadState === 'ready' && !hasPaint);
   455→
   456→  return createPortal(
   457→    <div
   458→      className={`reader-overlay${immersive ? ' reader-immersive' : ''}${immersive && chromeOn ? ' rd-chrome-on' : ''}`}
   459→      role="dialog"
   460→      aria-modal="true"
   461→      aria-label={`Reading ${file?.title ?? 'PDF'}`}
   462→    >
   463→      {/* ------- header ------- */}
   464→      <header className="rd-header">
   465→        <div className="rd-title-wrap">
   466→          <div className="rd-title-row">
   467→            <span className="rd-file-badge">
   468→              <FileText size={15} />
   469→            </span>
   470→            <span className="rd-title" title={file?.title ?? 'PDF'}>
   471→              {file?.title ?? 'PDF'}
   472→            </span>
   473→            <span className="rd-pages">
   474→              p.{page} of {numPages}
   475→            </span>
   476→          </div>
   477→          <div
   478→            className="rd-progress"
   479→            role="progressbar"
   480→            aria-valuenow={fileProgress}
   481→            aria-valuemin={0}
   482→            aria-valuemax={100}
   483→            aria-label="Read pages"
   484→          >
   485→            <div className="rd-progress-fill" style={{ width: `${fileProgress}%` }} />
   486→          </div>
   487→          <div className="rd-progress-meta">
   488→            {readCount}/{numPages} pages read · {fileProgress}%
   489→          </div>
   490→        </div>
   491→
   492→        <div className="rd-header-side">
   493→          <span className="rd-clock-chip" title="Reading session time">
   494→            <Clock size={13} />
   495→            <span className="rd-mono">{fmtClock(elapsedSec)}</span>
   496→          </span>
   497→          <span className="rd-clock-chip rd-sub" title="Time on this page">
   498→            <Eye size={13} />
   499→            <span>
   500→              {timeOnPage}s on page
   501→            </span>
   502→          </span>
   503→
   504→          {deadlineChip && <span className={`rd-chip rd-${deadlineChip.tone}`}>{deadlineChip.text}</span>}
   505→
   506→          {linkedItems.length > 0 && (
   507→            <div className="rd-items">
   508→              {linkedItems.slice(0, 3).map((it) => (
   509→                <button
   510→                  key={it.id}
   511→                  type="button"
   512→                  className="rd-item-chip"
   513→                  onClick={() => openLinked(it.id)}
   514→                  title={`Open ${it.title}`}
   515→                >
   516→                  {ITEM_ICON[it.type]}
   517→                  <span className="rd-item-title">{it.title}</span>
   518→                </button>
   519→              ))}
   520→              {linkedItems.length > 3 && <span className="rd-chip">+{linkedItems.length - 3}</span>}
   521→            </div>
   522→          )}
   523→
   524→          {!activeTimer ? (
   525→            <Btn variant="primary" size="sm" onClick={startFocusRead}>
   526→              <Play size={13} /> Focus read
   527→            </Btn>
   528→          ) : focusHere && focusTiming ? (
   529→            <div className="rd-focus-box">
   530→              <span className={`rd-phase-badge ${phaseTone}`}>{focusTiming.phaseLabel}</span>
   531→              <span className="rd-mono rd-focus-rem">{fmtRem(focusTiming.remaining)}</span>
   532→              {focusTiming.paused ? (
   533→                <IconBtn small title="Resume pomodoro" onClick={resumePomodoro}>
   534→                  <Play size={13} />
   535→                </IconBtn>
   536→              ) : (
   537→                <IconBtn small title="Pause pomodoro" onClick={pausePomodoro}>
   538→                  <Pause size={13} />
   539→                </IconBtn>
   540→              )}
   541→              <IconBtn small title="Skip to next phase" onClick={advancePomodoro}>
   542→                <SkipForward size={13} />
   543→              </IconBtn>
   544→              <IconBtn small danger title="End focus session" onClick={endFocusRead}>
   545→                <Square size={13} />
   546→              </IconBtn>
   547→            </div>
   548→          ) : (
   549→            <span className="rd-chip rd-elsewhere" title="A timer is running outside the reader">
   550→              <Timer size={12} /> Timer running elsewhere
   551→            </span>
   552→          )}
   553→
   554→          <IconBtn title="Close reader (Esc)" onClick={closeReader}>
   555→            <X size={16} />
   556→          </IconBtn>
   557→        </div>
   558→      </header>
   559→
   560→      {/* ------- page stage ------- */}
   561→      <div className="rd-stage-wrap" onClick={onStageClick}>
   562→        <div className="rd-stage" ref={stageRef} role="region" aria-label="PDF page view">
   563→          {file && loadState === 'ready' && (
   564→            <canvas
   565→              ref={canvasRef}
   566→              className="rd-canvas"
   567→              style={{ visibility: hasPaint ? 'visible' : 'hidden' }}
   568→              aria-label={`Page ${page} of ${numPages}`}
   569→            />
   570→          )}
   571→        </div>
   572→
   573→        {showSpinner && (
   574→          <div className="rd-centered">
   575→            <div className="rd-spinner" role="status" aria-label="Loading PDF" />
   576→            <div className="rd-load-label">Opening PDF…</div>
   577→          </div>
   578→        )}
   579→
   580→        {loadState === 'error' && (
   581→          <div className="rd-centered">
   582→            <div className="rd-error-card">
   583→              <FileWarning size={22} />
   584→              <div className="rd-error-title">Couldn't open this file</div>
   585→              <div className="rd-error-msg">{loadError}</div>
   586→              <Btn variant="primary" size="sm" onClick={closeReader}>
   587→                Close
   588→              </Btn>
   589→            </div>
   590→          </div>
   591→        )}
   592→
   593→        {!file && loadState === 'ready' && (
   594→          <div className="rd-centered">
   595→            <div className="rd-error-card">
   596→              <FileWarning size={22} />
   597→              <div className="rd-error-title">File not found</div>
   598→              <div className="rd-error-msg">This file may have been deleted from your library.</div>
   599→              <Btn variant="primary" size="sm" onClick={closeReader}>
   600→                Close
   601→              </Btn>
   602→            </div>
   603→          </div>
   604→        )}
   605→
   606→        {rendering && hasPaint && (
   607→          <div className="rd-rendering-chip">
   608→            <span className="rd-spinner rd-spinner-sm" /> Rendering…
   609→          </div>
   610→        )}
   611→
   612→        {immersive && !chromeOn && (
   613→          <div className="rd-immersive-hint">Immersive reading · move mouse or tap for controls · Esc exits</div>
   614→        )}
   615→      </div>
   616→
   617→      {/* ------- footer toolbar ------- */}
   618→      <footer className="rd-footer">
   619→        <div className="rd-toolbar">
   620→          <div className="rd-group" role="group" aria-label="Zoom">
   621→            <IconBtn title="Zoom out" onClick={() => zoomBy(1 / ZOOM_STEP)}>
   622→              <ZoomOut size={15} />
   623→            </IconBtn>
   624→            <span className="rd-zoom-label rd-mono">{Math.round(effScale * 100)}%</span>
   625→            <IconBtn title="Zoom in" onClick={() => zoomBy(ZOOM_STEP)}>
   626→              <ZoomIn size={15} />
   627→            </IconBtn>
   628→            <Btn size="sm" variant={fit ? 'primary' : 'default'} onClick={() => setFit(true)} title="Fit page width">
   629→              <MoveHorizontal size={13} /> Fit
   630→            </Btn>
   631→          </div>
   632→
   633→          <span className="rd-sep" aria-hidden="true" />
   634→
   635→          <div className="rd-group" role="group" aria-label="Page navigation">
   636→            <IconBtn title="Previous page (←)" onClick={() => goToPage(page - 1)} disabled={page <= 1}>
   637→              <ChevronLeft size={15} />
   638→            </IconBtn>
   639→            <span className="rd-page-box">
   640→              <input
   641→                ref={pageInputRef}
   642→                className="rd-page-input"
   643→                type="text"
   644→                inputMode="numeric"
   645→                value={pageInput}
   646→                aria-label="Page number — press Enter to go"
   647→                onChange={(e) => setPageInput(e.target.value)}
   648→                onKeyDown={(e) => {
   649→                  if (e.key === 'Enter') {
   650→                    e.preventDefault();
   651→                    applyPageInput();
   652→                    pageInputRef.current?.blur();
   653→                  }
   654→                }}
   655→              />
   656→              <span className="rd-page-total">/ {numPages}</span>
   657→            </span>
   658→            <IconBtn title="Next page (→)" onClick={() => goToPage(page + 1)} disabled={page >= numPages}>
   659→              <ChevronRight size={15} />
   660→            </IconBtn>
   661→          </div>
   662→
   663→          <span className="rd-sep" aria-hidden="true" />
   664→
   665→          <Btn
   666→            size="sm"
   667→            variant={pageIsRead ? 'primary' : 'default'}
   668→            onClick={() => markFilePage(fileId, page, !pageIsRead)}
   669→            title={pageIsRead ? 'Mark this page unread' : 'Mark this page read'}
   670→          >
   671→            {pageIsRead ? <BookCheck size={13} /> : <BookPlus size={13} />}
   672→            {pageIsRead ? 'Read' : 'Mark read'}
   673→          </Btn>
   674→
   675→          <Btn
   676→            size="sm"
   677→            variant={immersive ? 'primary' : 'default'}
   678→            onClick={toggleImmersive}
   679→            title="Immersive reading — hides the toolbar until you move the mouse"
   680→          >
   681→            {immersive ? <PanelTopOpen size={13} /> : <PanelTopClose size={13} />} Immersive
   682→          </Btn>
   683→
   684→          <span className="rd-sep" aria-hidden="true" />
   685→
   686→          <span className="rd-session-clock" title="Reading session time">
   687→            <Clock size={13} />
   688→            <span className="rd-mono">{fmtClock(elapsedSec)}</span>
   689→          </span>
   690→        </div>
   691→      </footer>
   692→    </div>,
   693→    document.body
   694→  );
   695→}
