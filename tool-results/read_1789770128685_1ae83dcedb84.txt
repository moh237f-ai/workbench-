   174→function Ring({ pct }: { pct: number }) {
   175→  const r = 15.5;
   176→  const c = 2 * Math.PI * r;
   177→  return (
   178→    <div className="rm-ring" aria-hidden="true">
   179→      <svg viewBox="0 0 38 38" width={38} height={38}>
   180→        <circle className="rm-ring-track" cx={19} cy={19} r={r} />
   181→        <circle
   182→          className="rm-ring-val"
   183→          cx={19} cy={19} r={r}
   184→          strokeDasharray={`${(c * pct) / 100} ${c}`}
   185→          transform="rotate(-90 19 19)"
   186→        />
   187→      </svg>
   188→      <span className="rm-ring-pct mono">{pct}%</span>

   359→function RoadmapWorkspace({ roadmap }: { roadmap: Roadmap }) {
   360→  const items = useWB((s) => s.items);
   361→  const files = useWB((s) => s.files);
   362→  const filePlans = useWB((s) => s.filePlans);
   363→  const updateRoadmap = useWB((s) => s.updateRoadmap);
   364→  const updateRoadmapNode = useWB((s) => s.updateRoadmapNode);
   365→  const removeRoadmapNode = useWB((s) => s.removeRoadmapNode);
   366→  const addRoadmapNode = useWB((s) => s.addRoadmapNode);
   367→  const addRoadmapEdge = useWB((s) => s.addRoadmapEdge);
   368→  const removeRoadmapEdge = useWB((s) => s.removeRoadmapEdge);
   369→  const updateRoadmapEdge = useWB((s) => s.updateRoadmapEdge);
   370→  const setActiveRoadmap = useWB((s) => s.setActiveRoadmap);
   371→  const updateItem = useWB((s) => s.updateItem);
   372→  const addFilePlan = useWB((s) => s.addFilePlan);
   373→
   374→  const canvasRef = useRef<RoadmapCanvasHandle>(null);
   375→  const timelineRef = useRef<RoadmapTimelineHandle>(null);
   376→  const [viewMode, setViewMode] = useState<ViewMode>('canvas');
   377→  const [selection, setSelection] = useState<RmSelection | null>(null);
   378→  const [linkMode, setLinkMode] = useState(false);
   379→  const [linkSource, setLinkSource] = useState<string | null>(null);
   380→  const [orderOpen, setOrderOpen] = useState(false);
   381→  const [pop, setPop] = useState<PopState>(null);
   382→  const [popSearch, setPopSearch] = useState('');
   383→  const [layoutPop, setLayoutPop] = useState(false);
   384→  const [zoom, setZoom] = useState(1);
   385→  const [fitKey, setFitKey] = useState(0);
   386→
   387→  const liveMap = useMemo(() => {
   388→    const m = new Map<string, RmNodeLive>();
   389→    roadmap.nodes.forEach((n) => m.set(n.id, nodeLive(n, items, files)));
   390→    return m;
   391→  }, [roadmap.nodes, items, files]);
   392→
   393→  const selectedNode =
   394→    selection?.kind === 'node' ? roadmap.nodes.find((n) => n.id === selection.id) || null : null;
   395→  const selectedEdge =
   396→    selection?.kind === 'edge' ? roadmap.edges.find((e) => e.id === selection.id) || null : null;
   397→
   398→  /* ---------- derived stats (shared by stats bar + order panel) ---------- */
   399→
   400→  const orderRows = useMemo(() => computeOrder(roadmap, liveMap), [roadmap, liveMap]);
   401→  const progress = useMemo(() => roadmapProgress(roadmap, liveMap), [roadmap, liveMap]);
   402→  const nextUp = useMemo(() => nextUpRow(orderRows.rows), [orderRows]);
   403→
   404→  /* keyboard: Esc closes pops / cancels link / clears selection; Delete removes */
   405→  useEffect(() => {
   406→    const onKey = (e: KeyboardEvent) => {
   407→      const t = e.target as HTMLElement;
   408→      if (['INPUT', 'TEXTAREA', 'SELECT'].includes(t?.tagName) || t?.isContentEditable) return;
   409→      if (e.key === 'Escape') {
   410→        if (pop) setPop(null);
   411→        else if (layoutPop) setLayoutPop(false);
   412→        else if (linkSource) setLinkSource(null);
   413→        else if (selection) setSelection(null);
   414→        return;
   415→      }
   416→      if ((e.key === 'Delete' || e.key === 'Backspace') && selection) {
   417→        e.preventDefault();
   418→        if (selection.kind === 'node') {
   419→          removeRoadmapNode(roadmap.id, selection.id);
   420→          toast('Node removed');
   421→        } else {
   422→          removeRoadmapEdge(roadmap.id, selection.id);
   423→          toast('Link removed');
   424→        }
   425→        setSelection(null);
   426→      }
   427→    };
   428→    window.addEventListener('keydown', onKey);
   429→    return () => window.removeEventListener('keydown', onKey);
   430→  }, [pop, layoutPop, linkSource, selection, roadmap.id, removeRoadmapNode, removeRoadmapEdge]);
   431→
   432→  /* ---------- focus a node in whichever view is active ---------- */
   433→
   434→  const focusNode = (id: string) => {
   435→    setSelection({ kind: 'node', id });
   436→    if (viewMode === 'canvas') canvasRef.current?.centerNode(id);
   437→    else timelineRef.current?.scrollToNode(id);
   438→  };
   439→
   440→  /* ---------- node creation ---------- */
   441→
   442→  const spawn = (partial: Omit<RoadmapNode, 'id'>) => {
   443→    const spot = canvasRef.current?.dropSpot() ?? { x: 80, y: 80 };
   444→    const node = addRoadmapNode(roadmap.id, { ...partial, x: spot.x, y: spot.y });
   445→    setPop(null);
   446→    setPopSearch('');
   447→    setSelection({ kind: 'node', id: node.id });
   448→    return node;
   449→  };
   450→
   451→  const pickList = useMemo(() => {
   452→    if (!pop || pop.mode !== 'pick') return [] as { id: string; title: string; meta: string[] }[];
   453→    const q = popSearch.trim().toLowerCase();
   454→    if (pop.kind === 'file') {
   455→      return files
   456→        .filter((f) => !f.archived)
   457→        .filter((f) => !q || f.title.toLowerCase().includes(q) || f.fileName.toLowerCase().includes(q))
   458→        .slice(0, 40)
   459→        .map((f) => ({
   460→          id: f.id,
   461→          title: f.title,
   462→          meta: [`${f.pageCount} pages`, f.deadline ? `due ${f.deadline.slice(5)}` : 'no deadline'],
   463→        }));
   464→    }
   465→    return items
   466→      .filter((i) => i.type === pop.kind)
   467→      .filter((i) => !['archived'].includes(i.status))
   468→      .filter((i) => !q || i.title.toLowerCase().includes(q))
   469→      .sort((a, b) => (a.status === 'done' ? 1 : 0) - (b.status === 'done' ? 1 : 0))
   470→      .slice(0, 40)
   471→      .map((i) => ({
   472→        id: i.id,
   473→        title: i.title,
   474→        meta: [
   475→          i.status.replace('_', ' '),
   476→          i.deadline ? `due ${i.deadline.slice(5)}` : 'no deadline',
   477→          i.estimateMin ? `${i.estimateMin}m` : '',
   478→        ].filter(Boolean),
   479→      }));
   480→  }, [pop, popSearch, items, files]);
   481→
   482→  /* ---------- auto layout (two strategies) ---------- */
   483→
   484→  const layoutByLayers = () => {
   485→    const order = topoOrder(roadmap.nodes, roadmap.edges);
   486→    const depth = new Map<string, number>();
   487→    roadmap.nodes.forEach((n) => depth.set(n.id, 0));
   488→    order.forEach((id) => {
   489→      roadmap.edges.forEach((e) => {
   490→        if (e.from !== id) return;
   491→        depth.set(e.to, Math.max(depth.get(e.to) ?? 0, (depth.get(id) ?? 0) + 1));
   492→      });
   493→    });
   494→    const layers = new Map<number, RoadmapNode[]>();
   495→    roadmap.nodes.forEach((n) => {
   496→      const d = depth.get(n.id) ?? 0;
   497→      const arr = layers.get(d) || [];
   498→      arr.push(n);
   499→      layers.set(d, arr);
   500→    });
   501→    layers.forEach((arr, d) => {
   502→      arr.sort((a, b) => a.y - b.y);
   503→      arr.forEach((n, i) => updateRoadmapNode(roadmap.id, n.id, { x: 60 + d * 290, y: 60 + i * 150 }));
   504→    });
   505→    setFitKey((k) => k + 1);
   506→    toast('Auto-layout applied', 'Layers follow your connections.');
   507→  };
   508→
   509→  const layoutByKind = () => {
   510→    KIND_ORDER.forEach((k, row) => {
   511→      const arr = roadmap.nodes.filter((n) => n.kind === k).sort((a, b) => a.y - b.y);
   512→      arr.forEach((n, i) => updateRoadmapNode(roadmap.id, n.id, { x: 60 + i * 290, y: 60 + row * 150 }));
   513→    });
   514→    setFitKey((k) => k + 1);
   515→    toast('Auto-layout applied', 'Swimlanes group nodes by kind.');
   516→  };
   517→
   518→  /* ---------- schedule a row onto tomorrow's planner ---------- */
   519→
   520→  const scheduleRow = (r: OrderRow) => {
   521→    const node = r.node;
   522→    if (node.kind === 'milestone' || node.kind === 'note' || !node.refId) {
   523→      toast("Milestones aren't schedulable — link a task first.");
   524→      return;
   525→    }
   526→    const tomorrow = addDaysStr(todayStr(), 1);
   527→    const title = r.live?.title || node.label || 'Untitled';
   528→    if (node.kind === 'file') {
   529→      const f = files.find((x) => x.id === node.refId);
   530→      if (!f) {

  1052→function EdgeLabelInput({ initial, onCommit }: { initial: string; onCommit: (v: string) => void }) {
  1053→  const [value, setValue] = useState(initial);

  1060→      onChange={(e) => setValue(e.target.value)}
  1061→      onBlur={() => onCommit(value.trim())}
  1062→      onKeyDown={(e) => {
  1063→        if (e.key === 'Enter') (e.target as HTMLInputElement).blur();
  1064→      }}

  1069→function MiniForm({
  1070→  fields,
  1071→  submitLabel,
  1072→  onSubmit,
  1073→  onCancel,
  1074→}: {
  1075→  fields: { key: string; placeholder: string; type?: string; autoFocus?: boolean }[];
  1076→  submitLabel: string;
  1077→  onSubmit: (values: Record<string, string>) => void;
  1078→  onCancel: () => void;
  1079→}) {
  1080→  const [values, setValues] = useState<Record<string, string>>({});
  1081→  return (
  1082→    <div className="rm-mini-form">
  1083→      {fields.map((f) => (
  1084→        <input
  1085→          key={f.key}
  1086→          className="rm-mini-input"
  1087→          type={f.type || 'text'}
  1088→          autoFocus={f.autoFocus}
  1089→          placeholder={f.placeholder}
  1090→          value={values[f.key] || ''}
  1091→          onChange={(e) => setValues((v) => ({ ...v, [f.key]: e.target.value }))}
  1092→          onKeyDown={(e) => {
  1093→            if (e.key === 'Enter') onSubmit(values);
  1094→            if (e.key === 'Escape') onCancel();
  1095→          }}
  1096→        />
  1097→      ))}
  1098→      <div className="rm-mini-actions">
  1099→        <Btn size="sm" onClick={onCancel}>Cancel</Btn>
  1100→        <Btn size="sm" variant="primary" onClick={() => onSubmit(values)}>{submitLabel}</Btn>
  1101→      </div>
  1102→    </div>
  1103→  );
  1104→}
