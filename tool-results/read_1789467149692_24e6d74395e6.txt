   253→function deleteCascade(
   254→  items: Item[],
   255→  labels: Label[],
   256→  expanded: Record<string, boolean>,
   257→  id: string
   258→): Item[] {
   259→  const desc = (iid: string): string[] => {
   260→    const out: string[] = [];
   261→    items.filter((i) => i.parentId === iid).forEach((c) => {
   262→      out.push(c.id);
   263→      out.push(...desc(c.id));
   264→    });
   265→    return out;
   266→  };
   267→  const ids = [id, ...desc(id)];
   268→  labels.forEach((l) => {
   269→    l.itemIds = l.itemIds.filter((x) => !ids.includes(x));
   270→  });
   271→  ids.forEach((i) => delete expanded[i]);
   272→  return items.filter((i) => !ids.includes(i.id));
   273→}

   553→          roadmaps: s.roadmaps.map((r) =>
   554→            r.id === id ? { ...r, ...patch, updatedAt: new Date().toISOString() } : r

   999→export function wbItem(items: Item[], id: string): Item | undefined {

  1002→export function wbRootItems(items: Item[]): Item[] {

  1005→export function wbDescendantIds(items: Item[], id: string): string[] {
  1006→  const out: string[] = [];
  1007→  items.filter((i) => i.parentId === id).forEach((c) => {
  1008→    out.push(c.id);
  1009→    out.push(...wbDescendantIds(items, c.id));
  1010→  });
  1011→  return out;
  1012→}
  1013→export function wbIsBlocked(items: Item[], item: Item | undefined): boolean {
  1014→  if (!item) return false;
  1015→  return (item.dependsOn || []).some((depId) => {
  1016→    const d = wbItem(items, depId);
  1017→    return !!d && d.status !== 'done' && d.status !== 'archived';
  1018→  });
  1019→}
  1020→export function wbBlockedCount(items: Item[], item: Item): number {
  1021→  return (item.dependsOn || []).filter((depId) => {
  1022→    const d = wbItem(items, depId);
  1023→    return !!d && d.status !== 'done' && d.status !== 'archived';
  1024→  }).length;
  1025→}
  1026→export function wbCanDependOn(items: Item[], itemId: string, candidateId: string): boolean {
  1027→  if (!candidateId || candidateId === itemId) return false;
  1028→  if (wbDescendantIds(items, itemId).includes(candidateId)) return false;
  1029→  if (!wbItem(items, candidateId)) return false;
  1030→  // no cycles: candidate must not (transitively) depend on itemId
  1031→  const reaches = (startId: string, targetId: string, seen = new Set<string>()): boolean => {
  1032→    if (startId === targetId) return true;
  1033→    if (seen.has(startId)) return false;
  1034→    seen.add(startId);
  1035→    const it = wbItem(items, startId);
  1036→    return !!it && (it.dependsOn || []).some((d) => reaches(d, targetId, seen));
  1037→  };
  1038→  return !reaches(candidateId, itemId);
  1039→}
  1040→export function wbReadyItems(items: Item[]): Item[] {
  1041→  return items.filter(
  1042→    (i) => !['done', 'archived', 'inbox'].includes(i.status) && !wbIsBlocked(items, i)
  1043→  );
  1044→}
  1045→export function wbPlannedItems(items: Item[], date: string): Item[] {
  1046→  return items.filter((i) => i.type === 'task' && i.startDate === date && i.status !== 'done' && i.status !== 'archived');
  1047→}
  1048→export function wbPlannedMinutes(items: Item[], date: string): number {
  1049→  return wbPlannedItems(items, date).reduce((s, i) => s + (Number(i.estimateMin) || 30), 0);

  1059→export function wbPickNextAction(items: Item[]): Item | null {
  1060→  const today = todayStr();
  1061→  const pool = wbPlannedItems(items, today).filter((i) => !wbIsBlocked(items, i));
  1062→  const source = pool.length ? pool : wbReadyItems(items).filter((i) => i.type === 'task');
  1063→  return (
  1064→    source
  1065→      .filter((i) => i.type !== 'goal' || wbChildren(items, i.id).length === 0)
  1066→      .sort((a, b) => {
  1067→        const p = wbPriorityWeight(a.priority) - wbPriorityWeight(b.priority);
  1068→        if (p) return p;
  1069→        const ad = a.deadline ? new Date(a.deadline).getTime() : Infinity;
  1070→        const bd = b.deadline ? new Date(b.deadline).getTime() : Infinity;
  1071→        if (ad !== bd) return ad - bd;
  1072→        return (b.updatedAt || '').localeCompare(a.updatedAt || '');
  1073→      })[0] || null
  1074→  );
  1075→}
  1076→export function wbInsights(items: Item[], entries: Entry[], capacity: number): string[] {
  1077→  const today = todayStr();
  1078→  const a: string[] = [];
  1079→  const overdue = items.filter(
  1080→    (i) => i.deadline && i.status !== 'done' && i.status !== 'archived' && i.deadline < today
  1081→  ).length;
  1082→  const blocked = items.filter((i) => wbIsBlocked(items, i)).length;
  1083→  const planned = wbPlannedMinutes(items, today);
  1084→  const logged = entriesInRange(entries, today, today).reduce((s, e) => s + e.durationMin, 0);
  1085→  if (overdue) a.push(`${overdue} item${overdue === 1 ? ' is' : 's are'} overdue — pull the most important one into your next focus block.`);
  1086→  if (blocked) a.push(`${blocked} item${blocked === 1 ? ' is' : 's are'} blocked by dependencies. Resolve them before adding more work.`);
  1087→  if (planned > capacity) a.push(`Today is over capacity by ${planned - capacity} minutes. Trim the plan before starting another block.`);
  1088→  if (logged >= 240) a.push(`${Math.round(logged / 60)}h logged today — protect your recovery time.`);
  1089→  if (!a.length) a.push('The system is stable. Protect your next action and keep capacity realistic.');
  1090→  return a;
  1091→}
