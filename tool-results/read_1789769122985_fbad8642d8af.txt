   288→function deleteCascade(
   289→  items: Item[],
   290→  labels: Label[],
   291→  expanded: Record<string, boolean>,
   292→  id: string
   293→): Item[] {
   294→  const desc = (iid: string): string[] => {
   295→    const out: string[] = [];
   296→    items.filter((i) => i.parentId === iid).forEach((c) => {
   297→      out.push(c.id);
   298→      out.push(...desc(c.id));
   299→    });
   300→    return out;
   301→  };
   302→  const ids = [id, ...desc(id)];
   303→  labels.forEach((l) => {
   304→    l.itemIds = l.itemIds.filter((x) => !ids.includes(x));
   305→  });
   306→  ids.forEach((i) => delete expanded[i]);
   307→  return items.filter((i) => !ids.includes(i.id));
   308→}

   539→            const readPages = read
   540→              ? Array.from(new Set([...f.readPages, key]))
   541→              : f.readPages.filter((p) => p !== key);
   542→            return { ...f, readPages };
   543→          }),
   544→        })),
   545→      logReadingSession: (session) => {

   550→            ...s.stats,
   551→            readingMinutes: s.stats.readingMinutes + session.minutes,

   553→          },
   554→        }));
   555→      },

  1142→export function wbDescendantIds(items: Item[], id: string): string[] {
  1143→  const out: string[] = [];
  1144→  items.filter((i) => i.parentId === id).forEach((c) => {
  1145→    out.push(c.id);
  1146→    out.push(...wbDescendantIds(items, c.id));
  1147→  });

  1150→export function wbIsBlocked(items: Item[], item: Item | undefined): boolean {
  1151→  if (!item) return false;
  1152→  return (item.dependsOn || []).some((depId) => {
  1153→    const d = wbItem(items, depId);
  1154→    return !!d && d.status !== 'done' && d.status !== 'archived';
  1155→  });
  1156→}
  1157→export function wbBlockedCount(items: Item[], item: Item): number {
  1158→  return (item.dependsOn || []).filter((depId) => {
  1159→    const d = wbItem(items, depId);
  1160→    return !!d && d.status !== 'done' && d.status !== 'archived';
  1161→  }).length;
  1162→}
  1163→export function wbCanDependOn(items: Item[], itemId: string, candidateId: string): boolean {
  1164→  if (!candidateId || candidateId === itemId) return false;
  1165→  if (wbDescendantIds(items, itemId).includes(candidateId)) return false;
  1166→  if (!wbItem(items, candidateId)) return false;
  1167→  // no cycles: candidate must not (transitively) depend on itemId
  1168→  const reaches = (startId: string, targetId: string, seen = new Set<string>()): boolean => {
  1169→    if (startId === targetId) return true;
  1170→    if (seen.has(startId)) return false;
  1171→    seen.add(startId);
  1172→    const it = wbItem(items, startId);
  1173→    return !!it && (it.dependsOn || []).some((d) => reaches(d, targetId, seen));
  1174→  };
  1175→  return !reaches(candidateId, itemId);
  1176→}

  1193→export function wbPriorityWeight(p: string): number {
  1194→  return ({ p0: 0, p1: 1, p2: 2, p3: 3 } as Record<string, number>)[p] ?? 2;
  1195→}
  1196→export function wbPickNextAction(items: Item[]): Item | null {
  1197→  const today = todayStr();
  1198→  const pool = wbPlannedItems(items, today).filter((i) => !wbIsBlocked(items, i));
  1199→  const source = pool.length ? pool : wbReadyItems(items).filter((i) => i.type === 'task');
  1200→  return (
  1201→    source
  1202→      .filter((i) => i.type !== 'goal' || wbChildren(items, i.id).length === 0)
  1203→      .sort((a, b) => {
  1204→        const p = wbPriorityWeight(a.priority) - wbPriorityWeight(b.priority);
  1205→        if (p) return p;
  1206→        const ad = a.deadline ? new Date(a.deadline).getTime() : Infinity;
  1207→        const bd = b.deadline ? new Date(b.deadline).getTime() : Infinity;
  1208→        if (ad !== bd) return ad - bd;
  1209→        return (b.updatedAt || '').localeCompare(a.updatedAt || '');
  1210→      })[0] || null
  1211→  );
  1212→}
  1213→export function wbInsights(items: Item[], entries: Entry[], capacity: number): string[] {
  1214→  const today = todayStr();
  1215→  const a: string[] = [];
  1216→  const overdue = items.filter(
  1217→    (i) => i.deadline && i.status !== 'done' && i.status !== 'archived' && i.deadline < today
  1218→  ).length;
  1219→  const blocked = items.filter((i) => wbIsBlocked(items, i)).length;
  1220→  const planned = wbPlannedMinutes(items, today);
  1221→  const logged = entriesInRange(entries, today, today).reduce((s, e) => s + e.durationMin, 0);
  1222→  if (overdue) a.push(`${overdue} item${overdue === 1 ? ' is' : 's are'} overdue — pull the most important one into your next focus block.`);
  1223→  if (blocked) a.push(`${blocked} item${blocked === 1 ? ' is' : 's are'} blocked by dependencies. Resolve them before adding more work.`);
  1224→  if (planned > capacity) a.push(`Today is over capacity by ${planned - capacity} minutes. Trim the plan before starting another block.`);
  1225→  if (logged >= 240) a.push(`${Math.round(logged / 60)}h logged today — protect your recovery time.`);
  1226→  if (!a.length) a.push('The system is stable. Protect your next action and keep capacity realistic.');
  1227→  return a;
  1228→}

  1234→export function wbFileReadPages(file: ResourceFile): number {

  1236→  for (let p = 1; p <= file.pageCount; p++) {

  1238→    if (file.readPages.includes(key) || (file.pageSeconds[key] || 0) >= PAGE_READ_THRESHOLD_SEC) count++;
  1239→  }
  1240→  return count;
  1241→}
