     1→     1→/* ==================== Library: Reader Overlay (pdf.js canvas) ====================
     2→     2→   Full-screen reading surface for a stored PDF.
     3→     3→   - Fit-width / zoom rendering with serialized page paints (no render races)
     4→     4→   - Per-page reading-second tracking (flushed to the store every 10s / page change / close)
     5→     5→   - Focus-read integration (pomodoro started from the reader)
     6→     6→   - Immersive mode with auto-hiding chrome */

    43→    43→function clampPage(p: number, total: number): number {
    44→    44→  const max = Math.max(1, total);
    45→    45→  const n = Math.round(p);
    46→    46→  return Math.max(1, Math.min(max, Number.isFinite(n) ? n : 1));
    47→    47→}

    49→    49→function fmtRem(secs: number): string {
    50→    50→  const s = Math.max(0, Math.floor(secs));
    51→    51→  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`;
    52→    52→}

    54→    54→export function ReaderOverlay({ fileId, onClose }: { fileId: string; onClose: () => void }) {
    55→    55→  /* ---------- store ---------- */
    56→    56→  const file = useWB((s) => s.files.find((f) => f.id === fileId));
    57→    57→  const items = useWB((s) => s.items);
    58→    58→  const activeTimer = useWB((s) => s.activeTimer);
    59→    59→  const setFilePosition = useWB((s) => s.setFilePosition);
    60→    60→  const addFilePageSeconds = useWB((s) => s.addFilePageSeconds);
    61→    61→  const markFilePage = useWB((s) => s.markFilePage);
    62→    62→  const logReadingSession = useWB((s) => s.logReadingSession);
    63→    63→  const startFileFocus = useWB((s) => s.startFileFocus);
    64→    64→  const pausePomodoro = useWB((s) => s.pausePomodoro);
    65→    65→  const resumePomodoro = useWB((s) => s.resumePomodoro);
    66→    66→  const advancePomodoro = useWB((s) => s.advancePomodoro);
    67→    67→  const stopTimer = useWB((s) => s.stopTimer);
    68→    68→
    69→    69→  const now = useNow();
    70→    70→
    71→    71→  /* ---------- view state ---------- */
    72→    72→  const [doc, setDoc] = useState<PDFDocumentProxy | null>(null);
    73→    73→  const [loadState, setLoadState] = useState<LoadState>('loading');
    74→    74→  const [loadError, setLoadError] = useState('');
    75→    75→  const [page, setPage] = useState(() => clampPage(file?.currentPage ?? 1, file?.pageCount ?? 1));
    76→    76→  const [pageInput, setPageInput] = useState(() => String(clampPage(file?.currentPage ?? 1, file?.pageCount ?? 1)));
    77→    77→  const [fit, setFit] = useState(true);
    78→    78→  const [scale, setScale] = useState(1);
    79→    79→  const [baseW, setBaseW] = useState(0);
    80→    80→  const [stageW, setStageW] = useState(0);
    81→    81→  const [rendering, setRendering] = useState(false);
    82→    82→  const [hasPaint, setHasPaint] = useState(false);
    83→    83→  const [immersive, setImmersive] = useState(false);
    84→    84→  const [chromeOn, setChromeOn] = useState(true);
    85→    85→
    86→    86→  const numPages = doc?.numPages ?? file?.pageCount ?? 1;
    87→    87→  const effScale = fit && baseW > 0 && stageW > 0 ? Math.max(0.05, stageW / baseW) : scale;
    88→    88→
    89→    89→  /* ---------- refs (session tracking + render serialization) ---------- */
    90→    90→  const stageRef = useRef<HTMLDivElement | null>(null);
    91→    91→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
    92→    92→  const pageInputRef = useRef<HTMLInputElement | null>(null);
    93→    93→  const sessionStartRef = useRef(Date.now());
    94→    94→  const startPageRef = useRef(page);
    95→    95→  const pageRef = useRef(page);
    96→    96→  const pageSecsRef = useRef(0); // visible seconds on the current page (display)
    97→    97→  const pendingRef = useRef<Map<number, number>>(new Map()); // page -> unflushed seconds
    98→    98→  const sessionTotalsRef = useRef<Map<number, number>>(new Map()); // page -> session seconds
    99→    99→  const visitedRef = useRef<Set<number>>(new Set());
   100→   100→  const newlyReadRef = useRef<Set<number>>(new Set());
   101→   101→  const renderSeqRef = useRef(0);
   102→   102→  const renderChainRef = useRef<Promise<void>>(Promise.resolve());
   103→   103→  const idleTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   104→   104→
   105→   105→  /* immutable snapshot of pageSeconds/readPages at mount — threshold crossing math */
   106→   106→  const snapshotRef = useRef<{ secs: Record<string, number>; read: Set<string> } | null>(null);
   107→   107→  if (file && snapshotRef.current === null) {
   108→   108→    snapshotRef.current = { secs: { ...file.pageSeconds }, read: new Set(file.readPages) };
   109→   109→  }
   110→   110→
   111→   111→  /* ---------- derived ---------- */
   112→   112→  const elapsedSec = Math.max(0, Math.floor((now.getTime() - sessionStartRef.current) / 1000));
   113→   113→  const timeOnPage = pageSecsRef.current;
   114→   114→  const fileProgress = file ? wbFileProgress(file) : 0;
   115→   115→  const readCount = file ? wbFileReadPages(file) : 0;
   116→   116→  const pageIsRead = !!file && file.readPages.includes(String(page));
   117→   117→  const linkedItems = useMemo(() => wbItemsForFile(items, fileId), [items, fileId]);
   118→   118→
   119→   119→  const focusHere = !!activeTimer && activeTimer.fileId === fileId;
   120→   120→  const focusTiming = focusHere && activeTimer ? getFocusTiming(activeTimer, now.getTime()) : null;
   121→   121→  const ftPaused = focusTiming?.paused ?? false;
   122→   122→  const ftRemaining = focusTiming?.remaining ?? 0;
   123→   123→  const ftPhase = focusTiming?.phaseLabel ?? '';
   124→   124→  const phaseTone = activeTimer?.pomodoro?.phase === 'break' ? 'break' : 'work';
   125→   125→
   126→   126→  const deadlineChip = useMemo(() => {
   127→   127→    if (!file?.deadline) return null;
   128→   128→    const status = wbFileDeadlineStatus(file);
   129→   129→    if (!status) return null;
   130→   130→    if (fileProgress >= 100) return { text: 'Complete', tone: 'ok' as const };
   131→   131→    const d = daysUntil(file.deadline);
   132→   132→    if (d < 0) return { text: `${Math.abs(d)}d overdue`, tone: 'over' as const };
   133→   133→    if (d === 0) return { text: 'Due today', tone: 'soon' as const };
   134→   134→    return { text: `${d}d left`, tone: status === 'soon' ? ('soon' as const) : ('ok' as const) };
   135→   135→  }, [file, fileProgress]);
   136→   136→
   137→   137→  /* ---------- reading-time tracking ---------- */
   138→   138→
   139→   139→  const flushPending = useCallback(() => {
   140→   140→    const pending = pendingRef.current;
   141→   141→    if (pending.size === 0) return;
   142→   142→    const snap = snapshotRef.current;
   143→   143→    for (const [pg, secs] of pending) {
   144→   144→      pending.delete(pg);
   145→   145→      if (secs <= 0) continue;
   146→   146→      addFilePageSeconds(fileId, pg, secs);
   147→   147→      const sess = (sessionTotalsRef.current.get(pg) || 0) + secs;
   148→   148→      sessionTotalsRef.current.set(pg, sess);
   149→   149→      if (snap) {
   150→   150→        const before = snap.secs[String(pg)] || 0;
   151→   151→        if (!snap.read.has(String(pg)) && before < PAGE_READ_THRESHOLD_SEC && before + sess >= PAGE_READ_THRESHOLD_SEC) {
   152→   152→          newlyReadRef.current.add(pg);
   153→   153→        }
   154→   154→      }
   155→   155→    }
   156→   156→  }, [fileId, addFilePageSeconds]);
   157→   157→
   158→   158→  /* 1-second tick — only while the tab is visible */
   159→   159→  useEffect(() => {
   160→   160→    const id = setInterval(() => {
   161→   161→      if (document.visibilityState !== 'visible') return;
   162→   162→      const p = pageRef.current;
   163→   163→      pendingRef.current.set(p, (pendingRef.current.get(p) || 0) + 1);
   164→   164→      pageSecsRef.current += 1;
   165→   165→    }, 1000);
   166→   166→    return () => clearInterval(id);
   167→   167→  }, []);
   168→   168→
   169→   169→  /* mark the entry page as visited */
   170→   170→  useEffect(() => {
   171→   171→    visitedRef.current.add(pageRef.current);
   172→   172→  }, []);
   173→   173→
   174→   174→  /* periodic flush */
   175→   175→  useEffect(() => {
   176→   176→    const id = setInterval(() => flushPending(), FLUSH_EVERY_MS);
   177→   177→    return () => clearInterval(id);
   178→   178→  }, [flushPending]);
   179→   179→
   180→   180→  /* final flush + session log on unmount */
   181→   181→  useEffect(() => {
   182→   182→    const sessionStart = sessionStartRef.current;
   183→   183→    const startPage = startPageRef.current;
   184→   184→    return () => {
   185→   185→      flushPending();
   186→   186→      const elapsed = Date.now() - sessionStart;
   187→   187→      if (elapsed < MIN_SESSION_MS) return;
   188→   188→      logReadingSession({
   189→   189→        fileId,
   190→   190→        date: isoDate(new Date(sessionStart)),
   191→   191→        startedAt: new Date(sessionStart).toISOString(),
   192→   192→        endedAt: new Date().toISOString(),
   193→   193→        minutes: Math.max(1, Math.round(elapsed / 60000)),
   194→   194→        pagesVisited: visitedRef.current.size,
   195→   195→        pagesRead: newlyReadRef.current.size,
   196→   196→        startPage,
   197→   197→        endPage: pageRef.current,
   198→   198→      });
   199→   199→    };
   200→   200→  }, [fileId, flushPending, logReadingSession]);
   201→   201→
   202→   202→  /* lock background scroll while the overlay is up */
   203→   203→  useEffect(() => {
   204→   204→    const prev = document.body.style.overflow;
   205→   205→    document.body.style.overflow = 'hidden';
   206→   206→    return () => {
   207→   207→      document.body.style.overflow = prev;
   208→   208→    };
   209→   209→  }, []);
   210→   210→
   211→   211→  /* ---------- document loading ---------- */
   212→   212→
   213→   213→  useEffect(() => {
   214→   214→    let alive = true;
   215→   215→    loadFileDoc(fileId)
   216→   216→      .then((d) => {
   217→   217→        if (!alive) return;
   218→   218→        setDoc(d);
   219→   219→        setLoadState('ready');
   220→   220→      })
   221→   221→      .catch((err: unknown) => {
   222→   222→        if (!alive) return;
   223→   223→        setLoadState('error');
   224→   224→        setLoadError(err instanceof Error ? err.message : 'Could not open this PDF.');
   225→   225→      });
   226→   226→    return () => {
   227→   227→      alive = false;
   228→   228→    };
   229→   229→  }, [fileId]);
   230→   230→
   231→   231→  /* base page width (scale 1) for fit-width math */
   232→   232→  useEffect(() => {
   233→   233→    if (!doc) return;
   234→   234→    let alive = true;
   235→   235→    doc
   236→   236→      .getPage(clampPage(page, doc.numPages))
   237→   237→      .then((pg) => {
   238→   238→        if (alive) setBaseW(pg.getViewport({ scale: 1 }).width);
   239→   239→      })
   240→   240→      .catch(() => {});
   241→   241→    return () => {
   242→   242→      alive = false;
   243→   243→    };
   244→   244→  }, [doc, page]);
   245→   245→
   246→   246→  /* clamp page if the doc turns out shorter than stored metadata */
   247→   247→  useEffect(() => {
   248→   248→    if (doc && page > doc.numPages) {
   249→   249→      const target = clampPage(page, doc.numPages);
   250→   250→      setPage(target);
   251→   251→      setPageInput(String(target));
   252→   252→      pageRef.current = target;
   253→   253→      pageSecsRef.current = 0;
   254→   254→    }
   255→   255→  }, [doc, page]);
   256→   256→
   257→   257→  /* available stage width (debounced, exact against computed padding) */
   258→   258→  useEffect(() => {
   259→   259→    const stage = stageRef.current;
   260→   260→    if (!stage) return;
   261→   261→    let timer: ReturnType<typeof setTimeout> | null = null;
   262→   262→    const measure = () => {
   263→   263→      const cs = window.getComputedStyle(stage);
   264→   264→      const pad = parseFloat(cs.paddingLeft || '0') + parseFloat(cs.paddingRight || '0');
   265→   265→      setStageW(Math.max(0, stage.clientWidth - pad - 12));
   266→   266→    };
   267→   267→    const ro = new ResizeObserver(() => {
   268→   268→      if (timer) clearTimeout(timer);
   269→   269→      timer = setTimeout(measure, 140);
   270→   270→    });
   271→   271→    ro.observe(stage);
   272→   272→    measure();
   273→   273→    return () => {
   274→   274→      ro.disconnect();
   275→   275→      if (timer) clearTimeout(timer);
   276→   276→    };
   277→   277→  }, []);
   278→   278→
   279→   279→  /* serialized canvas painting — latest request wins, no concurrent renders */
   280→   280→  useEffect(() => {
   281→   281→    if (!doc || loadState !== 'ready' || effScale <= 0) return;
   282→   282→    const seq = ++renderSeqRef.current;
   283→   283→    const prev = renderChainRef.current;
   284→   284→    const run = (async () => {
   285→   285→      await prev;
   286→   286→      if (seq !== renderSeqRef.current) return;
   287→   287→      const canvas = canvasRef.current;
   288→   288→      if (!canvas) return;
   289→   289→      setRendering(true);
   290→   290→      try {
   291→   291→        await renderPageToCanvas(doc, page, canvas, { scale: effScale });
   292→   292→        if (seq === renderSeqRef.current) {
   293→   293→          setRendering(false);
   294→   294→          setHasPaint(true);
   295→   295→        }
   296→   296→      } catch {
   297→   297→        if (seq === renderSeqRef.current) setRendering(false);
   298→   298→      }
   299→   299→    })();
   300→   300→    renderChainRef.current = run.catch(() => {});
   301→   301→  }, [doc, page, effScale, loadState]);
   302→   302→
   303→   303→  /* ---------- navigation ---------- */
   304→   304→
   305→   305→  const goToPage = useCallback(
   306→   306→    (next: number) => {
   307→   307→      const total = doc?.numPages ?? file?.pageCount ?? 1;
   308→   308→      const target = clampPage(next, total);
   309→   309→      if (target !== page) {
   310→   310→        flushPending();
   311→   311→        visitedRef.current.add(target);
   312→   312→        pageRef.current = target;
   313→   313→        pageSecsRef.current = 0;
   314→   314→        setPage(target);
   315→   315→        setFilePosition(fileId, target);
   316→   316→        stageRef.current?.scrollTo({ top: 0 });
   317→   317→      }
   318→   318→      setPageInput(String(target));
   319→   319→    },
   320→   320→    [doc, file, fileId, page, flushPending, setFilePosition]
   321→   321→  );
   322→   322→
   323→   323→  const applyPageInput = useCallback(() => {
   324→   324→    const n = parseInt(pageInput, 10);
   325→   325→    if (Number.isFinite(n)) goToPage(n);
   326→   326→    else setPageInput(String(page));
   327→   327→  }, [pageInput, goToPage, page]);
   328→   328→
   329→   329→  const closeReader = useCallback(() => {
   330→   330→    onClose();
   331→   331→  }, [onClose]);
   332→   332→
   333→   333→  /* ---------- keyboard ---------- */
   334→   334→
   335→   335→  useEffect(() => {
   336→   336→    const onKey = (e: KeyboardEvent) => {
   337→   337→      const t = e.target as HTMLElement | null;
   338→   338→      const inField =
   339→   339→        !!t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.tagName === 'SELECT' || t.isContentEditable);
   340→   340→      if (e.key === 'Escape') {
   341→   341→        if (inField) {
   342→   342→          pageInputRef.current?.blur();
   343→   343→          setPageInput(String(page));
   344→   344→          return;
   345→   345→        }
   346→   346→        if (document.querySelector('.modal-overlay')) return; // a modal above owns Esc
   347→   347→        e.preventDefault();
   348→   348→        if (immersive) {
   349→   349→          setImmersive(false);
   350→   350→          setChromeOn(true);
   351→   351→        } else {
   352→   352→          closeReader();
   353→   353→        }
   354→   354→        return;
   355→   355→      }
   356→   356→      if (inField || document.querySelector('.modal-overlay')) return;
   357→   357→      if (e.key === 'ArrowLeft' || e.key === 'PageUp') {
   358→   358→        e.preventDefault();
   359→   359→        goToPage(page - 1);
   360→   360→      } else if (e.key === 'ArrowRight' || e.key === 'PageDown') {
   361→   361→        e.preventDefault();
   362→   362→        goToPage(page + 1);
   363→   363→      } else if (e.key === 'Home') {
   364→   364→        e.preventDefault();
   365→   365→        goToPage(1);
   366→   366→      } else if (e.key === 'End') {
   367→   367→        e.preventDefault();
   368→   368→        goToPage(numPages);
   369→   369→      }
   370→   370→    };
   371→   371→    window.addEventListener('keydown', onKey);
   372→   372→    return () => window.removeEventListener('keydown', onKey);
   373→   373→  }, [page, numPages, goToPage, immersive, closeReader]);
   374→   374→
   375→   375→  /* ---------- immersive chrome ---------- */
   376→   376→
   377→   377→  const bumpChrome = useCallback(() => {
   378→   378→    setChromeOn(true);
   379→   379→    if (idleTimerRef.current) clearTimeout(idleTimerRef.current);
   380→   380→    idleTimerRef.current = setTimeout(() => setChromeOn(false), CHROME_IDLE_MS);
   381→   381→  }, []);
   382→   382→
   383→   383→  useEffect(() => {
   384→   384→    if (!immersive) return;
   385→   385→    bumpChrome();
   386→   386→    const onMove = () => bumpChrome();
   387→   387→    window.addEventListener('mousemove', onMove);
   388→   388→    return () => {
   389→   389→      window.removeEventListener('mousemove', onMove);
   390→   390→      if (idleTimerRef.current) {
   391→   391→        clearTimeout(idleTimerRef.current);
   392→   392→        idleTimerRef.current = null;
   393→   393→      }
   394→   394→    };
   395→   395→  }, [immersive, bumpChrome]);
   396→   396→
   397→   397→  const toggleImmersive = useCallback(() => {
   398→   398→    const next = !immersive;
   399→   399→    setImmersive(next);
   400→   400→    if (!next) {
   401→   401→      if (idleTimerRef.current) {
   402→   402→        clearTimeout(idleTimerRef.current);
   403→   403→        idleTimerRef.current = null;
   404→   404→      }
   405→   405→      setChromeOn(true);
   406→   406→    }
   407→   407→  }, [immersive]);
   408→   408→
   409→   409→  /* tap/click on the page reveals chrome while immersive (touch support) */
   410→   410→  const onStageClick = () => {
   411→   411→    if (immersive && !chromeOn) bumpChrome();
   412→   412→  };
   413→   413→
   414→   414→  /* ---------- zoom ---------- */
   415→   415→
   416→   416→  const zoomBy = useCallback(
   417→   417→    (mult: number) => {
   418→   418→      setScale(Math.min(ZOOM_MAX, Math.max(ZOOM_MIN, effScale * mult)));
   419→   419→      setFit(false);
   420→   420→    },
   421→   421→    [effScale]
   422→   422→  );
   423→   423→
   424→   424→  /* ---------- focus read ---------- */
   425→   425→
   426→   426→  const startFocusRead = () => {
   427→   427→    if (startFileFocus(fileId)) toast('Focus read started', 'Pomodoro running — page time still counts.');
   428→   428→    else toast('A timer is already running', 'End it before starting a focus read.');
   429→   429→  };
   430→   430→
   431→   431→  const endFocusRead = () => {
   432→   432→    stopTimer();
   433→   433→    toast('Session logged');
   434→   434→  };
   435→   435→
   436→   436→  /* auto-advance the pomodoro while focus-reading here */
   437→   437→  useEffect(() => {
   438→   438→    if (!focusHere || ftPaused || ftRemaining > 0) return;
   439→   439→    advancePomodoro();
   440→   440→    toast(
   441→   441→      ftPhase === 'Focus' ? 'Break time! ☕' : 'Back to work! 🧠',
   442→   442→      ftPhase === 'Focus' ? 'Focus phase logged.' : 'Break over — next cycle starts now.'
   443→   443→    );
   444→   444→  }, [focusHere, ftPaused, ftRemaining, ftPhase, advancePomodoro]);
   445→   445→
