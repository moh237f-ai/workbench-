    30→const GREG_WD = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
    31→
    32→/** weekday index inside the system week: Shamsi Sat=0…Fri=6, Gregorian Mon=0…Sun=6 */
    33→function weekdayIndex(d: Date, system: CalendarSystem): number {
    34→  return system === 'shamsi' ? (d.getDay() + 1) % 7 : (d.getDay() + 6) % 7;
    35→}
    36→
    37→/** ISO of the week start (Saturday for Shamsi, Monday for Gregorian) containing d. */
    38→function weekStartIso(d: Date, system: CalendarSystem): string {
    39→  const r = new Date(d.getFullYear(), d.getMonth(), d.getDate());
    40→  r.setDate(r.getDate() - weekdayIndex(r, system));
    41→  return isoDate(r);
    42→}
    43→
    44→function gregShort(iso: string): string {
    45→  return parseDate(iso).toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: 'numeric' });
    46→}
    47→
    48→function shamsiShort(iso: string): string {
    49→  return formatShamsi(dateToShamsi(parseDate(iso)), true);
    50→}
    51→
    52→function dualDate(iso: string, system: CalendarSystem): string {
    53→  return system === 'shamsi' ? `${shamsiShort(iso)} · ${gregShort(iso)}` : `${gregShort(iso)} · ${shamsiShort(iso)}`;
    54→}
    55→
    56→function weekdayShort(d: Date, system: CalendarSystem): string {
    57→  return system === 'shamsi' ? SHAMSI_WEEKDAYS_SHORT[(d.getDay() + 1) % 7] : GREG_WD[(d.getDay() + 6) % 7];
    58→}
    59→
    60→function countDaysBetween(a: string, b: string): number {
    61→  if (!a || !b || b < a) return 0;
    62→  return Math.round((parseDate(b).getTime() - parseDate(a).getTime()) / 86400000) + 1;
    63→}
    64→
    65→/** contribution cell intensity: 0 → empty, 1: 1-30m, 2: 30-60m, 3: 1-2h, 4: 2h+ */
    66→function cgLevel(m: number): number {
    67→  if (m <= 0) return 0;
    68→  if (m <= 30) return 1;
    69→  if (m <= 60) return 2;
    70→  if (m <= 120) return 3;
    71→  return 4;
    72→}

   154→/** Average progress across a roadmap's measurable nodes
   155→ *  (goal/task → live item progress, file → file progress, milestone → done, note → excluded). */
   156→function roadmapProgress(rm: Roadmap, items: Item[], files: ResourceFile[]): number {
   157→  const vals: number[] = [];
   158→  rm.nodes.forEach((n) => {
   159→    if (n.kind === 'note') return;
   160→    if (n.kind === 'milestone') {
   161→      vals.push(n.done ? 100 : 0);
   162→      return;
   163→    }
   164→    if (!n.refId) return;
   165→    if (n.kind === 'file') {
   166→      const f = files.find((x) => x.id === n.refId);
   167→      if (f) vals.push(wbFileProgress(f));
   168→      return;
   169→    }
   170→    const it = items.find((x) => x.id === n.refId);
   171→    if (it) vals.push(computeProgress(it, (id) => wbChildren(items, id)));
   172→  });
   173→  return vals.length ? Math.round(vals.reduce((a, b) => a + b, 0) / vals.length) : 0;
   174→}

   176→/** Planned timebox duration in minutes (plannedEnd − plannedStart, 30m fallback when open-ended). */
   177→function timeboxMinutes(i: Item): number {
   178→  if (!i.plannedStart) return 0;
   179→  const start = hhmmToMin(i.plannedStart.slice(11, 16));
   180→  const end = i.plannedEnd ? hhmmToMin(i.plannedEnd.slice(11, 16)) : start + 30;
   181→  return Math.max(0, end - start);
   182→}

   236→  const stats = useWB((s) => s.stats);

   239→  const system = useWB((s) => s.calendarSystem);
   240→  const capacity = useWB((s) => s.dailyCapacityMin);

   242→  const readingSessions = useWB((s) => s.readingSessions);
   243→  const filePlans = useWB((s) => s.filePlans);
   244→  const roadmaps = useWB((s) => s.roadmaps);
   245→  const habits = useWB((s) => s.habits);
   246→  const [logOpen, setLogOpen] = useState(false);
   247→
   248→  const [from, to] = useMemo(() => rangeToDates(range, reportCustom), [range, reportCustom]);
   249→  const isAllTime = from === '0000-01-01';
   250→
   251→  /* ----- per-day activity maps ----- */
   252→  const entriesByDay = useMemo(() => {
   253→    const m: Record<string, number> = {};
   254→    entries.forEach((e) => { m[e.date] = (m[e.date] || 0) + e.durationMin; });
   255→    return m;
   256→  }, [entries]);
   257→
   258→  const readingByDay = useMemo(() => {
   259→    const m: Record<string, number> = {};
   260→    readingSessions.forEach((r) => { m[r.date] = (m[r.date] || 0) + r.minutes; });
   261→    return m;
   262→  }, [readingSessions]);
   263→
   264→  const minutesByDay = useMemo(() => {
   265→    const m: Record<string, number> = { ...entriesByDay };
   266→    readingSessions.forEach((r) => { m[r.date] = (m[r.date] || 0) + r.minutes; });
   267→    return m;
   268→  }, [entriesByDay, readingSessions]);
   269→
   270→  const activeMap = useMemo(() => {
   271→    const m: Record<string, boolean> = {};
   272→    Object.keys(minutesByDay).forEach((d) => { if (minutesByDay[d] > 0) m[d] = true; });
   273→    return m;
   274→  }, [minutesByDay]);
   275→
   276→  const currentStreak = useMemo(() => computeStreak(activeMap), [activeMap]);
   277→
   278→  /* ----- effective span (all-time → earliest activity … today) ----- */
   279→  const effectiveFrom = useMemo(() => {
   280→    if (!isAllTime) return from;
   281→    const keys = Object.keys(minutesByDay).sort();
   282→    return keys[0] || todayStr();
   283→  }, [isAllTime, from, minutesByDay]);
   284→
   285→  const effectiveTo = to === '9999-12-31' ? todayStr() : to;
   286→  const statTo = useMemo(() => {
   287→    const t = todayStr();
   288→    return effectiveTo > t ? t : effectiveTo;
   289→  }, [effectiveTo]);
   290→  const statDays = useMemo(() => Math.max(0, countDaysBetween(effectiveFrom, statTo)), [effectiveFrom, statTo]);
   291→
   292→  /* ----- range-scoped entries + totals ----- */
   293→  const scoped = useMemo(() => entriesInRange(entries, from, to), [entries, from, to]);
   294→  const readingMinutes = wbReadingMinutesInRange(readingSessions, from, to);
   295→  const readingPages = wbReadingPagesInRange(readingSessions, from, to);
   296→
   297→  const rangeStats = useMemo(() => {
   298→    const totalLogged = scoped.reduce((s, e) => s + e.durationMin, 0);
   299→    const totalProductive = totalLogged + readingMinutes;
   300→    const days = statDays;
   301→    const activeDays = Object.keys(minutesByDay).filter(
   302→      (d) => d >= effectiveFrom && d <= statTo && minutesByDay[d] > 0
   303→    ).length;
   304→    const avgPerDay = days > 0 ? totalProductive / days : 0;
   305→    const zeroDays = Math.max(0, days - activeDays);
   306→    let bestIso = '';
   307→    let bestVal = 0;
   308→    Object.keys(minutesByDay).forEach((d) => {
   309→      if (d < effectiveFrom || d > statTo) return;
   310→      if (minutesByDay[d] > bestVal) { bestVal = minutesByDay[d]; bestIso = d; }
   311→    });
   312→    /* longest consecutive active run inside the range */
   313→    let bestRun = 0;
   314→    let run = 0;
   315→    if (days > 0) {
   316→      let cur = parseDate(effectiveFrom);
   317→      const end = parseDate(statTo);
   318→      let guard = 0;
   319→      while (cur <= end && guard < 4000) {
   320→        if (minutesByDay[isoDate(cur)] > 0) { run += 1; if (run > bestRun) bestRun = run; }
   321→        else run = 0;
   322→        cur = new Date(cur.getFullYear(), cur.getMonth(), cur.getDate() + 1);
   323→        guard += 1;
   324→      }
   325→    }
   326→    return { totalLogged, totalProductive, activeDays, avgPerDay, zeroDays, bestIso, bestVal, bestRun };
   327→  }, [scoped, readingMinutes, statDays, effectiveFrom, statTo, minutesByDay]);
   328→
   329→  /* ----- by-label / by-item (range-scoped) ----- */
   330→  const byLabel = useMemo(() => {
   331→    const m: Record<string, number> = {};
   332→    scoped.forEach((e) => {
   333→      if (!e.labelIds.length) {
   334→        m._none = (m._none || 0) + e.durationMin;
   335→        return;
   336→      }
   337→      e.labelIds.forEach((lid) => {
   338→        m[lid] = (m[lid] || 0) + e.durationMin;
   339→      });
   340→    });
   341→    return Object.entries(m).sort((a, b) => b[1] - a[1]);
   342→  }, [scoped]);
   343→
   344→  const byItem = useMemo(() => {
   345→    const m: Record<string, number> = {};
   346→    scoped.forEach((e) => {
   347→      e.itemIds.forEach((iid) => {
   348→        m[iid] = (m[iid] || 0) + e.durationMin;
   349→      });
   350→    });
   351→    return Object.entries(m).sort((a, b) => b[1] - a[1]).slice(0, 8);
   352→  }, [scoped]);
   353→
   354→  const maxLabelMin = Math.max(1, ...byLabel.map((r) => r[1]));
   355→  const maxItemMin = Math.max(1, ...byItem.map((r) => r[1]));
   356→
   357→  /* ----- day list (capped to the most recent 800 days) ----- */
   358→  const allDays = useMemo(() => {
   359→    let startIso = effectiveFrom;
   360→    if (countDaysBetween(startIso, effectiveTo) > 800) {
   361→      const d = parseDate(effectiveTo);
   362→      d.setDate(d.getDate() - 799);
   363→      startIso = isoDate(d);
   364→    }
   365→    const out: string[] = [];
   366→    let cur = parseDate(startIso);
   367→    const end = parseDate(effectiveTo);
   368→    let guard = 0;
   369→    while (cur <= end && guard < 810) {
   370→      out.push(isoDate(cur));
   371→      cur = new Date(cur.getFullYear(), cur.getMonth(), cur.getDate() + 1);
   372→      guard += 1;
   373→    }
   374→    return out;
   375→  }, [effectiveFrom, effectiveTo]);
   376→
   377→  /* ----- by day / by week chart data (weekly buckets for long ranges) ----- */
   378→  const dayChartData: DayChartData = useMemo(() => {
   379→    if (allDays.length <= 62) {
   380→      return {
   381→        weekly: false,
   382→        points: allDays.map((iso) => {
   383→          const d = parseDate(iso);
   384→          return {
   385→            key: iso,
   386→            minutes: minutesByDay[iso] || 0,
   387→            reading: readingByDay[iso] || 0,
   388→            label: String(d.getDate()),
   389→            sub: system === 'shamsi' ? String(dateToShamsi(d).jd) : null,
   390→            tip: `${weekdayShort(d, system)} ${dualDate(iso, system)}`,
   391→          };
   392→        }),
   393→      };
   394→    }
   395→    const buckets: { key: string; minutes: number; reading: number }[] = [];
   396→    let curKey = '';
   397→    allDays.forEach((iso) => {
   398→      const k = weekStartIso(parseDate(iso), system);
   399→      if (k !== curKey) { buckets.push({ key: k, minutes: 0, reading: 0 }); curKey = k; }
   400→      const b = buckets[buckets.length - 1];
   401→      b.minutes += minutesByDay[iso] || 0;
   402→      b.reading += readingByDay[iso] || 0;
   403→    });
   404→    return {
   405→      weekly: true,
   406→      points: buckets.map((b) => {
   407→        const d = parseDate(b.key);
   408→        return {
   409→          key: b.key,
   410→          minutes: b.minutes,
   411→          reading: b.reading,
   412→          label: String(d.getDate()),
   413→          sub: system === 'shamsi' ? String(dateToShamsi(d).jd) : null,
   414→          tip: `Week of ${weekdayShort(d, system)} ${dualDate(b.key, system)}`,
   415→        };
   416→      }),
   417→    };
   418→  }, [allDays, system, minutesByDay, readingByDay]);
   419→
   420→  const maxDayMin = Math.max(1, ...dayChartData.points.map((p) => p.minutes));
   421→  const maxReadingDay = Math.max(1, ...dayChartData.points.map((p) => p.reading));
   422→
   423→  /* ----- by-month section (yearly report) ----- */
   424→  const monthRows: MonthRow[] | null = useMemo(() => {
   425→    const spanDays = countDaysBetween(effectiveFrom, effectiveTo);
   426→    const showMonths = range === 'year' || range === 'lastyear' || spanDays >= 183;
   427→    if (!showMonths) return null;
   428→
   429→    const out: MonthRow[] = [];
   430→    const end = parseDate(effectiveTo);
   431→    if (system === 'shamsi') {
   432→      let cur = dateToShamsi(parseDate(effectiveFrom));
   433→      for (let guard = 0; guard < 64; guard++) {
   434→        const a = shamsiToGreg(cur.jy, cur.jm, 1);
   435→        const b = shamsiToGreg(cur.jy, cur.jm, shamsiMonthDays(cur.jy, cur.jm));

   437→          key: `s-${cur.jy}-${cur.jm}`,
   438→          label: SHAMSI_MONTHS_SHORT[cur.jm - 1],
   439→          full: `${SHAMSI_MONTHS[cur.jm - 1]} ${cur.jy}`,
   440→          yearLabel: cur.jm === 1 ? String(cur.jy) : null,
   441→          startIso: isoDate(new Date(a.gy, a.gm - 1, a.gd)),
   442→          endIso: isoDate(new Date(b.gy, b.gm - 1, b.gd)),
   443→          logged: 0,
   444→          reading: 0,
   445→          entryCount: 0,
   446→        });
   447→        if (parseDate(out[out.length - 1].endIso) >= end) break;
   448→        cur = cur.jm === 12 ? { jy: cur.jy + 1, jm: 1, jd: 1 } : { jy: cur.jy, jm: cur.jm + 1, jd: 1 };
   449→      }
   450→    } else {
   451→      let y = parseDate(effectiveFrom).getFullYear();
   452→      let m = parseDate(effectiveFrom).getMonth();
   453→      for (let guard = 0; guard < 64; guard++) {
   454→        out.push({
   455→          key: `g-${y}-${m}`,
   456→          label: new Date(y, m, 15).toLocaleDateString('en-GB', { month: 'short' }),
   457→          full: new Date(y, m, 15).toLocaleDateString('en-GB', { month: 'long', year: 'numeric' }),
   458→          yearLabel: m === 0 ? String(y) : null,
   459→          startIso: isoDate(new Date(y, m, 1)),
   460→          endIso: isoDate(new Date(y, m + 1, 0)),
   461→          logged: 0,
   462→          reading: 0,
   463→          entryCount: 0,
   464→        });
   465→        if (parseDate(out[out.length - 1].endIso) >= end) break;
   466→        m += 1;
   467→        if (m > 11) { m = 0; y += 1; }
   468→      }
   469→    }

   472→      const monthEntries = entriesInRange(entries, b.startIso, b.endIso);

   474→      const reading = readingSessions
   475→        .filter((r) => r.date >= b.startIso && r.date <= b.endIso)
   476→        .reduce((s, r) => s + r.minutes, 0);
   477→      return { ...b, logged, reading, entryCount: monthEntries.length };
   478→    });
   479→  }, [range, system, effectiveFrom, effectiveTo, entries, readingSessions]);
   480→
   481→  const maxMonthMin = monthRows ? Math.max(1, ...monthRows.map((m) => m.logged)) : 1;
   482→
   483→  /* ----- contribution graph (heatmap of productive minutes per day) ----- */
   484→  const contribution = useMemo(() => {
   485→    const rangeDayCount = countDaysBetween(effectiveFrom, effectiveTo);
   486→    const weeks = Math.min(26, Math.max(8, Math.ceil(rangeDayCount / 7)));
   487→    const endIso = effectiveTo;
   488→    const ws = parseDate(weekStartIso(parseDate(endIso), system));
   489→    ws.setDate(ws.getDate() - (weeks - 1) * 7);
   490→
   491→    const cols: CgCol[] = [];
   492→    for (let w = 0; w < weeks; w++) {
   493→      const days: CgDay[] = [];
   494→      for (let r = 0; r < 7; r++) {
   495→        const d = new Date(ws.getFullYear(), ws.getMonth(), ws.getDate() + w * 7 + r);
   496→        const iso = isoDate(d);
   497→        days.push({ iso, minutes: minutesByDay[iso] || 0, off: iso > endIso });

   522→    Object.keys(minutesByDay).forEach((d) => {

   525→      if (v > 0) activeDays += 1;
