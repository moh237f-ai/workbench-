    43→    43→function roadmapProgress(rm: Roadmap, liveMap: Map<string, RmNodeLive>): number {
    44→    44→  const vals: number[] = [];
    45→    45→  rm.nodes.forEach((n) => {
    46→    46→    const live = liveMap.get(n.id);
    47→    47→    if (!live) return;
    48→    48→    if (n.kind === 'note') return; // notes carry no progress
    49→    49→    vals.push(live.pct >= 0 ? live.pct : live.done ? 100 : 0);
    50→    50→  });
    51→    51→  if (!vals.length) return 0;
    52→    52→  return Math.round(vals.reduce((a, b) => a + b, 0) / vals.length);
    53→    53→}

    55→    55→/* ---------- execution order computation (module-scope pure) ---------- */
    56→    56→
    57→    57→interface OrderRow {
    58→    58→  id: string;
    59→    59→  idx: number;
    60→    60→  node: RoadmapNode;
    61→    61→  live: RmNodeLive | undefined;
    62→    62→  est: number;
    63→    63→  cum: number;
    64→    64→  blocked: boolean;
    65→    65→}
    66→    66→
    67→    67→function computeOrder(
    68→    68→  roadmap: Roadmap,
    69→    69→  liveMap: Map<string, RmNodeLive>
    70→    70→): { rows: OrderRow[]; totalEst: number; critical: Set<string> } {
    71→    71→  const order = topoOrder(roadmap.nodes, roadmap.edges);
    72→    72→  const estOf = (id: string): number => liveMap.get(id)?.estMin ?? 0;
    73→    73→
    74→    74→  /* running cumulative down the list */
    75→    75→  const rows: OrderRow[] = [];
    76→    76→  let cum = 0;
    77→    77→  order.forEach((id, idx) => {
    78→    78→    const node = roadmap.nodes.find((n) => n.id === id);
    79→    79→    if (!node) return;
    80→    80→    const est = estOf(id);
    81→    81→    cum += est;
    82→    82→    const blocked = roadmap.edges.some((e) => {
    83→    83→      if (e.to !== id) return false;
    84→    84→      const st = liveMap.get(e.from)?.status;
    85→    85→      return !!st && st !== 'done' && st !== 'none';
    86→    86→    });
    87→    87→    rows.push({ id, idx, node, live: liveMap.get(id), est, cum, blocked });
    88→    88→  });
    89→    89→
    90→    90→  /* critical path: longest est chain (DP over topo order) */
    91→    91→  const best = new Map<string, { total: number; prev: string | null }>();
    92→    92→  order.forEach((id) => {
    93→    93→    let prev: string | null = null;
    94→    94→    let prevTotal = -1;
    95→    95→    roadmap.edges.forEach((e) => {
    96→    96→      if (e.to !== id) return;
    97→    97→      const t = best.get(e.from)?.total ?? 0;
    98→    98→      if (t > prevTotal) {
    99→    99→        prevTotal = t;
   100→   100→        prev = e.from;
   101→   101→      }
   102→   102→    });
   103→   103→    best.set(id, { total: estOf(id) + Math.max(0, prevTotal), prev });
   104→   104→  });
   105→   105→  let end: string | null = null;
   106→   106→  let max = 0;
   107→   107→  best.forEach((v, id) => {
   108→   108→    if (v.total > max) {
   109→   109→      max = v.total;
   110→   110→      end = id;
   111→   111→    }
   112→   112→  });
   113→   113→  const critical = new Set<string>();
   114→   114→  let cur: string | null = end && max > 0 ? end : null;
   115→   115→  while (cur) {
   116→   116→    critical.add(cur);
   117→   117→    cur = best.get(cur)?.prev ?? null;
   118→   118→  }
   119→   119→
   120→   120→  return { rows, totalEst: cum, critical };
   121→   121→}

   281→   281→function RoadmapWorkspace({ roadmap }: { roadmap: Roadmap }) {

   288→   288→  const addRoadmapEdge = useWB((s) => s.addRoadmapEdge);
   289→   289→  const removeRoadmapEdge = useWB((s) => s.removeRoadmapEdge);
   290→   290→  const updateRoadmapEdge = useWB((s) => s.updateRoadmapEdge);
   291→   291→  const setActiveRoadmap = useWB((s) => s.setActiveRoadmap);
   292→   292→
   293→   293→  const canvasRef = useRef<RoadmapCanvasHandle>(null);
   294→   294→  const [selection, setSelection] = useState<RmSelection | null>(null);

   299→   299→  const [popSearch, setPopSearch] = useState('');
   300→   300→  const [zoom, setZoom] = useState(1);
   301→   301→  const [fitKey, setFitKey] = useState(0);
   302→   302→
   303→   303→  const liveMap = useMemo(() => {
   304→   304→    const m = new Map<string, RmNodeLive>();
   305→   305→    roadmap.nodes.forEach((n) => m.set(n.id, nodeLive(n, items, files)));
   306→   306→    return m;
   307→   307→  }, [roadmap.nodes, items, files]);
   308→   308→
   309→   309→  const selectedNode =
   310→   310→    selection?.kind === 'node' ? roadmap.nodes.find((n) => n.id === selection.id) || null : null;
   311→   311→  const selectedEdge =
   312→   312→    selection?.kind === 'edge' ? roadmap.edges.find((e) => e.id === selection.id) || null : null;
   313→   313→
   314→   314→  /* keyboard: Esc closes pop / cancels link / clears selection; Delete removes */
   315→   315→  useEffect(() => {
   316→   316→    const onKey = (e: KeyboardEvent) => {
   317→   317→      const t = e.target as HTMLElement;
   318→   318→      if (['INPUT', 'TEXTAREA', 'SELECT'].includes(t?.tagName) || t?.isContentEditable) return;
   319→   319→      if (e.key === 'Escape') {
   320→   320→        if (pop) setPop(null);
   321→   321→        else if (linkSource) setLinkSource(null);
   322→   322→        else if (selection) setSelection(null);
   323→   323→        return;
   324→   324→      }
   325→   325→      if ((e.key === 'Delete' || e.key === 'Backspace') && selection) {
   326→   326→        e.preventDefault();
   327→   327→        if (selection.kind === 'node') {
   328→   328→          removeRoadmapNode(roadmap.id, selection.id);
   329→   329→          toast('Node removed');
   330→   330→        } else {
   331→   331→          removeRoadmapEdge(roadmap.id, selection.id);
   332→   332→          toast('Link removed');
   333→   333→        }
   334→   334→        setSelection(null);

   336→   336→    };

   338→   338→    return () => window.removeEventListener('keydown', onKey);

   340→   340→
   341→   341→  /* ---------- node creation ---------- */
   342→   342→
   343→   343→  const spawn = (partial: Omit<RoadmapNode, 'id'>) => {

   351→   351→

   357→   357→        .filter((f) => !f.archived)

   359→   359→        .slice(0, 40)

   372→   372→      .filter((i) => !['archived'].includes(i.status))
   373→   373→      .filter((i) => !q || i.title.toLowerCase().includes(q))
   374→   374→      .sort((a, b) => (a.status === 'done' ? 1 : 0) - (b.status === 'done' ? 1 : 0))
   375→   375→      .slice(0, 40)
   376→   376→      .map((i) => ({
   377→   377→        id: i.id,
   378→   378→        title: i.title,
   379→   379→        meta: [
   380→   380→          i.status.replace('_', ' '),
   381→   381→          i.deadline ? `due ${i.deadline.slice(5)}` : 'no deadline',
   382→   382→          i.estimateMin ? `${i.estimateMin}m` : '',
   383→   383→        ].filter(Boolean),
   384→   384→      }));
   385→   385→  }, [pop, popSearch, items, files]);
   386→   386→
   387→   387→  /* ---------- auto layout ---------- */
   388→   388→
   389→   389→  const autoLayout = () => {
   390→   390→    const order = topoOrder(roadmap.nodes, roadmap.edges);
   391→   391→    const depth = new Map<string, number>();
   392→   392→    roadmap.nodes.forEach((n) => depth.set(n.id, 0));
   393→   393→    order.forEach((id) => {
   394→   394→      roadmap.edges.forEach((e) => {
   395→   395→        if (e.from !== id) return;
   396→   396→        depth.set(e.to, Math.max(depth.get(e.to) ?? 0, (depth.get(id) ?? 0) + 1));
   397→   397→      });
   398→   398→    });
   399→   399→    const layers = new Map<number, RoadmapNode[]>();
   400→   400→    roadmap.nodes.forEach((n) => {
   401→   401→      const d = depth.get(n.id) ?? 0;
   402→   402→      const arr = layers.get(d) || [];
   403→   403→      arr.push(n);
   404→   404→      layers.set(d, arr);
   405→   405→    });
   406→   406→    layers.forEach((arr, d) => {
   407→   407→      arr.sort((a, b) => a.y - b.y);
   408→   408→      arr.forEach((n, i) => updateRoadmapNode(roadmap.id, n.id, { x: 60 + d * 290, y: 60 + i * 150 }));
   409→   409→    });
   410→   410→    setFitKey((k) => k + 1);
   411→   411→    toast('Auto-layout applied', 'Layers follow your connections.');
   412→   412→  };
   413→   413→
   414→   414→  /* ---------- order panel data ---------- */
   415→   415→
   416→   416→  const orderRows = useMemo(
   417→   417→    () => computeOrder(roadmap, liveMap),
   418→   418→    [roadmap, liveMap]
   419→   419→  );
   420→   420→
   421→   421→  /* ---------- render ---------- */
   422→   422→
   423→   423→  return (
   424→   424→    <div className="rm-screen fade-in">
   425→   425→      <div className="rm-toolbar">
   426→   426→        <Btn size="sm" onClick={() => setActiveRoadmap(null)} title="Back to all roadmaps">
   427→   427→          <ChevronLeft size={14} /> Roadmaps
   428→   428→        </Btn>
   429→   429→        <div className="rm-toolbar-sep" />
   430→   430→        <div className="rm-toolbar-titles">
   431→   431→          <input
   432→   432→            className="rm-title-input"
   433→   433→            value={roadmap.title}
   434→   434→            aria-label="Roadmap title"
   435→   435→            onChange={(e) => updateRoadmap(roadmap.id, { title: e.target.value })}
   436→   436→          />
   437→   437→          <input
   438→   438→            className="rm-desc-input"
   439→   439→            value={roadmap.description}
   440→   440→            placeholder="Add a short description…"
   441→   441→            aria-label="Roadmap description"
   442→   442→            onChange={(e) => updateRoadmap(roadmap.id, { description: e.target.value })}
   443→   443→          />
   444→   444→        </div>
   445→   445→
   446→   446→        {/* add node */}
   447→   447→        <div className="rm-pop-wrap">
   448→   448→          <Btn
   449→   449→            size="sm"
   450→   450→            variant="primary"
   451→   451→            onClick={() => setPop(pop ? null : { mode: 'menu' })}
   452→   452→            title="Add a node"
   453→   453→          >
   454→   454→            <Plus size={14} /> Add node

   791→   791→function EdgeLabelInput({ initial, onCommit }: { initial: string; onCommit: (v: string) => void }) {

   808→   808→function MiniForm({
   809→   809→  fields,
   810→   810→  submitLabel,
   811→   811→  onSubmit,
   812→   812→  onCancel,
   813→   813→}: {
   814→   814→  fields: { key: string; placeholder: string; type?: string; autoFocus?: boolean }[];
   815→   815→  submitLabel: string;
   816→   816→  onSubmit: (values: Record<string, string>) => void;
   817→   817→  onCancel: () => void;
   818→   818→}) {
   819→   819→  const [values, setValues] = useState<Record<string, string>>({});
   820→   820→  return (
   821→   821→    <div className="rm-mini-form">
   822→   822→      {fields.map((f) => (
   823→   823→        <input
   824→   824→          key={f.key}
   825→   825→          className="rm-mini-input"
   826→   826→          type={f.type || 'text'}
   827→   827→          autoFocus={f.autoFocus}
   828→   828→          placeholder={f.placeholder}
   829→   829→          value={values[f.key] || ''}
   830→   830→          onChange={(e) => setValues((v) => ({ ...v, [f.key]: e.target.value }))}
   831→   831→          onKeyDown={(e) => {
   832→   832→            if (e.key === 'Enter') onSubmit(values);
   833→   833→            if (e.key === 'Escape') onCancel();
   834→   834→          }}
   835→   835→        />
   836→   836→      ))}
   837→   837→      <div className="rm-mini-actions">
   838→   838→        <Btn size="sm" onClick={onCancel}>Cancel</Btn>
   839→   839→        <Btn size="sm" variant="primary" onClick={() => onSubmit(values)}>{submitLabel}</Btn>
   840→   840→      </div>
   841→   841→    </div>
   842→   842→  );
   843→   843→}
