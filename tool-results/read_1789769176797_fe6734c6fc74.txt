    60→function clampPage(p: number, total: number): number {
    61→  const max = Math.max(1, total);
    62→  const n = Math.round(p);
    63→  return Math.max(1, Math.min(max, Number.isFinite(n) ? n : 1));
    64→}

    66→function fmtRem(secs: number): string {
    67→  const s = Math.max(0, Math.floor(secs));
    68→  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`;
    69→}

    71→function themeFilterOf(mode: ReaderThemeMode, custom: { brightness: number; contrast: number; sepia: number }): string {
    72→  switch (mode) {
    73→    case 'dark':
    74→      return 'invert(1) hue-rotate(180deg) brightness(0.92) contrast(1.06)';
    75→    case 'warm':
    76→      return 'sepia(0.45) brightness(1.05)';
    77→    case 'custom':
    78→      return `brightness(${custom.brightness / 100}) contrast(${custom.contrast / 100}) sepia(${custom.sepia / 100})`;
    79→    default:
    80→      return 'none';
    81→  }
    82→}

   289→  }, [scheduleSave]);
   290→
   291→  const redo = useCallback(() => {
   292→    const cmd = redoRef.current.pop();

   294→    setAnnos((prev) => {
   295→      const next = applyAnnoCommand(prev, cmd);
   296→      annosRef.current = next;
   297→      return next;
   298→    });
   299→    undoRef.current.push(cmd);
   300→    dirtyRef.current = true;

   450→      });

   452→      alive = false;

   461→      setPageInput(String(target));
   462→      pageRef.current = target;
   463→      pageSecsRef.current = 0;
   464→    }
   465→  }, [doc, page]);
   466→
   467→  /* ---------- stage measurement (single stage OR strip scroller) ---------- */
   468→
   469→  useEffect(() => {
   470→    const el = viewMode === 'single' ? stageRef.current : stripScrollRef.current;
   471→    if (!el) return;
   472→    let timer: ReturnType<typeof setTimeout> | null = null;

   478→    const ro = new ResizeObserver(() => {
   479→      if (timer) clearTimeout(timer);
   480→      timer = setTimeout(measure, 140);
   481→    });
   482→    ro.observe(el);
   483→    measure();
   484→    return () => {
   485→      ro.disconnect();
   486→      if (timer) clearTimeout(timer);
   487→    };
   488→  }, [viewMode, loadState, pagesOpen]);
   489→
   490→  /* ---------- dominant page (scroll mode) ---------- */
   491→
   492→  const changeDominant = useCallback((next: number) => {
   493→    if (next === pageRef.current) return;
   494→    flushPending();
   495→    visitedRef.current.add(next);
   496→    pageRef.current = next;
   497→    pageSecsRef.current = 0;
   498→    setPage(next);
   499→    setPageInput(String(next));
