    91→export function nodeLive(node: RoadmapNode, items: Item[], files: ResourceFile[]): RmNodeLive {
    92→  const base: RmNodeLive = {
    93→    title: node.label,
    94→    missing: false,
    95→    status: 'todo',
    96→    pct: -1,
    97→    deadline: null,
    98→    deadlineTone: null,
    99→    estMin: 0,
   100→    done: false,
   101→    pagesRead: 0,
   102→    pagesTotal: 0,
   103→  };
   104→
   105→  if (node.kind === 'goal' || node.kind === 'task') {
   106→    const item = items.find((i) => i.id === node.refId);
   107→    if (!item) return { ...base, missing: true, title: node.label || 'Deleted item' };
   108→    const pct = computeProgress(item, (id) => wbChildren(items, id));
   109→    const blocked = wbIsBlocked(items, item);
   110→    const done = item.status === 'done' || (pct >= 100 && item.status !== 'inbox');
   111→    const status: RmNodeStatus = done
   112→      ? 'done'
   113→      : blocked || item.status === 'waiting'
   114→        ? 'blocked'
   115→        : item.status === 'in_progress'
   116→          ? 'progress'
   117→          : 'todo';
   118→    return {
   119→      ...base,
   120→      title: item.title,
   121→      status,
   122→      pct,
   123→      deadline: item.deadline,
   124→      deadlineTone: deadlineStatus(item, pct),
   125→      estMin: item.estimateMin || 30,
   126→      done,
   127→    };
   128→  }
   129→
   130→  if (node.kind === 'file') {
   131→    const f = files.find((x) => x.id === node.refId);
   132→    if (!f) return { ...base, missing: true, title: node.label || 'Deleted file' };
   133→    const pct = wbFileProgress(f);
   134→    const read = wbFileReadPages(f);
   135→    const remaining = Math.max(0, f.pageCount - read);
   136→    const avgSec = wbFileAvgSecPerPage(f);
   137→    const perPageMin = avgSec > 0 ? Math.max(1, Math.round(avgSec / 60)) : 1; // fallback: 1 min/page
   138→    const tone = wbFileDeadlineStatus(f);
   139→    const status: RmNodeStatus = pct >= 100 ? 'done' : tone === 'over' ? 'blocked' : tone === 'soon' ? 'progress' : 'todo';
   140→    return {
   141→      ...base,
   142→      title: f.title,
   143→      status,
   144→      pct,
   145→      deadline: f.deadline,
   146→      deadlineTone: tone,
   147→      estMin: remaining * perPageMin,
   148→      done: pct >= 100,
   149→      pagesRead: read,
   150→      pagesTotal: f.pageCount,
   151→    };
   152→  }
   153→
   154→  if (node.kind === 'milestone') {
   155→    const done = node.done;
   156→    let tone: RmNodeLive['deadlineTone'] = null;
   157→    if (node.milestoneDate) {
   158→      if (done) tone = 'done';
   159→      else {
   160→        const du = daysUntil(node.milestoneDate);
   161→        tone = du < 0 ? 'over' : du <= 3 ? 'soon' : 'ok';
   162→      }
   163→    }
   164→    return {
   165→      ...base,
   166→      title: node.label || 'Milestone',
   167→      status: done ? 'done' : 'todo',
   168→      pct: done ? 100 : -1,
   169→      deadline: node.milestoneDate,
   170→      deadlineTone: tone,
   171→      estMin: 0,
   172→      done,
   173→    };
   174→  }
   175→
   176→  // note
   177→  return { ...base, title: node.label || 'Note', status: 'none' };
   178→}

   180→/* ---------- topological order (Kahn, cycle-safe) ---------- */
   181→
   182→export function topoOrder(nodes: RoadmapNode[], edges: RoadmapEdge[]): string[] {
   183→  const ids = nodes.map((n) => n.id);
   184→  if (!ids.length) return [];
   185→  const idSet = new Set(ids);
   186→  const indeg = new Map<string, number>();
   187→  const adj = new Map<string, string[]>();
   188→  ids.forEach((id) => {
   189→    indeg.set(id, 0);
   190→    adj.set(id, []);
   191→  });
   192→  edges.forEach((e) => {
   193→    if (!idSet.has(e.from) || !idSet.has(e.to) || e.from === e.to) return;
   194→    indeg.set(e.to, (indeg.get(e.to) ?? 0) + 1);
   195→    adj.get(e.from)!.push(e.to);
   196→  });
   197→
   198→  const orderIndex = new Map(ids.map((id, i) => [id, i]));
   199→  const ready: string[] = [];
   200→  const pushReady = (id: string) => {
   201→    ready.push(id);
   202→    ready.sort((a, b) => (orderIndex.get(a) ?? 0) - (orderIndex.get(b) ?? 0));
   203→  };
   204→  ids.forEach((id) => {
   205→    if ((indeg.get(id) ?? 0) === 0) pushReady(id);
   206→  });
   207→
   208→  const out: string[] = [];
   209→  const emitted = new Set<string>();
   210→  while (ready.length) {
   211→    const id = ready.shift()!;
   212→    out.push(id);
   213→    emitted.add(id);
   214→    for (const t of adj.get(id) ?? []) {
   215→      const d = (indeg.get(t) ?? 0) - 1;
   216→      indeg.set(t, d);
   217→      if (d === 0) pushReady(t);
   218→    }
   219→  }
   220→
   221→  // cycle remains → append leftover nodes by x-position (never crash)
   222→  if (out.length < ids.length) {
   223→    const byId = new Map(nodes.map((n) => [n.id, n]));
   224→    const rest = ids.filter((id) => !emitted.has(id));
   225→    rest.sort((a, b) => (byId.get(a)?.x ?? 0) - (byId.get(b)?.x ?? 0));
   226→    out.push(...rest);
   227→  }
   228→  return out;
   229→}

   295→  const viewportRef = useRef(viewport);
   296→  const sizeRef = useRef(size);
   297→  const roadmapRef = useRef(roadmap);
   298→
   299→  /* keep refs in sync after render (compiler-safe) */
   300→  useEffect(() => { viewportRef.current = viewport; }, [viewport]);
   301→  useEffect(() => { sizeRef.current = size; }, [size]);
   302→  useEffect(() => { roadmapRef.current = roadmap; }, [roadmap]);
   303→
   304→  const dragInfoRef = useRef<{ id: string; px: number; py: number; ox: number; oy: number; moved: boolean } | null>(null);
   305→  const panInfoRef = useRef<{ px: number; py: number; x: number; y: number } | null>(null);

   315→
   316→  const nodeById = useMemo(() => new Map(roadmap.nodes.map((n) => [n.id, n])), [roadmap.nodes]);
   317→
   318→  /* selected / dragging / connecting node renders last (on top) */
   319→  const renderNodes = useMemo(() => {

   321→      ?? (selection?.kind === 'node' ? selection.id : null)
   322→      ?? portDrag?.hoverId
   323→      ?? portDrag?.fixedId;

   325→    const top: RoadmapNode[] = [];

   335→    const el = wrapRef.current;
   336→    if (!el) return;
   337→    const ro = new ResizeObserver((entries) => {
   338→      const cr = entries[0]?.contentRect;
   339→      if (cr) setSize({ w: cr.width, h: cr.height });

   342→    return () => ro.disconnect();

   411→      const wy = (h / 2 - v.y) / v.zoom;
   412→      return { x: w / 2 - wx * nz, y: h / 2 - wy * nz, zoom: nz };

   422→    const cx = (w / 2 - v.x) / v.zoom;
   423→    const cy = (h / 2 - v.y) / v.zoom;
   424→    return {
   425→      x: Math.round(cx + (n % 4) * 260 - RM_NODE_W / 2),
   426→      y: Math.round(cy + Math.floor(n / 4) * 140 - RM_NODE_H / 2),
   427→    };
   428→  }, []);
   429→
   430→  useImperativeHandle(ref, () => ({ fitView, centerNode, zoomBy, dropSpot }), [fitView, centerNode, zoomBy, dropSpot]);
   431→

   440→  }, [size.w, size.h, fitView]);

   458→

   464→      const n = nodes[i];

   467→        return n.id;
   468→      }
   469→    }
   470→    return null;

   473→  /* ---------- background pan + double-click quick-add ---------- */

   488→  };
